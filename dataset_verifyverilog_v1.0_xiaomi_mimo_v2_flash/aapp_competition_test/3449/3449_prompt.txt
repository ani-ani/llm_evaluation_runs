    input wire [2:0] height,     // H: ceiling height (1-5)

    input wire [5:0] addr,       // Address for 64 cells (8x8 grid)
    input wire wr,               // Write enable for strength memory
    output reg done,
    output reg [15:0] cost_out   // Total fencing cost
);

// Parameters

// Internal memory for strengths (8x8 grid)
reg [3:0] strength_ram [0:63];

// State machine states
localparam S_IDLE = 3'd0;
localparam S_COMPUTE_LIGHT = 3'd1;

// Fixed-point format: Q8.8 (8 integer bits, 8 fractional bits)
// Weight table: 1/(dr² + dc² + H²) for all possible values
// Max denominator: 7² + 7² + 5² = 49+49+25 = 123
// We store weights as Q8.8 integers (weight * 256)
reg [15:0] weight_table [0:123];  // Precomputed weights

// Helper: get row and column from index
function [5:0] get_row(input [5:0] idx);
    get_row = idx / GRID_C;
endfunction

function [5:0] get_col(input [5:0] idx);
    get_col = idx % GRID_C;
endfunction

// Helper: compute weight index from dr, dc, H
function [6:0] weight_index(input [3:0] dr, input [3:0] dc, input [2:0] H);
    reg [6:0] dr_sq, dc_sq, H_sq;
    begin
        dr_sq = dr * dr;
        dc_sq = dc * dc;
        H_sq = H * H;
        weight_index = dr_sq + dc_sq + H_sq;
    end
endfunction

// Main state machine
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= S_IDLE;
        done <= 0;
        cost_out <= 0;
        cell_idx <= 0;
        source_idx <= 0;
        light_sum <= 0;
        meets_std <= 0;
    end else begin
        case (state)
            S_IDLE: begin
                if (start) begin
                    state <= S_COMPUTE_LIGHT;
                    cell_idx <= 0;
                    source_idx <= 0;
                    light_sum <= 0;
                    meets_std <= 0;
                end
            end
            
            S_COMPUTE_LIGHT: begin
                if (cell_idx < 64) begin
                    if (source_idx < 64) begin
                        // Compute contribution from source_idx to cell_idx
                        // Skip if source_idx == cell_idx (handled separately)

                        end

                    end else begin
                        // After summing all sources, apply H² correction and compare
                        // Add contribution from self: strength / H²
                        // Then check threshold
                        // Store results
                        if (cell_idx < 64) begin
                            light_levels[cell_idx] <= light_sum;
                            if (light_sum >= {threshold, 8'h00}) // Convert to Q8.8
                                meets_std[cell_idx] <= 1;
                            cell_idx <= cell_idx + 1;
                            source_idx <= 0;
                            light_sum <= 0;
                        end else begin
                            state <= S_COMPUTE_FENCE;
                            cell_idx <= 0;
                            cost_out <= 0;
                        end
                    end
                end else begin
                    state <= S_COMPUTE_FENCE;
                    cell_idx <= 0;
                    cost_out <= 0;
                end
            end
            
            S_COMPUTE_FENCE: begin
                // Compute fencing cost for adjacent cells
                // Horizontal edges: for each cell, check right neighbor
                // Vertical edges: for each cell, check bottom neighbor
                // Skip border cells
                // This is complex but doable with nested loops

                if (cell_idx < 64) begin
                    // Process edges from this cell
                    // ... (combinational logic to add cost)
                    cell_idx <= cell_idx + 1;
                end else begin
                    state <= S_DONE;
                    done <= 1;
                end
            end
            
            S_DONE: begin
                // Hold done high for one cycle, then return to idle
                state <= S_IDLE;
                done <= 0;
            end
            
            default: state <= S_IDLE;
        endcase
    end
end

// Memory write logic
always @(posedge clk) begin
    if (wr) begin
        strength_ram[addr] <= data_in;
    end
