module bandwidth_allocator (
  input clk, rst_n, start,
  input [31:0] a0, a1, a2, a3,
  input [31:0] b0, b1, b2, b3,
  input [31:0] d0, d1, d2, d3,
  input [31:0] t,
  output reg [31:0] x0, x1, x2, x3,
  output reg done
);

// State encoding
localparam IDLE = 0;
localparam COMPUTE_TOTAL_D = 1;
localparam COMPUTE_Y = 2;
localparam INIT_X = 3;
localparam CHECK_ITERATION = 4;
localparam COMPUTE_RESIDUAL = 5;
localparam COMPUTE_TOTAL_Y_FREE = 6;
localparam UPDATE_X = 7;
localparam DONE = 8;

reg [3:0] state;
reg [1:0] index; // 0 to 3
reg [31:0] total_d;
reg [31:0] y [0:3];
reg [31:0] x [0:3];
reg [3:0] fixed; // one bit per species
reg [31:0] residual;
reg [31:0] total_y_free;
reg changed;

always @(posedge clk or negedge rst_n) begin
  if (!rst_n) begin
    state <= IDLE;
    done <= 0;
    fixed <= 0;
    changed <= 0;
    index <= 0;
    total_d <= 0;
    residual <= 0;
    total_y_free <= 0;
    x0 <= 0; x1 <= 0; x2 <= 0; x3 <= 0;
  end else begin
    case (state)
      IDLE: 
        if (start) begin
          state <= COMPUTE_TOTAL_D;
          index <= 0;
          total_d <= 0;
        end

      COMPUTE_TOTAL_D:
        begin
          case (index)
            0: total_d <= total_d + d0;
            1: total_d <= total_d + d1;
            2: total_d <= total_d + d2;
            3: total_d <= total_d + d3;
          endcase
          index <= index + 1;
          if (index == 3) begin
            state <= COMPUTE_Y;
            index <= 0;
          end
        end

      COMPUTE_Y:
        begin
          case (index)
            0: y[0] <= (t * d0) / total_d;
            1: y[1] <= (t * d1) / total_d;
            2: y[2] <= (t * d2) / total_d;
            3: y[3] <= (t * d3) / total_d;
          endcase
          index <= index + 1;
          if (index == 3) begin
            state <= INIT_X;
            index <= 0;
          end
        end

      INIT_X:
        begin
          case (index)
            0: begin
                 x[0] <= y[0];
                 if (y[0] < a0) begin
                   x[0] <= a0;
                   fixed[0] <= 1;
                 end else if (y[0] > b0) begin
                   x[0] <= b0;
                   fixed[0] <= 1;
                 end
               end
            1: begin
                 x[1] <= y[1];
                 if (y[1] < a1) begin
                   x[1] <= a1;
                   fixed[1] <= 1;
                 end else if (y[1] > b1) begin
                   x[1] <= b1;
                   fixed[1] <= 1;
                 end
               end
            2: begin
                 x[2] <= y[2];
                 if (y[2] < a2) begin
                   x[2] <= a2;
                   fixed[2] <= 1;
                 end else if (y[2] > b2) begin
                   x[2] <= b2;
                   fixed[2] <= 1;
                 end
               end
            3: begin
                 x[3] <= y[3];
                 if (y[3] < a3) begin
                   x[3] <= a3;
                   fixed[3] <= 1;
                 end else if (y[3] > b3) begin
                   x[3] <= b3;
                   fixed[3] <= 1;
                 end
               end
          endcase
          index <= index + 1;
          if (index == 3) begin
            state <= CHECK_ITERATION;
            index <= 0;
            changed <= 0;
          end
        end

      CHECK_ITERATION:
        begin
          if (fixed == 4'b1111) begin
            state <= DONE;
          end else begin
            state <= COMPUTE_RESIDUAL;
            index <= 0;
            residual <= t;
          end
        end

      COMPUTE_RESIDUAL:
        begin
          case (index)
            0: if (fixed[0]) residual <= residual - x[0];
            1: if (fixed[1]) residual <= residual - x[1];
            2: if (fixed[2]) residual <= residual - x[2];
            3: if (fixed[3]) residual <= residual - x[3];
          endcase
          index <= index + 1;
          if (index == 3) begin
            state <= COMPUTE_TOTAL_Y_FREE;
            index <= 0;
            total_y_free <= 0;
          end
        end

      COMPUTE_TOTAL_Y_FREE:
        begin
          case (index)
            0: if (!fixed[0]) total_y_free <= total_y_free + y[0];
            1: if (!fixed[1]) total_y_free <= total_y_free + y[1];
            2: if (!fixed[2]) total_y_free <= total_y_free + y[2];
            3: if (!fixed[3]) total_y_free <= total_y_free + y[3];
          endcase
          index <= index + 1;
          if (index == 3) begin
            state <= UPDATE_X;
            index <= 0;
          end
        end

      UPDATE_X:
        begin
          case (index)
            0: if (!fixed[0]) begin
                 x[0] <= (residual * y[0]) / total_y_free;
                 if (x[0] < a0) begin
                   x[0] <= a0;
                   fixed[0] <= 1;
                   changed <= 1;
                 end else if (x[0] > b0) begin
                   x[0] <= b0;
                   fixed[0] <= 1;
                   changed <= 1;
                 end
               end
            1: if (!fixed[1]) begin
                 x[1] <= (residual * y[1]) / total_y_free;
                 if (x[1] < a1) begin
                   x[1] <= a1;
                   fixed[1] <= 1;
                   changed <= 1;
                 end else if (x[1] > b1) begin
                   x[1] <= b1;
                   fixed[1] <= 1;
                   changed <= 1;
                 end
               end
            2: if (!fixed[2]) begin
                 x[2] <= (residual * y[2]) / total_y_free;
                 if (x[2] < a2) begin
                   x[2] <= a2;
                   fixed[2] <= 1;
                   changed <= 1;
                 end else if (x[2] > b2) begin
                   x[2] <= b2;
                   fixed[2] <= 1;
                   changed <= 1;
                 end
               end
            3: if (!fixed[3]) begin
                 x[3] <= (residual * y[3]) / total_y_free;
                 if (x[3] < a3) begin
                   x[3] <= a3;
                   fixed[3] <= 1;
                   changed <= 1;
                 end else if (x[3] > b3) begin
                   x[3] <= b3;
                   fixed[3] <= 1;
                   changed <= 1;
                 end
               end
          endcase
          index <= index + 1;
          if (index == 3) begin
            if (changed) begin
              state <= COMPUTE_RESIDUAL;
              index <= 0;
              residual <= t;
            end else begin
              state <= DONE;
            end
          end
        end

      DONE:
        begin
          done <= 1;
          x0 <= x[0];
          x1 <= x[1];
          x2 <= x[2];
          x3 <= x[3];
        end

    endcase
  end
end

endmodule
