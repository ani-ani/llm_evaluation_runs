module election #(
    parameter MAX_S = 8,
    parameter DATA_WIDTH = 32,
    parameter DELEGATE_WIDTH = 12
) (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] S,
    input wire [DELEGATE_WIDTH-1:0] D [MAX_S-1:0],
    input wire [DATA_WIDTH-1:0] C [MAX_S-1:0],
    input wire [DATA_WIDTH-1:0] F [MAX_S-1:0],
    input wire [DATA_WIDTH-1:0] U [MAX_S-1:0],
    output reg [DATA_WIDTH-1:0] result,
    output reg done
);
localparam INF = 32'h7FFFFFFF;
wire [DATA_WIDTH-1:0] cost_wire [MAX_S-1:0];
generate
for (genvar i=0;i<MAX_S;i=i+1) begin : gen_cost
assign cost_wire[i] = calc_cost(C[i],F[i],U[i]);
end
endgenerate
function automatic [DATA_WIDTH-1:0] calc_cost;
input [DATA_WIDTH-1:0] C_val,F_val,U_val;
reg [DATA_WIDTH:0] diff;
begin
if (C_val > F_val + U_val) calc_cost = 0;
else if (C_val + U_val <= F_val) calc_cost = INF;
else begin diff = F_val + U_val - C_val; calc_cost = (diff>>1)+1; end
end
endfunction
wire [DELEGATE_WIDTH-1:0] total_delegates_comb;
wire [DELEGATE_WIDTH-1:0] required_comb;
wire [7:0] max_mask_comb;
assign total_delegates_comb = ( (S>0?D[0]:0)+(S>1?D[1]:0)+(S>2?D[2]:0)+(S>3?D[3]:0)+(S>4?D[4]:0)+(S>5?D[5]:0)+(S>6?D[6]:0)+(S>7?D[7]:0) );
assign required_comb = (total_delegates_comb>>1)+1;
assign max_mask_comb = (8'b00000001<<S)-1;
reg [7:0] mask;
reg [2:0] i;
reg [DELEGATE_WIDTH-1:0] sum_delegates;
reg [DATA_WIDTH-1:0] sum_cost;
reg [DATA_WIDTH-1:0] min_cost;
reg [DELEGATE_WIDTH-1:0] total_delegates_latched;
reg [DELEGATE_WIDTH-1:0] required_latched;
reg [7:0] max_mask_latched;
reg [3:0] S_latched;
reg [DELEGATE_WIDTH-1:0] D_latched [MAX_S-1:0];
reg [DATA_WIDTH-1:0] cost_latched [MAX_S-1:0];
localparam S_IDLE=0,S_INIT=1,S_START_MASK=2,S_ADD_STATE=3,S_CHECK=4,S_DONE=5;
reg [2:0] state;
always @(posedge clk or negedge rst_n) begin
if (!rst_n) begin
state<=S_IDLE; done<=0; result<=0; mask<=0; i<=0; sum_delegates<=0; sum_cost<=0; min_cost<=INF;
total_delegates_latched<=0; required_latched<=0; max_mask_latched<=0; S_latched<=0;
end else begin
case (state)
S_IDLE: begin
done<=0;
if (start) begin
S_latched<=S;
total_delegates_latched<=total_delegates_comb;
required_latched<=required_comb;
max_mask_latched<=max_mask_comb;
if (S>0) begin D_latched[0]<=D[0]; cost_latched[0]<=cost_wire[0]; end else begin D_latched[0]<=0; cost_latched[0]<=INF; end
if (S>1) begin D_latched[1]<=D[1]; cost_latched[1]<=cost_wire[1]; end else begin D_latched[1]<=0; cost_latched[1]<=INF; end
if (S>2) begin D_latched[2]<=D[2]; cost_latched[2]<=cost_wire[2]; end else begin D_latched[2]<=0; cost_latched[2]<=INF; end
if (S>3) begin D_latched[3]<=D[3]; cost_latched[3]<=cost_wire[3]; end else begin D_latched[3]<=0; cost_latched[3]<=INF; end
if (S>4) begin D_latched[4]<=D[4]; cost_latched[4]<=cost_wire[4]; end else begin D_latched[4]<=0; cost_latched[4]<=INF; end
if (S>5) begin D_latched[5]<=D[5]; cost_latched[5]<=cost_wire[5]; end else begin D_latched[5]<=0; cost_latched[5]<=INF; end
if (S>6) begin D_latched[6]<=D[6]; cost_latched[6]<=cost_wire[6]; end else begin D_latched[6]<=0; cost_latched[6]<=INF; end
if (S>7) begin D_latched[7]<=D[7]; cost_latched[7]<=cost_wire[7]; end else begin D_latched[7]<=0; cost_latched[7]<=INF; end
mask<=0; min_cost<=INF; state<=S_START_MASK; end
end
S_START_MASK: begin i<=0; sum_delegates<=0; sum_cost<=0; state<=S_ADD_STATE; end
S_ADD_STATE: begin
if (i < S_latched) begin
if (mask[i]) begin sum_delegates <= sum_delegates + D_latched[i]; sum_cost <= sum_cost + cost_latched[i]; end
i <= i + 1; state <= S_ADD_STATE; end
else state <= S_CHECK; end
S_CHECK: begin
if (sum_delegates >= required_latched) begin if (sum_cost < min_cost) min_cost <= sum_cost; end
if (mask < max_mask_latched) begin mask <= mask + 1; state <= S_START_MASK; end
else state <= S_DONE; end
S_DONE: begin result <= min_cost; done <= 1; state <= S_IDLE; end
default: state <= S_IDLE;
endcase
end
end
endmodule
