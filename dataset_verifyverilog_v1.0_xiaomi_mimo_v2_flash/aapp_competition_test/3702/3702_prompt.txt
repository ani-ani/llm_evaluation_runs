module fib_seq_solver(
    input [19:0] n,  // Unused but included for interface compatibility
    input [19:0] a,
    input [19:0] d,
    output [63:0] b,
    output [63:0] e
);

    // Constants
    wire [63:0] CONST = 368131125;
    wire [63:0] MOD = 1000000000;
    wire [63:0] SCALE = 12000000000;

    // Compute b and e
    assign b = ((CONST * a) % MOD) * SCALE + 1;
    assign e = ((CONST * d) % MOD) * SCALE;

endmodule
