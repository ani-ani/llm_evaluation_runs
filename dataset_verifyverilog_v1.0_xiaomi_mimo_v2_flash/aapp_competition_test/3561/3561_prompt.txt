module wolf_sheep_cabbage (
    input [31:0] W,
    input [31:0] S,
    input [31:0] C,
    input [31:0] K,
    output result
);
    // Condition: YES if S == 0 OR (K >= S AND W + C <= 2*K)
    wire condition = (S == 0) || ((K >= S) && ((W + C) <= (K << 1)));
    assign result = condition;
endmodule
