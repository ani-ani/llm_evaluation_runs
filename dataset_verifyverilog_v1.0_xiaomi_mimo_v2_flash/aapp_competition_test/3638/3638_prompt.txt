module pattern_sorter #(
    parameter NUM_PATTERNS = 4,
    parameter PATTERN_LENGTH = 8,
    parameter DATA_WIDTH = 2,  // 2 bits: 00=R, 01=P, 10=S
    parameter SCORE_WIDTH = 8,  // Max score = 1+2+...+7 = 28
    parameter N_WIDTH = 20
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [N_WIDTH-1:0] n,  // Number of rounds (unused in border score)
    input wire [DATA_WIDTH*PATTERN_LENGTH-1:0] patterns_in [0:NUM_PATTERNS-1],
    output wire done,
    output wire [DATA_WIDTH*PATTERN_LENGTH-1:0] patterns_out [0:NUM_PATTERNS-1]
);

// Functionality:
// 1. On start=1, compute border score for each pattern
// 2. Score = sum of lengths of all proper borders (prefix = suffix)
// 3. Sort patterns by score ascending (stable for ties)
// 4. Assert done when complete
// 5. Computation takes NUM_PATTERNS * PATTERN_LENGTH cycles

// Implementation: Sequential state machine
// - State 0: Idle
// - State 1: Compute KMP prefix function for current pattern
// - State 2: Sum border lengths
// - State 3: Compare scores and sort
// - State 4: Done

endmodule
