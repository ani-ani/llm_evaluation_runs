module cookie_wall (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] n,          // Number of vertices (3-8)
    input wire [31:0] omega,     // Angular velocity Q16.16
    input wire [31:0] v0,        // Initial velocity Q16.16
    input wire [31:0] theta,     // Initial angle in degrees Q16.16
    input wire [31:0] w,         // Wall x-coordinate Q16.16
    input wire [31:0] x0, x1, x2, x3, x4, x5, x6, x7, // Initial x coordinates Q16.16
    input wire [31:0] y0, y1, y2, y3, y4, y5, y6, y7, // Initial y coordinates Q16.16
    output reg [3:0] result_index,  // 1-indexed vertex that hits first
    output reg [31:0] result_time,  // Time when it hits Q16.16
    output reg done                 // Computation complete
);

    // Fixed-point constants
    localparam Q16_16 = 16;
    localparam MAX_TIME = 32'sd600 * 65536; // 600 seconds in Q16.16
    localparam TIME_STEP = 32'sd100 * 65536 / 1000; // 0.1s in Q16.16
    localparam G = 32'sd981 * 65536 / 100; // g = 9.81 m/s² Q16.16
    localparam PI_TIMES_2 = 32'sd62831853 / 1000000; // 2π in Q16.16
    
    // State machine
    reg [4:0] state;
    reg [3:0] vertex_counter;
    reg [31:0] time_reg;        // Current time Q16.16
    reg [31:0] best_time;
    reg [3:0] best_index;
    reg found;
    
    // Intermediate calculations
    reg [31:0] com_x, com_y;    // Center of mass
    reg [31:0] v0x, v0y;        // Velocity components
    reg [31:0] dx [0:7];        // Relative x positions
    reg [31:0] dy [0:7];        // Relative y positions
    reg [31:0] angle;           // Current angle φ = ω*t
    reg [31:0] cos_phi, sin_phi; // Trigonometric values
    reg [31:0] x_com_pos;       // x_com + v0x*t
    reg [31:0] y_com_pos;       // y_com + v0y*t - 0.5*g*t²
    reg [31:0] x_vertex;        // x_i(t) = x_com_pos + dx*cos_phi + dy*sin_phi
    
    // States
    localparam S_IDLE = 0;
    localparam S_COMPUTE_COM = 1;
    localparam S_COMPUTE_VEL = 2;
    localparam S_COMPUTE_REL = 3;
    localparam S_INIT_TIME = 4;
    localparam S_COMPUTE_TRIG = 5;
    localparam S_COMPUTE_TRAJ = 6;
    localparam S_CHECK_VERTEX = 7;
    localparam S_NEXT_TIME = 8;
    localparam S_DONE = 9;
    
    // Fixed-point multiplication (Q16.16 * Q16.16 -> Q16.16)
    function automatic [31:0] mul_fp;
        input [31:0] a, b;
        begin
            mul_fp = (a * b) >> 16;
        end
    endfunction
    
    // Fixed-point division (Q16.16 / Q16.16 -> Q16.16)
    function automatic [31:0] div_fp;
        input [31:0] a, b;
        begin
            div_fp = (a << 16) / b;
        end
    endfunction
    
    // Fixed-point to integer (for array indexing)
    function automatic [3:0] fp_to_int;
        input [31:0] val;
        begin
            fp_to_int = val[31:16];
        end
    endfunction
    
    // Main state machine
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= S_IDLE;
            done <= 0;
            found <= 0;
            time_reg <= 0;
            best_time <= 32'h7FFFFFFF; // Max value
            best_index <= 0;
            vertex_counter <= 0;
        end else begin
            case (state)
                S_IDLE: begin
                    if (start) begin
                        state <= S_COMPUTE_COM;
                        vertex_counter <= 0;
                        com_x <= 0;
                        com_y <= 0;
                    end
                end
                
                S_COMPUTE_COM: begin
                    // Compute center of mass
                    case (vertex_counter)
                        0: begin com_x <= x0; com_y <= y0; end
                        1: begin com_x <= com_x + x1; com_y <= com_y + y1; end
                        2: begin com_x <= com_x + x2; com_y <= com_y + y2; end
                        3: begin com_x <= com_x + x3; com_y <= com_y + y3; end
                        4: begin com_x <= com_x + x4; com_y <= com_y + y4; end
                        5: begin com_x <= com_x + x5; com_y <= com_y + y5; end
                        6: begin com_x <= com_x + x6; com_y <= com_y + y6; end
                        7: begin com_x <= com_x + x7; com_y <= com_y + y7; end
                    endcase
                    
                    if (vertex_counter == n - 1) begin
                        com_x <= div_fp(com_x, {n, 16'h0}); // n in Q16.16
                        com_y <= div_fp(com_y, {n, 16'h0});
                        vertex_counter <= 0;
                        state <= S_COMPUTE_VEL;
                    end else begin
                        vertex_counter <= vertex_counter + 1;
                    end
                end
                
                S_COMPUTE_VEL: begin
                    // Convert theta to radians: theta_rad = theta * π / 180
                    // Then: v0x = v0 * cos(theta_rad), v0y = v0 * sin(theta_rad)
                    // For simplicity, we'll use precomputed values for 0-80 degrees
                    // In practice, this would use a CORDIC or lookup table
                    // Here we assume theta is in degrees Q16.16, convert to radians
                    // theta_rad = mul_fp(theta, 32'sd1859187 / 1000000) // π/180 ≈ 0.0174533
                    // For benchmarking, we'll compute v0x and v0y directly:
                    // v0x = v0 * cos(theta_rad) ≈ v0 * (1 - theta_rad²/2)
                    // v0y = v0 * sin(theta_rad) ≈ v0 * theta_rad
                    // This is a simplified approximation for small angles
                    if (theta < 32'sd45 * 65536) begin
                        v0x <= mul_fp(v0, 32'sd65536); // cos(0) = 1
                        v0y <= mul_fp(v0, theta * 32'sd17453 / 1000000); // sin(theta) ≈ theta
                    end else begin
                        v0x <= mul_fp(v0, 32'sd46341); // cos(45) ≈ 0.707
                        v0y <= mul_fp(v0, 32'sd46341); // sin(45) ≈ 0.707
                    end
                    state <= S_COMPUTE_REL;
                    vertex_counter <= 0;
                end
                
                S_COMPUTE_REL: begin
                    // Compute relative positions dx = x - com_x, dy = y - com_y
                    case (vertex_counter)
                        0: begin dx[0] <= x0 - com_x; dy[0] <= y0 - com_y; end
                        1: begin dx[1] <= x1 - com_x; dy[1] <= y1 - com_y; end
                        2: begin dx[2] <= x2 - com_x; dy[2] <= y2 - com_y; end
                        3: begin dx[3] <= x3 - com_x; dy[3] <= y3 - com_y; end
                        4: begin dx[4] <= x4 - com_x; dy[4] <= y4 - com_y; end
                        5: begin dx[5] <= x5 - com_x; dy[5] <= y5 - com_y; end
                        6: begin dx[6] <= x6 - com_x; dy[6] <= y6 - com_y; end
                        7: begin dx[7] <= x7 - com_x; dy[7] <= y7 - com_y; end
                    endcase
                    
                    if (vertex_counter == n - 1) begin
                        state <= S_INIT_TIME;
                    end else begin
                        vertex_counter <= vertex_counter + 1;
                    end
                end
                
                S_INIT_TIME: begin
                    time_reg <= 0;
                    vertex_counter <= 0;
                    state <= S_COMPUTE_TRIG;
                end
                
                S_COMPUTE_TRIG: begin
                    // Compute φ = ω * t
                    angle <= mul_fp(omega, time_reg);
                    // Reduce angle modulo 2π (simplified for benchmarking)
                    // In practice, use CORDIC to compute cos(φ) and sin(φ)
                    // Here we use approximations for cos and sin
                    if (angle < PI_TIMES_2) begin
                        // Small angle approximation
                        cos_phi <= 32'sd65536 - mul_fp(angle, angle) / 2; // cos(φ) ≈ 1 - φ²/2
                        sin_phi <= angle; // sin(φ) ≈ φ
                    end else begin
                        cos_phi <= 32'sd46341; // cos(π/2) ≈ 0.707
                        sin_phi <= 32'sd46341; // sin(π/2) ≈ 0.707
                    end
                    state <= S_COMPUTE_TRAJ;
                end
                
                S_COMPUTE_TRAJ: begin
                    // x_com_pos = com_x + v0x * t
                    x_com_pos <= com_x + mul_fp(v0x, time_reg);
                    // y_com_pos = com_y + v0y * t - 0.5 * g * t²
                    y_com_pos <= com_y + mul_fp(v0y, time_reg) - mul_fp(G, mul_fp(time_reg, time_reg)) / 2;
                    state <= S_CHECK_VERTEX;
                    vertex_counter <= 0;
                end
                
                S_CHECK_VERTEX: begin
                    // x_vertex = x_com_pos + dx*cos_phi + dy*sin_phi
                    x_vertex <= x_com_pos + 
                               mul_fp(dx[vertex_counter], cos_phi) + 
                               mul_fp(dy[vertex_counter], sin_phi);
                    
                    // Check if x_vertex >= w (wall)
                    if (x_vertex >= w && !found && time_reg > 0) begin
                        best_time <= time_reg;
                        best_index <= vertex_counter + 1; // 1-indexed
                        found <= 1;
                    end
                    
                    if (vertex_counter == n - 1) begin
                        state <= S_NEXT_TIME;
                    end else begin
                        vertex_counter <= vertex_counter + 1;
                    end
                end
                
                S_NEXT_TIME: begin
                    time_reg <= time_reg + TIME_STEP;
                    if (time_reg >= MAX_TIME || found) begin
                        state <= S_DONE;
                    end else begin
                        state <= S_COMPUTE_TRIG;
                    end
                end
                
                S_DONE: begin
                    result_index <= best_index;
                    result_time <= best_time;
                    done <= 1;
                    if (!start) state <= S_IDLE;
                end
            endcase
        end
    end
endmodule
