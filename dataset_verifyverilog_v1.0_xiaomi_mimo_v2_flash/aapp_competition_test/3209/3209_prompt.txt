module TrainPathPlanner (
    input clk,
    input rst_n,
    input start,
    input [5:0] origin,      // 6-bit node ID (0-63)
    input [5:0] destination, // 6-bit node ID (0-63)
    input [3:0] num_edges,   // Number of edges (1-16)
    input [5:0] edge_from [15:0],  // From node IDs
    input [5:0] edge_to [15:0],    // To node IDs  
    input [7:0] edge_dep [15:0],   // Departure time (0-59)
    input [15:0] edge_time [15:0], // Standard time in Q8.8
    input [15:0] edge_p [15:0],    // Probability (0-100) in Q8.8
    input [15:0] edge_d [15:0],    // Max delay in Q8.8
    output reg [15:0] result,       // Expected duration in Q8.8
    output reg done,
    output reg impossible
);

// Parameters
parameter MAX_NODES = 8;
parameter MAX_EDGES = 16;
parameter ITERATIONS = 8;

// Internal state
reg [5:0] node_state [MAX_NODES-1:0]; // 0=unvisited, 1=visiting, 2=done
reg [15:0] expected_time [MAX_NODES-1:0]; // Q8.8
reg [5:0] current_node;
reg [3:0] edge_counter;
reg [2:0] iteration_counter;

// Temporary registers
reg [15:0] new_time;
reg [15:0] wait_time;
reg [15:0] base_time;
reg [15:0] delay_expected;
reg [15:0] prob_factor;
reg [15:0] no_delay_prob;
reg [15:0] delay_prob;
reg [15:0] sum_delay;
reg [4:0] delay_iter;

// State machine states
reg [3:0] state;
localparam IDLE = 0;
localparam INIT = 1;
localparam SELECT_NODE = 2;
localparam PROCESS_EDGES = 3;
localparam UPDATE_NODE = 4;
localparam CHECK_DONE = 5;
localparam FINISHED = 6;
localparam IMPOSSIBLE_STATE = 7;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        impossible <= 0;
        result <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    state <= INIT;
                    done <= 0;
                    impossible <= 0;
                end
            end
            
            INIT: begin
                // Initialize node states and expected times
                for (integer i = 0; i < MAX_NODES; i = i + 1) begin
                    node_state[i] <= (i == origin) ? 0 : 0; // All unvisited
                    expected_time[i] <= 16'h7FFF; // Max value
                end
                expected_time[origin] <= 0;
                iteration_counter <= 0;
                state <= SELECT_NODE;
            end
            
            SELECT_NODE: begin
                // Find unvisited node with minimum expected time
                current_node <= 6'h3F; // Invalid initially
                for (integer i = 0; i < MAX_NODES; i = i + 1) begin
                    if (node_state[i] == 0) begin
                        if (current_node == 6'h3F || expected_time[i] < expected_time[current_node]) begin
                            current_node <= i;
                        end
                    end
                end
                
                if (current_node == 6'h3F) begin
                    // No more nodes to visit
                    if (expected_time[destination] < 16'h7FFF) begin
                        state <= FINISHED;
                    end else begin
                        state <= IMPOSSIBLE_STATE;
                    end
                end else begin
                    edge_counter <= 0;
                    state <= PROCESS_EDGES;
                end
            end
            
            PROCESS_EDGES: begin
                if (edge_counter >= num_edges) begin
                    node_state[current_node] <= 2; // Mark as done
                    state <= SELECT_NODE;
                end else if (edge_from[edge_counter] == current_node) begin
                    // Calculate expected time for this edge
                    // wait_time = (departure - current_time) mod 60
                    // But we don't track current time modulo 60 in this simplified version
                    // Instead, we assume immediate departure for simplicity
                    wait_time <= 0;
                    
                    // base_time = standard_time
                    base_time <= edge_time[edge_counter];
                    
                    // delay_expected = (p/100) * (d/2) in Q8.8
                    // p in [0,100], d in [1,120] but scaled to Q8.8
                    delay_expected <= (edge_p[edge_counter] * edge_d[edge_counter]) >> 9; // (p*d/2)/100
                    
                    // no_delay_prob = 1 - p/100
                    no_delay_prob <= 16'h0100 - edge_p[edge_counter]; // 1.0 = 0x0100
                    
                    // delay_prob = p/100
                    delay_prob <= edge_p[edge_counter];
                    
                    // Calculate expected future cost with delays
                    // This is simplified: we assume the next node's expected time
                    // doesn't change significantly with arrival time variations
                    sum_delay <= 0;
                    delay_iter <= 0;
                    state <= UPDATE_NODE;
                end else begin
                    edge_counter <= edge_counter + 1;
                end
            end
            
            UPDATE_NODE: begin
                // For simplicity, we use a weighted average of no-delay and delay cases
                // In full implementation, we'd need to consider different arrival times
                new_time <= expected_time[current_node] + base_time + delay_expected;
                
                // Update destination node if better
                if (edge_to[edge_counter] < MAX_NODES && new_time < expected_time[edge_to[edge_counter]]) begin
                    expected_time[edge_to[edge_counter]] <= new_time;
                end
                
                edge_counter <= edge_counter + 1;
                state <= PROCESS_EDGES;
            end
            
            FINISHED: begin
                result <= expected_time[destination];
                done <= 1;
                state <= IDLE;
            end
            
            IMPOSSIBLE_STATE: begin
                impossible <= 1;
                done <= 1;
                state <= IDLE;
            end
            
            default: state <= IDLE;
        endcase
    end
end

endmodule
