Verilog module specification:

module subset_sum_mask (
    input clk, rst_n,
    input start,
    input [3:0] n,        // Number of coins (1-8)
    input [4:0] k,        // Target sum (1-16)
    input [7:0] coin_in,  // Coin value (8-bit)
    output done,
    output [16:0] result_mask  // Bitmask of possible x values
);

parameter MAX_N = 8;
parameter MAX_K = 16;

// State encoding
localparam IDLE = 2'b00;
localparam PROCESS = 2'b01;
localparam DONE = 2'b10;

reg [1:0] state, next_state;
reg [3:0] counter, next_counter;
reg [16:0] dp [0:MAX_K];  // dp[s] = bitmask of subset sums for total sum s
reg [4:0] k_reg;
reg [3:0] n_reg;

integer s;

// State and counter update
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        counter <= 0;
        k_reg <= 0;
        n_reg <= 0;
        // Reset dp: only dp[0] = 1 (subset sum 0 possible with total sum 0)
        for (s = 0; s <= MAX_K; s = s + 1) begin
            dp[s] <= 0;
        end
        dp[0] <= 1;
    end else begin
        state <= next_state;
        counter <= next_counter;
        if (state == IDLE && start) begin
            k_reg <= k;
            n_reg <= n;
        end
        // Process coin in PROCESS state
        if (state == PROCESS) begin
            for (s = 0; s <= MAX_K; s = s + 1) begin
                if (s < coin_in) begin
                    dp[s] <= dp[s];
                end else begin
                    // Update: existing subsets + subsets without coin + subsets with coin
                    dp[s] <= dp[s] | dp[s-coin_in] | (dp[s-coin_in] << coin_in);
                end
            end
        end
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    next_counter = counter;
    case (state)
        IDLE: 
            if (start) begin
                next_state = PROCESS;
                next_counter = 0;
            end
        PROCESS: 
            begin
                next_counter = counter + 1;
                if (counter + 1 == n_reg) begin
                    next_state = DONE;
                end
            end
        DONE: 
            next_state = DONE;
    endcase
end

assign done = (state == DONE);
assign result_mask = (state == DONE) ? dp[k_reg] : 0;

endmodule
