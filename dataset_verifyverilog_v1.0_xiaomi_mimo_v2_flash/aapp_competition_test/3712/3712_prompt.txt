module student_swap(
    input clk,
    input rst_n,
    input start,
    input [3:0] arr0, arr1, arr2, arr3, arr4, arr5, arr6, arr7,
    output reg [15:0] result,
    output reg done
);

// State encoding
localparam IDLE = 3'b000;
localparam COUNT = 3'b001;
localparam CALC1 = 3'b010;
localparam CALC2 = 3'b011;
localparam DONE = 3'b100;

reg [2:0] state;
reg [2:0] index;
reg [3:0] cnt0, cnt1, cnt2, cnt3, cnt4;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
        index <= 0;
        cnt0 <= 0; cnt1 <= 0; cnt2 <= 0; cnt3 <= 0; cnt4 <= 0;
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    index <= 0;
                    cnt0 <= 0; cnt1 <= 0; cnt2 <= 0; cnt3 <= 0; cnt4 <= 0;
                    state <= COUNT;
                end
            end

            COUNT: begin
                case (index)
                    0: case(arr0) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    1: case(arr1) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    2: case(arr2) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    3: case(arr3) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    4: case(arr4) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    5: case(arr5) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    6: case(arr6) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                    7: begin
                        case(arr7) 0: cnt0<=cnt0+1; 1: cnt1<=cnt1+1; 2: cnt2<=cnt2+1; 3: cnt3<=cnt3+1; 4: cnt4<=cnt4+1; endcase
                        state <= CALC1;
                    end
                endcase
                index <= index + 1;
            end

            CALC1: begin
                reg [7:0] total;
                total = cnt1 + (cnt2<<1) + (cnt3*3) + (cnt4<<2);
                if (total < 3 || total == 5) begin
                    result <= 16'hFFFF;
                    state <= DONE;
                end else begin
                    state <= CALC2;
                end
            end

            CALC2: begin
                reg [15:0] ans;
                reg [3:0] t1, t2, t3, groups, rem;
                if (cnt1 >= cnt2) begin
                    ans = cnt2;
                    t1 = cnt1 - cnt2;
                    t3 = cnt3 + cnt2;
                    groups = t1 / 3;
                    ans = ans + (groups << 1);
                    t3 = t3 + groups;
                    rem = t1 % 3;
                    if (rem == 0) result <= ans;
                    else if (t3 > 0) result <= ans + rem;
                    else result <= ans + 2;
                end else begin
                    ans = cnt1;
                    t2 = cnt2 - cnt1;
                    groups = t2 / 3;
                    ans = ans + (groups << 1);
                    rem = t2 % 3;
                    if (rem == 0) result <= ans;
                    else if (cnt4 > 0) result <= ans + rem;
                    else result <= ans + 2;
                end
                state <= DONE;
            end

            DONE: begin
                done <= 1;
            end
        endcase
    end
end
endmodule
