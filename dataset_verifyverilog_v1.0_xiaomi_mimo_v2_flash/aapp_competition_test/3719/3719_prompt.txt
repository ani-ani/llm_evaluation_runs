module spaceship_destroyer (
    input clk, rst_n, start,
    input signed [7:0] group1_0, group1_1,
    input signed [7:0] group2_0, group2_1,
    input [1:0] n, m,
    output reg [7:0] result,
    output reg done
);

// States
parameter IDLE = 2'b00;
parameter PREPARE_SUMS = 2'b01;
parameter ITERATE = 2'b10;
parameter DONE = 2'b11;

// Registers
reg signed [7:0] g1 [0:1];
reg signed [7:0] g2 [0:1];
reg signed [7:0] sums [0:3];
reg [1:0] i, j;
reg [7:0] best;
reg [1:0] state;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
    end else begin
        case (state)
            IDLE: if (start) begin
                g1[0] <= group1_0;
                g1[1] <= group1_1;
                g2[0] <= group2_0;
                g2[1] <= group2_1;
                state <= PREPARE_SUMS;
                done <= 0;
            end

            PREPARE_SUMS: begin
                // Only compute valid sums based on n and m
                sums[0] <= (n >= 1 && m >= 1) ? g1[0] + g2[0] : 8'sd0;
                sums[1] <= (n >= 1 && m >= 2) ? g1[0] + g2[1] : 8'sd0;
                sums[2] <= (n >= 2 && m >= 1) ? g1[1] + g2[0] : 8'sd0;
                sums[3] <= (n >= 2 && m >= 2) ? g1[1] + g2[1] : 8'sd0;
                i <= 0;
                j <= 0;
                best <= 0;
                state <= ITERATE;
            end

            ITERATE: begin
                // Compute destruction for current pair (i,j)
                // Group1 ship0
                bit0_group1 = (n >= 1) && (
                    (sums[i] - g1[0] == g2[0]) ||
                    (sums[i] - g1[0] == g2[1]) ||
                    (sums[j] - g1[0] == g2[0]) ||
                    (sums[j] - g1[0] == g2[1])
                );
                // Group1 ship1
                bit1_group1 = (n >= 2) && (
                    (sums[i] - g1[1] == g2[0]) ||
                    (sums[i] - g1[1] == g2[1]) ||
                    (sums[j] - g1[1] == g2[0]) ||
                    (sums[j] - g1[1] == g2[1])
                );
                // Group2 ship0
                bit0_group2 = (m >= 1) && (
                    (sums[i] - g2[0] == g1[0]) ||
                    (sums[i] - g2[0] == g1[1]) ||
                    (sums[j] - g2[0] == g1[0]) ||
                    (sums[j] - g2[0] == g1[1])
                );
                // Group2 ship1
                bit1_group2 = (m >= 2) && (
                    (sums[i] - g2[1] == g1[0]) ||
                    (sums[i] - g2[1] == g1[1]) ||
                    (sums[j] - g2[1] == g1[0]) ||
                    (sums[j] - g2[1] == g1[1])
                );
                
                total = bit0_group1 + bit1_group1 + bit0_group2 + bit1_group2;
                if (total > best) best <= total;
                
                // Update indices
                if (j < 3) begin
                    j <= j + 1;
                end else begin
                    j <= 0;
                    if (i < 3) begin
                        i <= i + 1;
                    end else begin
                        state <= DONE;
                    end
                end
            end

            DONE: begin
                result <= best;
                done <= 1;
            end
        endcase
    end
end
endmodule
