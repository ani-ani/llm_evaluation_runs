module bus_time_calculator (
    input wire clk,
    input wire rst_n,
    input wire start,
    
    // All values in Q16.16 fixed-point format
    input wire [31:0] n,   // Number of pupils (integer part only)
    input wire [31:0] l,   // Distance
    input wire [31:0] v1,  // Walking speed
    input wire [31:0] v2,  // Bus speed
    input wire [31:0] k,   // Bus capacity (integer part only)
    
    output reg [31:0] time,
    output reg done
);

// Internal state
reg [2:0] state;
reg [7:0] iter;
reg [31:0] m, L, R, M, S, T;

// Fixed-point multiplication (Q16.16 * Q16.16)
function [31:0] q_mul;
    input [31:0] a, b;
    begin
        q_mul = (a * b) >> 16;
    end
endfunction

localparam IDLE = 0, INIT = 1, LOOP = 2, CALC = 3, UPDATE = 4, DONE = 5;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        time <= 0;
    end else begin
        case (state)
            IDLE: if (start) state <= INIT;
            
            INIT: begin
                m <= (n + k - 1) / k;  // ceil(n/k)
                L <= 0;
                R <= l;
                iter <= 0;
                state <= LOOP;
            end
            
            LOOP: begin
                if (iter >= 16) begin
                    time <= M;
                    state <= DONE;
                end else begin
                    M <= (L + R) >> 1;  // Binary search midpoint
                    state <= CALC;
                end
            end
            
            CALC: begin
                S <= l - M;
                T <= q_mul(M, (m << 1) - 1) - l;
                state <= UPDATE;
            end
            
            UPDATE: begin
                if (q_mul(T, v1) > q_mul(S, v2)) begin
                    R <= M;
                end else begin
                    L <= M;
                end
                iter <= iter + 1;
                state <= LOOP;
            end
            
            DONE: begin
                done <= 1;
                state <= IDLE;
            end
            
            default: state <= IDLE;
        endcase
    end
end

endmodule
