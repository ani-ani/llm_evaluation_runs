module bug_fixing_optimizer #(
    parameter MAX_BUGS = 2,
    parameter MAX_TIME = 8,
    parameter STATE_SIZE = (MAX_TIME+2) ** MAX_BUGS,
    parameter DATA_WIDTH = 64,
    parameter FRACTION_WIDTH = 32
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [1:0] B,
    input wire [3:0] T,
    input wire [DATA_WIDTH-1:0] f,
    input wire [DATA_WIDTH-1:0] p [0:MAX_BUGS-1],
    input wire [DATA_WIDTH-1:0] s [0:MAX_BUGS-1],
    output reg [DATA_WIDTH-1:0] expected_value,
    output reg done
);

// State definitions
localparam S_IDLE = 0;
localparam S_PRECOMP = 1;
localparam S_INIT = 2;
localparam S_LOOP_T = 3;
localparam S_LOOP_STATE = 4;
localparam S_LOOP_BUG = 5;
localparam S_CALC = 6;
localparam S_UPDATE_BEST = 7;
localparam S_STORE = 8;
localparam S_OUTPUT = 9;
localparam S_DONE = 10;

reg [3:0] state, next_state;

// DP memory
reg [DATA_WIDTH-1:0] dp [0:MAX_TIME][0:STATE_SIZE-1];

// Precomputed powers
reg [DATA_WIDTH-1:0] f_power [0:MAX_TIME];

// Iteration variables
reg [3:0] t_idx;
reg [6:0] state_idx;
reg [1:0] bug_idx;

// Temporary registers
reg [DATA_WIDTH-1:0] p_current;
reg [DATA_WIDTH-1:0] one_minus_p;
reg [DATA_WIDTH-1:0] dp_success;
reg [DATA_WIDTH-1:0] dp_failure;
reg [DATA_WIDTH-1:0] term_success;
reg [DATA_WIDTH-1:0] term_failure;
reg [DATA_WIDTH-1:0] value;
reg [DATA_WIDTH-1:0] best;

// State decoding
wire [3:0] bug0_state = state_idx % (T+2);
wire [3:0] bug1_state = state_idx / (T+2);
wire bug0_fixed = (bug0_state == T+1);
wire bug1_fixed = (bug1_state == T+1);

wire [3:0] bug0_state_success = T+1;
wire [3:0] bug0_state_failure = (bug0_state < T) ? bug0_state + 1 : T;
wire [3:0] bug1_state_success = T+1;
wire [3:0] bug1_state_failure = (bug1_state < T) ? bug1_state + 1 : T;

wire [6:0] state_success_bug0 = bug0_state_success * (T+2) + bug1_state;
wire [6:0] state_failure_bug0 = bug0_state_failure * (T+2) + bug1_state;
wire [6:0] state_success_bug1 = bug0_state * (T+2) + bug1_state_success;
wire [6:0] state_failure_bug1 = bug0_state * (T+2) + bug1_state_failure;

// Fixed-point arithmetic functions
function [DATA_WIDTH-1:0] fp_mult;
    input [DATA_WIDTH-1:0] a, b;
    reg signed [127:0] prod;
    begin
        prod = a * b;
        fp_mult = prod[95:32];
    end
endfunction

function [DATA_WIDTH-1:0] fp_add;
    input [DATA_WIDTH-1:0] a, b;
    begin
        fp_add = a + b;
    end
endfunction

function [DATA_WIDTH-1:0] fp_sub;
    input [DATA_WIDTH-1:0] a, b;
    begin
        fp_sub = a - b;
    end
endfunction

// Main state machine
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= S_IDLE;
        done <= 0;
        expected_value <= 0;
    end else begin
        case (state)
            S_IDLE: begin
                if (start) state <= S_PRECOMP;
            end

            S_PRECOMP: begin
                // Compute f_power[k] for k=0..T sequentially
                // Assume done by combinational logic or separate state
                // For simplicity, we move to next state
                state <= S_INIT;
            end

            S_INIT: begin
                // Initialize dp[T][*] = 0
                for (integer i = 0; i < STATE_SIZE; i = i + 1) dp[T][i] <= 0;
                t_idx <= T - 1;
                state_idx <= 0;
                state <= S_LOOP_T;
            end

            S_LOOP_T: begin
                if (t_idx == 8'hFF) state <= S_OUTPUT; // t_idx underflows when done
                else state <= S_LOOP_STATE;
            end

            S_LOOP_STATE: begin
                if (state_idx >= STATE_SIZE) begin
                    t_idx <= t_idx - 1;
                    state_idx <= 0;
                    state <= S_LOOP_T;
                end else begin
                    bug_idx <= 0;
                    best <= 0;
                    state <= S_LOOP_BUG;
                end
            end

            S_LOOP_BUG: begin
                if (bug_idx >= MAX_BUGS) begin
                    dp[t_idx][state_idx] <= best;
                    state_idx <= state_idx + 1;
                    state <= S_LOOP_STATE;
                end else begin
                    if ((bug_idx == 0 && bug0_fixed) || (bug_idx == 1 && bug1_fixed)) begin
                        bug_idx <= bug_idx + 1;
                    end else begin
                        state <= S_CALC;
                    end
                end
            end

            S_CALC: begin
                // Combinational logic calculates value
                // Move to update state
                state <= S_UPDATE_BEST;
            end

            S_UPDATE_BEST: begin
                if (value > best) best <= value;
                bug_idx <= bug_idx + 1;
                state <= S_LOOP_BUG;
            end

            S_OUTPUT: begin
                expected_value <= dp[0][0];
                done <= 1;
                state <= S_DONE;
            end

            S_DONE: begin
                done <= 0;
                state <= S_IDLE;
            end

            default: state <= S_IDLE;
        endcase
    end
end

// Combinational logic for S_CALC
always @(*) begin
    value = 0;
    p_current = 0;
    one_minus_p = 0;
    dp_success = 0;
    dp_failure = 0;
    term_success = 0;
    term_failure = 0;

    if (state == S_CALC) begin
        if (bug_idx == 0) begin
            p_current = fp_mult(p[0], f_power[bug0_state]);
            one_minus_p = fp_sub(64'h0000000100000000, p_current);
            dp_success = dp[t_idx+1][state_success_bug0];
            dp_failure = dp[t_idx+1][state_failure_bug0];
        end else begin
            p_current = fp_mult(p[1], f_power[bug1_state]);
            one_minus_p = fp_sub(64'h0000000100000000, p_current);
            dp_success = dp[t_idx+1][state_success_bug1];
            dp_failure = dp[t_idx+1][state_failure_bug1];
        end
        term_success = fp_mult(p_current, fp_add(s[bug_idx], dp_success));
        term_failure = fp_mult(one_minus_p, dp_failure);
        value = fp_add(term_success, term_failure);
    end
end

endmodule
