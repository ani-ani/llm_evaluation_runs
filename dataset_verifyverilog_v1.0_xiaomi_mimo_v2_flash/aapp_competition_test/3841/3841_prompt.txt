module find_polynomial (
    input clk,
    input rst_n,
    input start,
    input [59:0] p_in,
    input [11:0] k_in,
    output reg [11:0] coeff,
    output reg valid,
    output reg done
);

// States
localparam IDLE = 0;
localparam COMPUTE = 1;

reg state;
reg signed [59:0] p_reg;
reg signed [11:0] k_reg;

// Combinational quotient and remainder
wire signed [59:0] quotient;
wire signed [11:0] remainder;

assign quotient = p_reg / k_reg;
assign remainder = p_reg % k_reg;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        p_reg <= 0;
        k_reg <= 0;
        coeff <= 0;
        valid <= 0;
        done <= 0;
    end else begin
        case (state)
            IDLE: 
                if (start) begin
                    p_reg <= $signed(p_in);
                    k_reg <= $signed(k_in);
                    done <= 0;
                    valid <= 0;
                    state <= COMPUTE;
                end

            COMPUTE: 
                if (p_reg != 0) begin
                    if (remainder < 0) begin
                        coeff <= $unsigned(remainder + k_reg);
                        p_reg <= - (quotient + 1);
                    end else begin
                        coeff <= $unsigned(remainder);
                        p_reg <= - quotient;
                    end
                    valid <= 1;
                    done <= 0;
                end else begin
                    done <= 1;
                    valid <= 0;
                    state <= IDLE;
                end

        endcase
    end
end

endmodule
