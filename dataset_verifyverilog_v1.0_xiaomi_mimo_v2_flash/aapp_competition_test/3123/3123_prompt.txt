module quotations_solver (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [1:0] n,          // Number of quote blocks (1 to 3)
    input wire [1:0] a_0,        // Quote block 0
    input wire [1:0] a_1,        // Quote block 1
    input wire [1:0] a_2,        // Quote block 2
    output reg [1:0] max_k,      // Maximum nesting level found
    output reg done              // Computation finished
);

    // State machine for sequential execution
    reg [1:0] state;
    localparam S_IDLE = 0;
    localparam S_CHECK = 1;
    localparam S_DONE = 2;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            max_k <= 2'b00;
            done <= 1'b0;
            state <= S_IDLE;
        end else begin
            case (state)
                S_IDLE: begin
                    if (start) begin
                        max_k <= 2'b00;
                        done <= 1'b0;
                        state <= S_CHECK;
                    end
                end

                S_CHECK: begin
                    // Check for k=2 (2-quotation)
                    // Requires: a_0 >= 2, a_{n-1} >= 2, and inner structure valid
                    // For n=3: requires a_1 == 1 (single 1-quotation inside)
                    if ((n == 3) && (a_0 >= 2) && (a_2 >= 2) && (a_1 == 1)) begin
                        max_k <= 2'b10;
                    end
                    // Check for k=1 (1-quotation)
                    // Requires: n==2, a_0 == 1, a_1 == 1
                    else if ((n == 2) && (a_0 == 1) && (a_1 == 1)) begin
                        max_k <= 2'b01;
                    end
                    // else max_k remains 00
                    state <= S_DONE;
                end

                S_DONE: begin
                    done <= 1'b1;
                    state <= S_IDLE;
                end

                default: state <= S_IDLE;
            endcase
        end
    end

endmodule
