module friendship_clique (
    input clk,
    input rst_n,
    input start,
    input [2:0] n,          // Number of guests (1-8)
    input [7:0] adj [0:7], // Adjacency matrix: adj[i][j] = 1 if i and j are friends
    output reg [3:0] steps, // Minimum number of steps
    output reg [7:0] sequence [0:7], // Sequence of guests selected
    output reg done         // Computation complete
);

// State definitions
localparam IDLE = 2'd0;
localparam COMPUTE = 2'd1;
localparam FINISH = 2'd2;

reg [1:0] state;
reg [2:0] current_guest;
reg [7:0] temp_adj [0:7];
reg [3:0] step_count;
reg [7:0] sequence_reg [0:7];
reg [2:0] seq_index;

// Combinational logic for BFS
reg [7:0] visited_mask;
reg [7:0] queue [0:7];
reg [2:0] queue_front, queue_back;
reg [7:0] mask; // Bitmask of current connected component

integer i, j, k;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        steps <= 0;
        done <= 0;
        for (i = 0; i < 8; i = i + 1) begin
            sequence[i] <= 0;
        end
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    state <= COMPUTE;
                    step_count <= 0;
                    seq_index <= 0;
                    for (i = 0; i < 8; i = i + 1) begin
                        sequence_reg[i] <= 0;
                        temp_adj[i] <= adj[i];
                    end
                    current_guest <= 0;
                    done <= 0;
                end
            end
            
            COMPUTE: begin
                // Check if graph is complete
                if (is_complete(temp_adj, n)) begin
                    state <= FINISH;
                end else if (current_guest < n) begin
                    // Try selecting current_guest
                    if (has_neighbors(temp_adj, current_guest, n)) begin
                        // Add to sequence
                        sequence_reg[seq_index] <= current_guest + 1; // 1-indexed
                        seq_index <= seq_index + 1;
                        step_count <= step_count + 1;
                        
                        // Update adjacency: make all neighbors pairwise connected
                        for (i = 0; i < n; i = i + 1) begin
                            if (temp_adj[current_guest][i]) begin
                                for (j = i + 1; j < n; j = j + 1) begin
                                    if (temp_adj[current_guest][j]) begin
                                        temp_adj[i][j] <= 1;
                                        temp_adj[j][i] <= 1;
                                    end
                                end
                            end
                        end
                    end
                    current_guest <= current_guest + 1;
                end else begin
                    state <= FINISH;
                end
            end
            
            FINISH: begin
                steps <= step_count;
                for (i = 0; i < 8; i = i + 1) begin
                    sequence[i] <= sequence_reg[i];
                end
                done <= 1;
                state <= IDLE;
            end
        endcase
    end
end

// Helper function: check if graph is complete
function automatic is_complete;
    input [7:0] adj_matrix [0:7];
    input [2:0] size;
    begin
        is_complete = 1;
        for (i = 0; i < size; i = i + 1) begin
            for (j = i + 1; j < size; j = j + 1) begin
                if (!adj_matrix[i][j]) begin
                    is_complete = 0;
                end
            end
        end
    end
endfunction

// Helper function: check if guest has neighbors
function automatic has_neighbors;
    input [7:0] adj_matrix [0:7];
    input [2:0] guest;
    input [2:0] size;
    begin
        has_neighbors = 0;
        for (i = 0; i < size; i = i + 1) begin
            if (adj_matrix[guest][i] && i != guest) begin
                has_neighbors = 1;
            end
        end
    end
endfunction

endmodule
