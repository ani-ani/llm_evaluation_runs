module statue_rearrangement (
    input [63:0] a_flat,  // 8 elements of 8 bits each (element 0: bits 7:0, element 1: bits 15:8, ...)
    input [63:0] b_flat,
    output result
);
    // Extract arrays
    wire [7:0] a [0:7];
    wire [7:0] b [0:7];
    generate
        genvar g;
        for (g=0; g<8; g=g+1) begin
            assign a[g] = a_flat[8*g +: 8];
            assign b[g] = b_flat[8*g +: 8];
        end
    endgenerate

    // Step 1: Remove zeros to form 7-element non-zero arrays
    reg [7:0] a_no_zero [0:6];
    reg [7:0] b_no_zero [0:6];
    integer i, j;

    always @(*) begin
        j = 0;
        for (i=0; i<8; i=i+1) begin
            if (a[i] != 0) begin
                a_no_zero[j] = a[i];
                j = j + 1;
            end
        end
    end

    always @(*) begin
        j = 0;
        for (i=0; i<8; i=i+1) begin
            if (b[i] != 0) begin
                b_no_zero[j] = b[i];
                j = j + 1;
            end
        end
    end

    // Step 2: Find rotation index k in a_no_zero for b_no_zero[0]
    reg [2:0] k;
    always @(*) begin
        if (a_no_zero[0] == b_no_zero[0]) k = 0;
        else if (a_no_zero[1] == b_no_zero[0]) k = 1;
        else if (a_no_zero[2] == b_no_zero[0]) k = 2;
        else if (a_no_zero[3] == b_no_zero[0]) k = 3;
        else if (a_no_zero[4] == b_no_zero[0]) k = 4;
        else if (a_no_zero[5] == b_no_zero[0]) k = 5;
        else if (a_no_zero[6] == b_no_zero[0]) k = 6;
        else k = 0;
    end

    // Step 3: Form rotated sequence and compare
    wire [7:0] a_rotated [0:6];
    wire [6:0] match;

    generate
        for (g=0; g<7; g=g+1) begin
            wire [2:0] idx = (k + g < 7) ? (k + g) : (k + g - 7);
            assign a_rotated[g] = a_no_zero[idx];
            assign match[g] = (a_rotated[g] == b_no_zero[g]);
        end
    endgenerate

    assign result = &match;
endmodule
