module min_toll (
    input clk,
    input rst_n,
    input start,
    input [7:0] entrances [0:7],
    input [7:0] exits [0:7],
    output reg [15:0] result,
    output reg done
);

parameter N = 8;

// States
localparam IDLE = 3'd0;
localparam SORT_ENTRANCES = 3'd1;
localparam SORT_EXITS = 3'd2;
localparam DP_INIT = 3'd3;
localparam DP_COMPUTE = 3'd4;
localparam DONE = 3'd5;

reg [2:0] state;
reg [7:0] sorted_entrances [0:7];
reg [7:0] sorted_exits [0:7];
reg [15:0] dp [0:8];
reg [3:0] i; // counter for DP
reg [3:0] sort_i, sort_j;
reg [7:0] temp;

// INF value for impossible states
localparam INF = 16'hFFFF;

// Helper function for absolute difference
function [15:0] abs_diff;
    input [7:0] a, b;
    begin
        if (a > b)
            abs_diff = a - b;
        else
            abs_diff = b - a;
    end
endfunction

// Bubble sort for one array
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    // Copy inputs to sorted arrays
                    for (integer k = 0; k < N; k = k + 1) begin
                        sorted_entrances[k] <= entrances[k];
                        sorted_exits[k] <= exits[k];
                    end
                    sort_i <= 0;
                    sort_j <= 0;
                    state <= SORT_ENTRANCES;
                end
            end

            SORT_ENTRANCES: begin
                if (sort_i < N - 1) begin
                    if (sort_j < N - 1 - sort_i) begin
                        if (sorted_entrances[sort_j] > sorted_entrances[sort_j + 1]) begin
                            temp <= sorted_entrances[sort_j];
                            sorted_entrances[sort_j] <= sorted_entrances[sort_j + 1];
                            sorted_entrances[sort_j + 1] <= temp;
                        end
                        sort_j <= sort_j + 1;
                    end else begin
                        sort_j <= 0;
                        sort_i <= sort_i + 1;
                    end
                end else begin
                    sort_i <= 0;
                    sort_j <= 0;
                    state <= SORT_EXITS;
                end
            end

            SORT_EXITS: begin
                if (sort_i < N - 1) begin
                    if (sort_j < N - 1 - sort_i) begin
                        if (sorted_exits[sort_j] > sorted_exits[sort_j + 1]) begin
                            temp <= sorted_exits[sort_j];
                            sorted_exits[sort_j] <= sorted_exits[sort_j + 1];
                            sorted_exits[sort_j + 1] <= temp;
                        end
                        sort_j <= sort_j + 1;
                    end else begin
                        sort_j <= 0;
                        sort_i <= sort_i + 1;
                    end
                end else begin
                    state <= DP_INIT;
                end
            end

            DP_INIT: begin
                dp[0] <= 0;
                // Initialize dp[1]
                if (sorted_entrances[0] != sorted_exits[0])
                    dp[1] <= abs_diff(sorted_entrances[0], sorted_exits[0]);
                else
                    dp[1] <= INF;
                i <= 1;
                state <= DP_COMPUTE;
            end

            DP_COMPUTE: begin
                if (i < N) begin
                    // Compute options for dp[i+1]
                    reg [15:0] option1, option2, option3, option4;
                    option1 = INF;
                    option2 = INF;
                    option3 = INF;
                    option4 = INF;

                    // Option 1: direct match
                    if (sorted_entrances[i] != sorted_exits[i])
                        option1 = dp[i] + abs_diff(sorted_entrances[i], sorted_exits[i]);

                    // Option 2: swap with previous
                    if (i >= 1) begin
                        if (sorted_entrances[i] != sorted_exits[i-1] && sorted_entrances[i-1] != sorted_exits[i])
                            option2 = dp[i-1] + abs_diff(sorted_entrances[i], sorted_exits[i-1]) + abs_diff(sorted_entrances[i-1], sorted_exits[i]);
                    end

                    // Option 3: three-cycle type 1
                    if (i >= 2) begin
                        if (sorted_entrances[i] != sorted_exits[i-2] && sorted_entrances[i-1] != sorted_exits[i] && sorted_entrances[i-2] != sorted_exits[i-1])
                            option3 = dp[i-2] + abs_diff(sorted_entrances[i], sorted_exits[i-2]) + abs_diff(sorted_entrances[i-1], sorted_exits[i]) + abs_diff(sorted_entrances[i-2], sorted_exits[i-1]);
                    end

                    // Option 4: three-cycle type 2
                    if (i >= 2) begin
                        if (sorted_entrances[i] != sorted_exits[i-1] && sorted_entrances[i-1] != sorted_exits[i-2] && sorted_entrances[i-2] != sorted_exits[i])
                            option4 = dp[i-2] + abs_diff(sorted_entrances[i], sorted_exits[i-1]) + abs_diff(sorted_entrances[i-1], sorted_exits[i-2]) + abs_diff(sorted_entrances[i-2], sorted_exits[i]);
                    end

                    // Find minimum
                    dp[i+1] <= option1;
                    if (option2 < dp[i+1]) dp[i+1] <= option2;
                    if (option3 < dp[i+1]) dp[i+1] <= option3;
                    if (option4 < dp[i+1]) dp[i+1] <= option4;

                    i <= i + 1;
                end else begin
                    result <= dp[N];
                    done <= 1;
                    state <= DONE;
                end
            end

            DONE: begin
                done <= 0;
                state <= IDLE;
            end
        endcase
    end
end

endmodule
