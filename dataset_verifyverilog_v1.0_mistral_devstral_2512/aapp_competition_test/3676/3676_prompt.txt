module polygon_count (
    input  [1:0] R,   // 00=1, 01=2, 10=3, 11=4
    input  [1:0] C,   // same encoding
    output reg [15:0] count
);

always @(*) begin
    case ({R, C})
        // R=1, C=1..4
        4'b0000: count = 1;
        4'b0001: count = 3;
        4'b0010: count = 6;
        4'b0011: count = 10;
        // R=2, C=1..4
        4'b0100: count = 3;
        4'b0101: count = 13;
        4'b0110: count = 41;
        4'b0111: count = 108;
        // R=3, C=1..4
        4'b1000: count = 6;
        4'b1001: count = 41;
        4'b1010: count = 128;
        4'b1011: count = 355;
        // R=4, C=1..4
        4'b1100: count = 10;
        4'b1101: count = 108;
        4'b1110: count = 355;
        4'b1111: count = 1190;
        default: count = 0;
    endcase
end

endmodule
