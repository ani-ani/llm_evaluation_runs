module haybale_sort (
    input clk,
    input rst_n,
    input start,
    input [7:0] s,   // 0 = C, 1 = P
    output reg [3:0] result,
    output reg done
);
    // Combinational logic to compute result
    reg [4:0] inv;          // inversion count, up to 16
    reg has_2_inv;          // 1 if any triple with 2 inversions exists
    reg [4:0] ceil_half;    // ceil(inv/2)
    reg [3:0] result_comb;  // combinational result

    // Compute inversion count
    integer i;
    reg [2:0] zeros_suffix [0:8]; // suffix zero count, index 8 = 0
    always_comb begin
        zeros_suffix[8] = 0;
        for (i = 7; i >= 0; i = i - 1) begin
            zeros_suffix[i] = zeros_suffix[i+1] + (s[i] ? 0 : 1);
        end
        inv = 0;
        for (i = 0; i < 8; i = i + 1) begin
            if (s[i]) inv = inv + zeros_suffix[i+1];
        end
    end

    // Check for PCC or PPC
    always_comb begin
        has_2_inv = 0;
        for (i = 0; i <= 5; i = i + 1) begin
            if ((s[i] && !s[i+1] && !s[i+2]) || (s[i] && s[i+1] && !s[i+2]))
                has_2_inv = 1;
        end
    end

    // Compute ceil(inv/2)
    always_comb begin
        ceil_half = (inv + 1) >> 1;
    end

    // Compute final combinational result
    always_comb begin
        if (inv == 0)
            result_comb = 0;
        else if (has_2_inv)
            result_comb = ceil_half;
        else if (inv[0])  // odd
            result_comb = ceil_half;
        else
            result_comb = ceil_half + 4'd1;
    end

    // Sequential state machine
    reg [1:0] state; // 00: IDLE, 01: COMPUTE, 10: DONE
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= 2'b00;
            result <= 4'b0;
            done <= 1'b0;
        end else begin
            case (state)
                2'b00: begin // IDLE
                    done <= 1'b0;
                    if (start) state <= 2'b01;
                end
                2'b01: begin // COMPUTE
                    result <= result_comb;
                    state <= 2'b10;
                end
                2'b10: begin // DONE
                    done <= 1'b1;
                    state <= 2'b00;
                end
                default: state <= 2'b00;
            endcase
        end
    end
endmodule
