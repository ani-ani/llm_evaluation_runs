module IOUSettlement #(
    parameter N = 4,                    // Maximum number of friends (nodes)
    parameter M = 8,                    // Maximum number of IOUs (edges)
    parameter WEIGHT_WIDTH = 10,        // Width for debt amounts (max 1000)
    parameter NODE_WIDTH = 3,           // Width for node IDs (2^3=8 >= 4)
    parameter CYCLE_COUNT = 20          // Number of precomputed cycles for N=4
)(
    input clk,                          // Clock, rising edge
    input rst_n,                        // Active-low synchronous reset
    input start,                        // Start pulse (1 cycle)
    input [NODE_WIDTH-1:0] n,          // Actual number of friends (1 to N)
    input [3:0] m,                      // Actual number of IOUs (0 to M)
    input [M*NODE_WIDTH-1:0] a_flat,   // Packed a values (a0,a1,...aM-1)
    input [M*NODE_WIDTH-1:0] b_flat,   // Packed b values (b0,b1,...bM-1)
    input [M*WEIGHT_WIDTH-1:0] c_flat, // Packed c values (c0,c1,...cM-1)
    output reg done,                    // Done signal
    output reg [WEIGHT_WIDTH-1:0] graph [0:N-1][0:N-1] // Output debt matrix
);

// State machine states
localparam STATE_IDLE = 3'b000;
localparam STATE_LOAD = 3'b001;
localparam STATE_CANCEL_PASS = 3'b010;
localparam STATE_CHECK_CYCLE = 3'b011;
localparam STATE_SUBTRACT = 3'b100;
localparam STATE_OUTPUT = 3'b101;

reg [2:0] state, next_state;

// Unpacked input arrays
reg [NODE_WIDTH-1:0] a_reg [0:M-1];
reg [NODE_WIDTH-1:0] b_reg [0:M-1];
reg [WEIGHT_WIDTH-1:0] c_reg [0:M-1];

// Cycle cancellation registers
reg [CYCLE_COUNT-1:0] cycle_idx;          // Current cycle index
reg [1:0] cycle_len;                       // Length of current cycle
reg [NODE_WIDTH-1:0] edge_u [0:3];        // u nodes of edges in cycle
reg [NODE_WIDTH-1:0] edge_v [0:3];        // v nodes of edges in cycle
reg [WEIGHT_WIDTH-1:0] min_weight;        // Minimum weight in cycle
reg [WEIGHT_WIDTH-1:0] temp_weight;       // Temporary weight for comparison
reg cancel_occurred;                       // Flag if any cancellation in pass
reg [9:0] pass_count;                      // Safety counter for max passes

// ROM for precomputed cycles (for N=4)
// Each entry: {len, edge0_u, edge0_v, edge1_u, edge1_v, edge2_u, edge2_v, edge3_u, edge3_v}
// len is 2 bits, each node is 3 bits, so 2 + 8*3 = 26 bits per cycle
reg [25:0] cycle_rom [0:CYCLE_COUNT-1];

// Initialize ROM with all 20 cycles for N=4
initial begin
    // Cycle 0: (0,1) -> (1,0)  length=2
    cycle_rom[0] = 26'b01_000_001_001_000_000_000_000_000;
    // Cycle 1: (0,2) -> (2,0)  length=2
    cycle_rom[1] = 26'b01_000_010_010_000_000_000_000_000;
    // Cycle 2: (0,3) -> (3,0)  length=2
    cycle_rom[2] = 26'b01_000_011_011_000_000_000_000_000;
    // Cycle 3: (1,2) -> (2,1)  length=2
    cycle_rom[3] = 26'b01_001_010_010_001_000_000_000_000;
    // Cycle 4: (1,3) -> (3,1)  length=2
    cycle_rom[4] = 26'b01_001_011_011_001_000_000_000_000;
    // Cycle 5: (2,3) -> (3,2)  length=2
    cycle_rom[5] = 26'b01_010_011_011_010_000_000_000_000;
    // Cycle 6: (0,1)->(1,2)->(2,0)  length=3
    cycle_rom[6] = 26'b10_000_001_001_010_010_000_000_000;
    // Cycle 7: (0,1)->(1,3)->(3,0)  length=3
    cycle_rom[7] = 26'b10_000_001_001_011_011_000_000_000;
    // Cycle 8: (0,2)->(2,1)->(1,0)  length=3
    cycle_rom[8] = 26'b10_000_010_010_001_001_000_000_000;
    // Cycle 9: (0,2)->(2,3)->(3,0)  length=3
    cycle_rom[9] = 26'b10_000_010_010_011_011_000_000_000;
    // Cycle 10: (0,3)->(3,1)->(1,0)  length=3
    cycle_rom[10] = 26'b10_000_011_011_001_001_000_000_000;
    // Cycle 11: (0,3)->(3,2)->(2,0)  length=3
    cycle_rom[11] = 26'b10_000_011_011_010_010_000_000_000;
    // Cycle 12: (1,2)->(2,3)->(3,1)  length=3
    cycle_rom[12] = 26'b10_001_010_010_011_011_001_000_000;
    // Cycle 13: (1,3)->(3,2)->(2,1)  length=3
    cycle_rom[13] = 26'b10_001_011_011_010_010_001_000_000;
    // Cycle 14: (0,1)->(1,2)->(2,3)->(3,0)  length=4
    cycle_rom[14] = 26'b11_000_001_001_010_010_011_011_000;
    // Cycle 15: (0,1)->(1,3)->(3,2)->(2,0)  length=4
    cycle_rom[15] = 26'b11_000_001_001_011_011_010_010_000;
    // Cycle 16: (0,2)->(2,1)->(1,3)->(3,0)  length=4
    cycle_rom[16] = 26'b11_000_010_010_001_001_011_011_000;
    // Cycle 17: (0,2)->(2,3)->(3,1)->(1,0)  length=4
    cycle_rom[17] = 26'b11_000_010_010_011_011_001_001_000;
    // Cycle 18: (0,3)->(3,1)->(1,2)->(2,0)  length=4
    cycle_rom[18] = 26'b11_000_011_011_001_001_010_010_000;
    // Cycle 19: (0,3)->(3,2)->(2,1)->(1,0)  length=4
    cycle_rom[19] = 26'b11_000_011_011_010_010_001_001_000;
end

// Unpack inputs into arrays
integer i;
always @(posedge clk) begin
    if (state == STATE_LOAD) begin
        for (i = 0; i < M; i = i + 1) begin
            a_reg[i] <= a_flat[i*NODE_WIDTH +: NODE_WIDTH];
            b_reg[i] <= b_flat[i*NODE_WIDTH +: NODE_WIDTH];
            c_reg[i] <= c_flat[i*WEIGHT_WIDTH +: WEIGHT_WIDTH];
        end
    end
end

// State transition and output logic
always @(posedge clk) begin
    if (!rst_n) begin
        state <= STATE_IDLE;
        done <= 0;
        graph <= '{default: 0};
        cycle_idx <= 0;
        cancel_occurred <= 0;
        pass_count <= 0;
    end else begin
        case (state)
            STATE_IDLE: begin
                done <= 0;
                if (start) begin
                    state <= STATE_LOAD;
                    // Clear graph
                    graph <= '{default: 0};
                    cycle_idx <= 0;
                    cancel_occurred <= 0;
                    pass_count <= 0;
                end
            end
            
            STATE_LOAD: begin
                // Load IOUs into graph
                for (i = 0; i < M; i = i + 1) begin
                    if (i < m) begin
                        graph[a_reg[i]][b_reg[i]] <= c_reg[i];
                    end
                end
                state <= STATE_CANCEL_PASS;
            end
            
            STATE_CANCEL_PASS: begin
                // Start a new pass
                cancel_occurred <= 0;
                cycle_idx <= 0;
                state <= STATE_CHECK_CYCLE;
                pass_count <= pass_count + 1;
            end
            
            STATE_CHECK_CYCLE: begin
                if (cycle_idx >= CYCLE_COUNT) begin
                    // Finished this pass
                    if (cancel_occurred && pass_count < 1000) begin
                        state <= STATE_CANCEL_PASS; // Another pass
                    end else begin
                        state <= STATE_OUTPUT;      // Done
                    end
                end else begin
                    // Read current cycle from ROM
                    {cycle_len, edge_u[0], edge_v[0], edge_u[1], edge_v[1], 
                     edge_u[2], edge_v[2], edge_u[3], edge_v[3]} <= cycle_rom[cycle_idx];
                    state <= STATE_SUBTRACT;
                    min_weight <= {WEIGHT_WIDTH{1'b1}}; // Set to max value
                end
            end
            
            STATE_SUBTRACT: begin
                // Check if all edges in cycle have positive weight
                if (graph[edge_u[0]][edge_v[0]] > 0 &&
                    (cycle_len < 2 || graph[edge_u[1]][edge_v[1]] > 0) &&
                    (cycle_len < 3 || graph[edge_u[2]][edge_v[2]] > 0) &&
                    (cycle_len < 4 || graph[edge_u[3]][edge_v[3]] > 0)) begin
                    // Find minimum weight
                    min_weight <= graph[edge_u[0]][edge_v[0]];
                    if (cycle_len >= 2 && graph[edge_u[1]][edge_v[1]] < min_weight)
                        min_weight <= graph[edge_u[1]][edge_v[1]];
                    if (cycle_len >= 3 && graph[edge_u[2]][edge_v[2]] < min_weight)
                        min_weight <= graph[edge_u[2]][edge_v[2]];
                    if (cycle_len >= 4 && graph[edge_u[3]][edge_v[3]] < min_weight)
                        min_weight <= graph[edge_u[3]][edge_v[3]];
                    // Subtract min from all edges
                    graph[edge_u[0]][edge_v[0]] <= graph[edge_u[0]][edge_v[0]] - min_weight;
                    if (cycle_len >= 2)
                        graph[edge_u[1]][edge_v[1]] <= graph[edge_u[1]][edge_v[1]] - min_weight;
                    if (cycle_len >= 3)
                        graph[edge_u[2]][edge_v[2]] <= graph[edge_u[2]][edge_v[2]] - min_weight;
                    if (cycle_len >= 4)
                        graph[edge_u[3]][edge_v[3]] <= graph[edge_u[3]][edge_v[3]] - min_weight;
                    cancel_occurred <= 1;
                end
                cycle_idx <= cycle_idx + 1;
                state <= STATE_CHECK_CYCLE;
            end
            
            STATE_OUTPUT: begin
                done <= 1;
                // Hold graph values for reading
                state <= STATE_IDLE;
            end
            
            default: state <= STATE_IDLE;
        endcase
    end
end

endmodule
