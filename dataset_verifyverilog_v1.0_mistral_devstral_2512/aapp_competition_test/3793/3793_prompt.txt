module cube_checker (
  input signed [23:0] in0_0, in0_1, in0_2,
  input signed [23:0] in1_0, in1_1, in1_2,
  input signed [23:0] in2_0, in2_1, in2_2,
  input signed [23:0] in3_0, in3_1, in3_2,
  input signed [23:0] in4_0, in4_1, in4_2,
  input signed [23:0] in5_0, in5_1, in5_2,
  input signed [23:0] in6_0, in6_1, in6_2,
  input signed [23:0] in7_0, in7_1, in7_2,
  output reg is_cube,
  output reg [23:0] out0_0, out0_1, out0_2,
  output reg [23:0] out1_0, out1_1, out1_2,
  output reg [23:0] out2_0, out2_1, out2_2,
  output reg [23:0] out3_0, out3_1, out3_2,
  output reg [23:0] out4_0, out4_1, out4_2,
  output reg [23:0] out5_0, out5_1, out5_2,
  output reg [23:0] out6_0, out6_1, out6_2,
  output reg [23:0] out7_0, out7_1, out7_2
);
  // Form points from inputs
  wire signed [23:0] points [0:7][0:2];
  assign points[0] = {in0_0, in0_1, in0_2};
  assign points[1] = {in1_0, in1_1, in1_2};
  assign points[2] = {in2_0, in2_1, in2_2};
  assign points[3] = {in3_0, in3_1, in3_2};
  assign points[4] = {in4_0, in4_1, in4_2};
  assign points[5] = {in5_0, in5_1, in5_2};
  assign points[6] = {in6_0, in6_1, in6_2};
  assign points[7] = {in7_0, in7_1, in7_2};

  // Compute all 28 squared distances
  wire signed [63:0] dist2 [0:27];
  wire signed [63:0] dx, dy, dz;
  reg [4:0] pair_idx;
  integer i, j, k;

  always @(*) begin
    pair_idx = 0;
    for (i = 0; i < 8; i = i + 1) begin
      for (j = i+1; j < 8; j = j + 1) begin
        dx = points[i][0] - points[j][0];
        dy = points[i][1] - points[j][1];
        dz = points[i][2] - points[j][2];
        dist2[pair_idx] = dx*dx + dy*dy + dz*dz;
        pair_idx = pair_idx + 1;
      end
    end
  end

  // Find minimal non-zero distance squared
  reg signed [63:0] min_dist2;
  reg found_min;
  always @(*) begin
    min_dist2 = 64'h7FFFFFFFFFFFFFFF; // Max value
    found_min = 0;
    for (k = 0; k < 28; k = k + 1) begin
      if (dist2[k] > 0 && dist2[k] < min_dist2) begin
        min_dist2 = dist2[k];
        found_min = 1;
      end
    end
  end

  // Verify distance ratios and counts
  reg [2:0] edge_count [0:7];
  reg [2:0] face_count [0:7];
  reg [2:0] space_count [0:7];
  reg [2:0] edge_errors, face_errors, space_errors;
  reg valid_distances;
  integer p_idx;

  always @(*) begin
    // Initialize counters
    for (i = 0; i < 8; i = i + 1) begin
      edge_count[i] = 0;
      face_count[i] = 0;
      space_count[i] = 0;
    end
    edge_errors = 0;
    face_errors = 0;
    space_errors = 0;
    valid_distances = 1;
    p_idx = 0;

    // Check all pairs
    for (i = 0; i < 8; i = i + 1) begin
      for (j = i+1; j < 8; j = j + 1) begin
        if (dist2[p_idx] == 0) valid_distances = 0;
        else if (dist2[p_idx] == min_dist2) begin
          edge_count[i] = edge_count[i] + 1;
          edge_count[j] = edge_count[j] + 1;
        end
        else if (dist2[p_idx] == 2*min_dist2) begin
          face_count[i] = face_count[i] + 1;
          face_count[j] = face_count[j] + 1;
        end
        else if (dist2[p_idx] == 3*min_dist2) begin
          space_count[i] = space_count[i] + 1;
          space_count[j] = space_count[j] + 1;
        end
        else valid_distances = 0;
        p_idx = p_idx + 1;
      end
    end

    // Verify counts per vertex
    for (i = 0; i < 8; i = i + 1) begin
      if (edge_count[i] != 3) edge_errors = edge_errors + 1;
      if (face_count[i] != 3) face_errors = face_errors + 1;
      if (space_count[i] != 1) space_errors = space_errors + 1;
    end
  end

  // Output assignments
  always @(*) begin
    // Default outputs: pass through inputs
    {out0_0, out0_1, out0_2} = {in0_0, in0_1, in0_2};
    {out1_0, out1_1, out1_2} = {in1_0, in1_1, in1_2};
    {out2_0, out2_1, out2_2} = {in2_0, in2_1, in2_2};
    {out3_0, out3_1, out3_2} = {in3_0, in3_1, in3_2};
    {out4_0, out4_1, out4_2} = {in4_0, in4_1, in4_2};
    {out5_0, out5_1, out5_2} = {in5_0, in5_1, in5_2};
    {out6_0, out6_1, out6_2} = {in6_0, in6_1, in6_2};
    {out7_0, out7_1, out7_2} = {in7_0, in7_1, in7_2};

    // Cube validation
    is_cube = valid_distances && 
              found_min &&
              (edge_errors == 0) && 
              (face_errors == 0) && 
              (space_errors == 0);
  end
endmodule
