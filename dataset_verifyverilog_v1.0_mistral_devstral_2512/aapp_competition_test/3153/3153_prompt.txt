module card_purchase_solver #(
    parameter MAX_CHILDREN = 8,
    parameter MAX_PURCHASES = 16,
    parameter MAX_OUTPUT_PURCHASES = 32,
    parameter DATA_WIDTH = 4,
    parameter PAIR_WIDTH = 4
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [DATA_WIDTH-1:0] target_counts [0:MAX_CHILDREN-1],
    input wire [PAIR_WIDTH-1:0] pair_a [0:MAX_PURCHASES-1],
    input wire [PAIR_WIDTH-1:0] pair_b [0:MAX_PURCHASES-1],
    input wire [3:0] valid_pairs_count,
    output reg [PAIR_WIDTH-1:0] out_child1,
    output reg [PAIR_WIDTH-1:0] out_child2,
    output reg [1:0] out_outcome,
    output reg out_valid,
    output reg done
);

    localparam IDLE = 3'b000, INIT = 3'b001, SEARCH = 3'b010, DECIDE = 3'b011,
               OUTPUT = 3'b100, UPDATE = 3'b101, CHECK = 3'b110, FINISHED = 3'b111;

    reg [2:0] state, next_state;
    reg [DATA_WIDTH-1:0] current_counts [0:MAX_CHILDREN-1];
    reg [3:0] search_idx;
    reg [5:0] purchase_counter;
    
    wire [PAIR_WIDTH-1:0] curr_a, curr_b;
    wire [DATA_WIDTH-1:0] target_a, target_b, current_a, current_b;
    wire needs_a, needs_b;
    wire [DATA_WIDTH-1:0] needed_a, needed_b;
    wire all_targets_met;

    assign curr_a = (search_idx < valid_pairs_count) ? pair_a[search_idx] : 0;
    assign curr_b = (search_idx < valid_pairs_count) ? pair_b[search_idx] : 0;
    assign target_a = (curr_a && curr_a <= MAX_CHILDREN) ? target_counts[curr_a-1] : 0;
    assign target_b = (curr_b && curr_b <= MAX_CHILDREN) ? target_counts[curr_b-1] : 0;
    assign current_a = (curr_a && curr_a <= MAX_CHILDREN) ? current_counts[curr_a-1] : 0;
    assign current_b = (curr_b && curr_b <= MAX_CHILDREN) ? current_counts[curr_b-1] : 0;
    assign needs_a = (current_a < target_a);
    assign needs_b = (current_b < target_b);
    assign needed_a = target_a - current_a;
    assign needed_b = target_b - current_b;

    integer i;
    reg met;
    always @(*) begin
        met = 1'b1;
        for (i = 0; i < MAX_CHILDREN; i = i + 1)
            if (current_counts[i] != target_counts[i]) met = 1'b0;
    end
    assign all_targets_met = met;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) state <= IDLE;
        else state <= next_state;
    end

    always @(*) begin
        case (state)
            IDLE: next_state = start ? INIT : IDLE;
            INIT: next_state = SEARCH;
            SEARCH: begin
                if (search_idx >= valid_pairs_count) next_state = FINISHED;
                else if (needs_a || needs_b) next_state = DECIDE;
                else next_state = SEARCH;
            end
            DECIDE: next_state = OUTPUT;
            OUTPUT: next_state = UPDATE;
            UPDATE: next_state = CHECK;
            CHECK: next_state = (all_targets_met || purchase_counter >= MAX_OUTPUT_PURCHASES-1) ? FINISHED : SEARCH;
            FINISHED: next_state = FINISHED;
            default: next_state = IDLE;
        endcase
    end

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            out_valid <= 1'b0; done <= 1'b0; purchase_counter <= 6'b0;
            search_idx <= 4'b0; out_child1 <= 4'b0; out_child2 <= 4'b0; out_outcome <= 2'b0;
            for (i = 0; i < MAX_CHILDREN; i = i + 1) current_counts[i] <= 4'b0;
        end else begin
            case (state)
                INIT: begin
                    for (i = 0; i < MAX_CHILDREN; i = i + 1) current_counts[i] <= 4'b0;
                    search_idx <= 4'b0; purchase_counter <= 6'b0; out_valid <= 1'b0; done <= 1'b0;
                end
                SEARCH: begin
                    if (!(needs_a || needs_b) && search_idx < valid_pairs_count)
                        search_idx <= search_idx + 1;
                    out_valid <= 1'b0;
                end
                DECIDE: begin
                    out_child1 <= curr_a; out_child2 <= curr_b;
                    if (needs_a && needs_b) out_outcome <= 2'b01;
                    else if (needs_a && !needs_b) out_outcome <= (needed_a >= 2) ? 2'b10 : 2'b01;
                    else if (!needs_a && needs_b) out_outcome <= 2'b00;
                    else out_outcome <= 2'b00;
                end
                OUTPUT: out_valid <= 1'b1;
                UPDATE: begin
                    if (curr_a && curr_a <= MAX_CHILDREN) current_counts[curr_a-1] <= current_counts[curr_a-1] + out_outcome;
                    if (curr_b && curr_b <= MAX_CHILDREN) current_counts[curr_b-1] <= current_counts[curr_b-1] + (2'b10 - out_outcome);
                    purchase_counter <= purchase_counter + 1; search_idx <= 4'b0; out_valid <= 1'b0;
                end
                FINISHED: done <= 1'b1;
            endcase
        end
    end
endmodule
