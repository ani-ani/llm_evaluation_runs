module bookcase_optimizer (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] book_count,  // Number of books (3-8)
    
    // Individual book inputs (8 books max)
    // Height: 9 bits (150-300), Thickness: 6 bits (5-30)
    input wire [8:0] h0, h1, h2, h3, h4, h5, h6, h7,
    input wire [5:0] t0, t1, t2, t3, t4, t5, t6, t7,
    
    output reg [19:0] min_area,  // Result: 20 bits (max ~216,000)
    output reg done
);

// Internal memory for book data
reg [8:0] height [0:7];
reg [5:0] thickness [0:7];

// State machine states
reg [2:0] state;
localparam IDLE = 0, LOAD = 1, COMPUTE = 2, DONE = 3;

// Iteration counter: 0 to 3^8-1 = 6560
reg [15:0] iter;

// Best result tracking
reg [19:0] best;

// Function to extract shelf assignment for book i from iteration counter
function [1:0] get_shelf(input [15:0] cnt, input [3:0] book_idx);
    reg [15:0] temp;
    integer j;
    begin
        temp = cnt;
        for (j = 0; j < 8; j = j + 1) begin
            if (j < book_idx) begin
                temp = temp / 3;
            end
        end
        get_shelf = temp % 3;
    end
endfunction

// Combinational block to compute area for current iteration
reg [8:0] max_h0, max_h1, max_h2;
reg [7:0] sum_t0, sum_t1, sum_t2;  // Max sum = 8*30 = 240
reg [19:0] current_area;

always @(*) begin
    // Reset max and sum
    max_h0 = 0; max_h1 = 0; max_h2 = 0;
    sum_t0 = 0; sum_t1 = 0; sum_t2 = 0;
    
    // Process each book
    for (integer i = 0; i < 8; i = i + 1) begin
        if (i < book_count) begin
            // Get shelf assignment for this book
            case(get_shelf(iter, i))
                2'b00: begin
                    if (height[i] > max_h0) max_h0 = height[i];
                    sum_t0 = sum_t0 + thickness[i];
                end
                2'b01: begin
                    if (height[i] > max_h1) max_h1 = height[i];
                    sum_t1 = sum_t1 + thickness[i];
                end
                2'b10: begin
                    if (height[i] > max_h2) max_h2 = height[i];
                    sum_t2 = sum_t2 + thickness[i];
                end
            endcase
        end
    end
    
    // Compute objective function
    current_area = (max_h0 + max_h1 + max_h2) * 
                   ((sum_t0 > sum_t1) ? 
                    ((sum_t0 > sum_t2) ? sum_t0 : sum_t2) :
                    ((sum_t1 > sum_t2) ? sum_t1 : sum_t2));
end

// Sequential logic
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        min_area <= 0;
        iter <= 0;
        best <= 20'hFFFFF;
        // Clear book storage
        for (integer i = 0; i < 8; i = i + 1) begin
            height[i] <= 0;
            thickness[i] <= 0;
        end
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    state <= LOAD;
                end
            end
            
            LOAD: begin
                // Load books from input ports
                height[0] <= h0; thickness[0] <= t0;
                height[1] <= h1; thickness[1] <= t1;
                height[2] <= h2; thickness[2] <= t2;
                height[3] <= h3; thickness[3] <= t3;
                height[4] <= h4; thickness[4] <= t4;
                height[5] <= h5; thickness[5] <= t5;
                height[6] <= h6; thickness[6] <= t6;
                height[7] <= h7; thickness[7] <= t7;
                
                iter <= 0;
                best <= 20'hFFFFF;
                state <= COMPUTE;
            end
            
            COMPUTE: begin
                if (iter < 16'd6561) begin  // 3^8 = 6561
                    // Update best if current area is smaller
                    if (current_area < best && iter > 0) begin
                        best <= current_area;
                    end
                    iter <= iter + 1;
                end else begin
                    // All iterations complete
                    min_area <= best;
                    done <= 1;
                    state <= DONE;
                end
            end
            
            DONE: begin
                if (!start) begin
                    state <= IDLE;
                    done <= 0;
                end
            end
        endcase
    end
end

endmodule
