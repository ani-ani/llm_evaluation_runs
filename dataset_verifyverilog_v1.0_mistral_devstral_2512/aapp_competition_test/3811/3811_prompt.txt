module wcd_finder #(
    parameter MAX_PAIRS = 8,
    parameter DATA_WIDTH = 32,
    parameter MAX_CANDIDATES = 8
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [DATA_WIDTH-1:0] a_in,
    input wire [DATA_WIDTH-1:0] b_in,
    input wire [2:0] pair_count,  // Number of pairs (1-8)
    output reg [DATA_WIDTH-1:0] wcd_result,
    output reg done,
    output reg valid
);

    // State definitions
    localparam IDLE = 3'b000;
    localparam FACTOR_INIT = 3'b001;
    localparam FACTOR_CHECK = 3'b010;
    localparam FACTOR_UPDATE = 3'b011;
    localparam RESULT_READY = 3'b100;
    localparam ERROR_STATE = 3'b101;

    // Internal registers
    reg [DATA_WIDTH-1:0] candidates [0:MAX_CANDIDATES-1];
    reg [2:0] state, next_state;
    reg [2:0] pair_index;
    reg [2:0] candidate_index;
    reg [2:0] valid_count;
    reg [DATA_WIDTH-1:0] stored_a, stored_b;
    reg [DATA_WIDTH-1:0] temp_result;
    
    // Helper wires for divisibility checks
    wire candidate_divides_a = (stored_a % candidates[candidate_index] == 0);
    wire candidate_divides_b = (stored_b % candidates[candidate_index] == 0);
    wire is_valid_candidate = candidate_divides_a || candidate_divides_b;

    integer i;

    // State transition logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            pair_index <= 0;
            candidate_index <= 0;
            valid_count <= 0;
            done <= 0;
            valid <= 0;
            wcd_result <= 0;
            stored_a <= 0;
            stored_b <= 0;
            temp_result <= 0;
            for (i = 0; i < MAX_CANDIDATES; i = i + 1) begin
                candidates[i] <= 0;
            end
        end else begin
            state <= next_state;
            
            case (state)
                IDLE: begin
                    done <= 0;
                    valid <= 0;
                    if (start) begin
                        pair_index <= 0;
                        candidate_index <= 0;
                        valid_count <= 0;
                        stored_a <= a_in;
                        stored_b <= b_in;
                        // Initialize candidates with simple divisors (2, 3, 5, 7, 11, 13, 17, 19)
                        // For simplicity, we'll use the actual values from first pair as candidates
                        // but only their prime factors (simplified: just use the values themselves)
                        candidates[0] <= (a_in > 1) ? a_in : 2;
                        candidates[1] <= (b_in > 1) ? b_in : 3;
                        candidates[2] <= 5;
                        candidates[3] <= 7;
                        candidates[4] <= 11;
                        candidates[5] <= 13;
                        candidates[6] <= 17;
                        candidates[7] <= 19;
                        valid_count <= 2; // Initially only first 2 are from pair
                    end
                end
                
                FACTOR_CHECK: begin
                    // Check if current candidate works for current pair
                    if (!is_valid_candidate) begin
                        // Mark as invalid by setting to 0
                        candidates[candidate_index] <= 0;
                        valid_count <= (valid_count > 0) ? valid_count - 1 : 0;
                    end
                    candidate_index <= candidate_index + 1;
                end
                
                FACTOR_UPDATE: begin
                    // Get next pair
                    if (pair_index < pair_count) begin
                        stored_a <= a_in;
                        stored_b <= b_in;
                        candidate_index <= 0;
                        pair_index <= pair_index + 1;
                    end
                end
                
                RESULT_READY: begin
                    // Find first valid candidate
                    if (candidates[candidate_index] > 1 && valid_count > 0) begin
                        wcd_result <= candidates[candidate_index];
                        valid <= 1;
                    end else begin
                        valid <= 0;
                    end
                    done <= 1;
                end
            endcase
        end
    end

    // Next state logic
    always @(*) begin
        next_state = state;
        case (state)
            IDLE: begin
                if (start) next_state = FACTOR_CHECK;
            end
            
            FACTOR_CHECK: begin
                if (candidate_index >= MAX_CANDIDATES - 1) begin
                    if (pair_index < pair_count - 1) begin
                        next_state = FACTOR_UPDATE;
                    end else begin
                        next_state = RESULT_READY;
                    end
                end else begin
                    next_state = FACTOR_CHECK;
                end
            end
            
            FACTOR_UPDATE: begin
                next_state = FACTOR_CHECK;
            end
            
            RESULT_READY: begin
                next_state = IDLE;
            end
            
            default: next_state = IDLE;
        endcase
    end

endmodule
