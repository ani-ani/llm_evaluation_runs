module tv_coverage (
    input clk, input rst_n, input start,
    input [7:0] N, input [15:0] D,
    input [7:0] transmitter_flags,
    input [15:0] X [0:7], input [15:0] H [0:7],
    output real covered_length, output reg done
);

    localparam IDLE=0, COMP=1, MERGE=2, DONE=3;
    reg [1:0] state;
    reg [3:0] i, j, k;
    real left_bounds[0:7], right_bounds[0:7];
    reg valid[0:7];
    real intervals_left[0:7], intervals_right[0:7];
    real total_length, current_left, current_right;
    real temp_left, temp_right;
    integer cnt;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE; done <= 0; covered_length <= 0;
        end else begin
            case (state)
                IDLE: if (start) begin
                    state <= COMP; i <= 0; j <= 0; cnt <= 0;
                    for (k=0; k<8; k=k+1) valid[k] <= 0;
                end
                COMP: if (i < N) begin
                    if (transmitter_flags[i]) begin
                        if (j==0) begin left_bounds[i]=0; right_bounds[i]=D; valid[i]<=1; end
                        if (j < N) begin
                            if (j != i) begin
                                real x_t = X[i], h_t = H[i], x_j = X[j], h_j = H[j];
                                if (x_j < x_t) begin
                                    real limit = x_j - (h_j/h_t)*(x_t-x_j);
                                    if (limit > left_bounds[i]) left_bounds[i] = limit;
                                end else if (x_j > x_t) begin
                                    real limit = x_j + (h_j/h_t)*(x_j-x_t);
                                    if (limit < right_bounds[i]) right_bounds[i] = limit;
                                end
                            end
                            j <= j + 1;
                        end else begin
                            j <= 0; i <= i + 1;
                        end
                    end else i <= i + 1;
                end else begin
                    state <= MERGE; cnt <= 0;
                    for (k=0; k<8; k=k+1) if (valid[k]) begin
                        intervals_left[cnt] <= left_bounds[k];
                        intervals_right[cnt] <= right_bounds[k];
                        cnt <= cnt + 1;
                    end
                    k <= 0;
                end
                MERGE: if (cnt > 0) begin
                    if (k < cnt-1) begin
                        if (intervals_left[k] > intervals_left[k+1]) begin
                            temp_left <= intervals_left[k]; temp_right <= intervals_right[k];
                            intervals_left[k] <= intervals_left[k+1]; intervals_right[k] <= intervals_right[k+1];
                            intervals_left[k+1] <= temp_left; intervals_right[k+1] <= temp_right;
                        end
                        k <= k + 1;
                    end else begin
                        current_left <= intervals_left[0]; current_right <= intervals_right[0];
                        total_length <= 0; k <= 1;
                        state <= 3;
                    end
                end else begin
                    covered_length <= 0; done <= 1; state <= IDLE;
                end
                3: if (k < cnt) begin
                    if (intervals_left[k] <= current_right) begin
                        if (intervals_right[k] > current_right) current_right <= intervals_right[k];
                    end else begin
                        total_length <= total_length + (current_right - current_left);
                        current_left <= intervals_left[k]; current_right <= intervals_right[k];
                    end
                    k <= k + 1;
                end else begin
                    total_length <= total_length + (current_right - current_left);
                    covered_length <= total_length;
                    done <= 1; state <= IDLE;
                end
                default: state <= IDLE;
            endcase
        end
    end
endmodule
