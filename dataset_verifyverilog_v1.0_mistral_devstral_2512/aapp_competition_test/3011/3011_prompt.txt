module hill_number_counter(
    input clk,
    input rst_n,
    input start,
    input [31:0] number,  // 8 digits packed as 8x4-bit (each digit 0-9)
    input [3:0] digit_count,  // actual number of digits (1-8)
    output reg [31:0] result,
    output reg done
);

// State machine for checking hill number and counting
reg [3:0] state;
reg [3:0] digit_pos;
reg [3:0] peak_pos;
reg [3:0] current_digit;
reg [3:0] prev_digit;
reg has_fallen;
reg [31:0] count;
reg [3:0] check_pos;
reg [3:0] check_state;  // 0=checking, 1=valid, 2=invalid

// Extract digits from packed number
function [3:0] get_digit;
    input [31:0] packed_num;
    input [3:0] pos;
    begin
        get_digit = packed_num >> (pos * 4) & 4'hF;
    end
endfunction

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= 4'd0;
        done <= 0;
        result <= 0;
        count <= 0;
        check_state <= 0;
    end else begin
        case (state)
            4'd0: begin  // Idle
                if (start) begin
                    state <= 4'd1;
                    digit_pos <= 0;
                    has_fallen <= 0;
                    check_state <= 0;
                    count <= 0;
                end
            end
            
            4'd1: begin  // Check if hill number
                if (digit_pos < digit_count - 1) begin
                    current_digit <= get_digit(number, digit_pos);
                    prev_digit <= get_digit(number, digit_pos - 1);
                    
                    if (digit_pos == 0) begin
                        // First digit, just move on
                        digit_pos <= digit_pos + 1;
                    end else begin
                        if (current_digit > prev_digit) begin
                            if (has_fallen) begin
                                check_state <= 2;  // Invalid
                                state <= 4'd8;    // Go to done
                            end else begin
                                digit_pos <= digit_pos + 1;
                            end
                        end else if (current_digit < prev_digit) begin
                            has_fallen <= 1;
                            digit_pos <= digit_pos + 1;
                        end else begin
                            digit_pos <= digit_pos + 1;
                        end
                    end
                end else begin
                    if (check_state != 2) check_state <= 1;  // Valid
                    state <= 4'd2;  // Go to counting
                    digit_pos <= 0;
                end
            end
            
            4'd2: begin  // Count hill numbers <= number
                if (check_state == 2) begin
                    result <= 32'hFFFFFFFF;  // -1
                    done <= 1;
                    state <= 4'd9;
                end else if (digit_pos < digit_count) begin
                    // Use digit DP to count
                    state <= 4'd3;
                    check_pos <= 0;
                    peak_pos <= 0;
                end else begin
                    result <= count;
                    done <= 1;
                    state <= 4'd9;
                end
            end
            
            4'd3: begin  // Count numbers with current prefix
                // This is simplified - would need full DP in real implementation
                // For now, count manually for small numbers
                if (check_pos < 8) begin
                    // Count valid numbers up to 8 digits
                    // This would be expanded in real implementation
                    count <= count + 1;  // Simplified counting
                    check_pos <= check_pos + 1;
                end else begin
                    digit_pos <= digit_pos + 1;
                    state <= 4'd2;
                end
            end
            
            4'd8: begin  // Invalid, set result to -1
                result <= 32'hFFFFFFFF;
                done <= 1;
                state <= 4'd9;
            end
            
            4'd9: begin  // Wait for start to go low
                if (!start) begin
                    done <= 0;
                    state <= 4'd0;
                end
            end
        endcase
    end
end

endmodule
