module WatchLaterMinClicks (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [7:0] video_types [0:15],  // ASCII characters, 16 videos max
    input wire [4:0] length,              // Number of videos (1-16)
    output reg [7:0] result,
    output reg done
);

// Parameters
parameter MAX_VIDEOS = 16;
parameter MAX_TYPES = 8;
parameter STATE_IDLE = 3'b000;
parameter STATE_COUNT_SEGMENTS = 3'b001;
parameter STATE_FIND_MIN = 3'b010;
parameter STATE_REMOVE_TYPE = 3'b011;
parameter STATE_UPDATE_RESULT = 3'b100;
parameter STATE_DONE = 3'b101;

// Registers
reg [2:0] state;
reg [7:0] current_videos [0:15];  // Current remaining videos
reg [4:0] remaining_length;
reg [7:0] segment_counts [0:7];    // Segment counts for each type (0-7, mapped from ASCII)
reg [7:0] min_segments;
reg [7:0] selected_type;
reg [3:0] type_index;              // Iterating through types
reg [4:0] video_index;             // Iterating through videos
reg [7:0] temp_result;
reg [7:0] last_type;
reg counting_done;

// Helper: Map ASCII to type index (0-7)
function automatic [2:0] char_to_type(input [7:0] c);
    begin
        // Map 'a'..'h' to 0..7 (simple mapping for small k)
        char_to_type = c[2:0];  // Use lower 3 bits for type index
    end
endfunction

// Helper: Check if two types are same
function automatic is_same_type(input [7:0] a, input [7:0] b);
    begin
        is_same_type = (a == b);
    end
endfunction

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= STATE_IDLE;
        result <= 8'b0;
        done <= 1'b0;
        remaining_length <= 5'b0;
        temp_result <= 8'b0;
        type_index <= 4'b0;
        video_index <= 5'b0;
        min_segments <= 8'hFF;
        selected_type <= 8'b0;
        last_type <= 8'b0;
        counting_done <= 1'b0;
    end else begin
        case (state)
            STATE_IDLE: begin
                done <= 1'b0;
                if (start && length > 0) begin
                    // Initialize current_videos from input
                    for (integer i = 0; i < MAX_VIDEOS; i = i + 1) begin
                        if (i < length)
                            current_videos[i] <= video_types[i];
                        else
                            current_videos[i] <= 8'hFF;  // Invalid marker
                    end
                    remaining_length <= length;
                    temp_result <= 8'b0;
                    state <= STATE_COUNT_SEGMENTS;
                    type_index <= 4'b0;
                    video_index <= 5'b0;
                    counting_done <= 1'b0;
                end
            end

            STATE_COUNT_SEGMENTS: begin
                // Count segments for current type_index
                if (type_index < MAX_TYPES) begin
                    if (!counting_done) begin
                        // Reset segment count for this type
                        segment_counts[type_index] <= 8'b0;
                        video_index <= 5'b0;
                        last_type <= 8'hFF;
                        counting_done <= 1'b1;
                    end else if (video_index < remaining_length) begin
                        if (current_videos[video_index] != 8'hFF) begin
                            if (char_to_type(current_videos[video_index]) == type_index[2:0]) begin
                                // This video is of current type
                                if (video_index == 0 || 
                                    last_type == 8'hFF || 
                                    !is_same_type(current_videos[video_index-1], current_videos[video_index])) begin
                                    // Start of new segment
                                    segment_counts[type_index] <= segment_counts[type_index] + 1;
                                end
                                last_type <= current_videos[video_index];
                            end
                        end
                        video_index <= video_index + 1;
                    end else begin
                        counting_done <= 1'b0;
                        type_index <= type_index + 1;
                    end
                end else begin
                    // Done counting all types
                    type_index <= 4'b0;
                    state <= STATE_FIND_MIN;
                    min_segments <= 8'hFF;
                    selected_type <= 8'hFF;
                end
            end

            STATE_FIND_MIN: begin
                // Find type with minimum segments (but > 0)
                if (type_index < MAX_TYPES) begin
                    if (segment_counts[type_index] > 0 && segment_counts[type_index] < min_segments) begin
                        min_segments <= segment_counts[type_index];
                        selected_type <= type_index[2:0];
                    end
                    type_index <= type_index + 1;
                end else begin
                    if (min_segments != 8'hFF && remaining_length > 0) begin
                        // Remove videos of selected_type
                        video_index <= 5'b0;
                        state <= STATE_REMOVE_TYPE;
                    end else begin
                        // No videos left or all removed
                        state <= STATE_DONE;
                    end
                end
            end

            STATE_REMOVE_TYPE: begin
                // Remove all videos of selected_type from current_videos
                if (video_index < remaining_length) begin
                    if (current_videos[video_index] != 8'hFF && 
                        char_to_type(current_videos[video_index]) == selected_type[2:0]) begin
                        // Remove this video by shifting remaining videos left
                        for (integer i = video_index; i < remaining_length - 1; i = i + 1) begin
                            current_videos[i] <= current_videos[i + 1];
                        end
                        current_videos[remaining_length - 1] <= 8'hFF;
                        remaining_length <= remaining_length - 1;
                        // Stay at same index after shift
                    end else begin
                        video_index <= video_index + 1;
                    end
                end else begin
                    state <= STATE_UPDATE_RESULT;
                end
            end

            STATE_UPDATE_RESULT: begin
                temp_result <= temp_result + min_segments;
                state <= STATE_COUNT_SEGMENTS;
                type_index <= 4'b0;
                video_index <= 5'b0;
                counting_done <= 1'b0;
            end

            STATE_DONE: begin
                result <= temp_result;
                done <= 1'b1;
                state <= STATE_IDLE;
            end

            default: state <= STATE_IDLE;
        endcase
    end
end

endmodule
