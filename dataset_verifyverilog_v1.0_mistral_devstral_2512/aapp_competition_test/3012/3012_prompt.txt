module move_sequences (
    input wire [63:0] a11,
    input wire [63:0] a12,
    input wire [63:0] a21,
    input wire [63:0] a22,
    input wire [63:0] N,
    output reg [63:0] result
);

    localparam MOD = 1000000007;

    // Helper function for modular exponentiation
    function [63:0] mod_pow;
        input [63:0] base;
        input [63:0] exp;
        reg [63:0] result;
        reg [63:0] b;
        reg [63:0] e;
        integer i;
    begin
        result = 1;
        b = base % MOD;
        e = exp;
        for (i = 0; i < 64; i = i + 1) begin
            if (e == 0) begin
                // break loop early
            end else begin
                if (e[0] == 1)
                    result = (result * b) % MOD;
                b = (b * b) % MOD;
                e = e >> 1;
            end
        end
        mod_pow = result;
    end
    endfunction

    always @(*) begin
        reg [63:0] P;
        reg [63:0] part1;
        reg [63:0] part2;
        
        // Check if all players have the same type
        if ((a11 > 0 && a12 == 0 && a21 == 0 && a22 == 0) ||
            (a12 > 0 && a11 == 0 && a21 == 0 && a22 == 0) ||
            (a21 > 0 && a11 == 0 && a12 == 0 && a22 == 0) ||
            (a22 > 0 && a11 == 0 && a12 == 0 && a21 == 0)) begin
            // Determine P
            if (a11 > 0) P = a11;
            else if (a12 > 0) P = a12;
            else if (a21 > 0) P = a21;
            else P = a22;
            
            if (N == 0) begin
                result = 1;
            end else if (P < 2) begin
                result = 0;
            end else if (N == 1) begin
                result = (P * (P - 1)) % MOD;
            end else begin
                part1 = (P * (P - 1)) % MOD;
                part2 = mod_pow(P - 2, N - 1);
                result = (part1 * part2) % MOD;
            end
        end else begin
            // Mixed types - output 0 (simplified)
            result = 0;
        end
    end

endmodule
