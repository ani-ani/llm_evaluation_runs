module schedule_generator (
    input clk,
    input rst_n,
    input start,
    input [4:0] n, // players per team (1-25)
    input [4:0] m, // teams (2-25)
    output reg valid,
    output reg [9:0] round,
    output reg [9:0] p1, // player1 ID: {team[4:0], player[4:0]}
    output reg [9:0] p2, // player2 ID
    output reg done
);

// Implementation notes:
// - The module implements a greedy edge coloring algorithm for the complete m-partite graph.
// - It uses a dual-port BRAM to store a 100x100 played matrix (each entry 1 bit).
// - Player ID = team * n + player (0-indexed).
// - When start is high, the module initializes the BRAM and begins scheduling.
// - For each round, it scans all players and tries to schedule a game with an available opponent.
// - Once no more games can be added to the current round, it outputs all scheduled games.
// - The output is streamed: for each game in the round, valid is high for one cycle.
// - After all games are scheduled, done is asserted.
// - Total rounds will be at most (m-1)*n + 1.
