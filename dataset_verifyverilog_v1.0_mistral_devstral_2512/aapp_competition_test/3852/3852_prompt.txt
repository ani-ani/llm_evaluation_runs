module non_decreasing_array #(
    parameter DATA_WIDTH = 24,
    parameter MAX_N = 50,
    parameter INDEX_WIDTH = 6
) (
    input clk,
    input rst_n,
    input start,
    input [5:0] N,
    input [DATA_WIDTH*MAX_N-1:0] array_in,
    output reg count_valid,
    output reg [7:0] count,
    output reg op_valid,
    output reg [INDEX_WIDTH-1:0] op_x,
    output reg [INDEX_WIDTH-1:0] op_y,
    output reg done
);

// State definitions
localparam IDLE = 0;
localparam FIND_MAX_MIN = 1;
localparam DECIDE = 2;
localparam OUTPUT_COUNT = 3;
localparam FIX = 4;
localparam PREFIX = 5;
localparam SUFFIX = 6;
localparam DONE = 7;

reg [2:0] state;
reg [DATA_WIDTH-1:0] array_reg [0:MAX_N-1];
reg [DATA_WIDTH-1:0] max_val, min_val;
reg [INDEX_WIDTH-1:0] max_index, min_index;
reg [6:0] neg_count, pos_count;
reg [6:0] total_count;
reg [1:0] strategy;
reg [INDEX_WIDTH-1:0] i;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        count_valid <= 0;
        op_valid <= 0;
        done <= 0;
        i <= 0;
        count <= 0;
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    state <= FIND_MAX_MIN;
                    i <= 0;
                    max_val <= {1'b0, {DATA_WIDTH-1{1'b1}}};
                    min_val <= {1'b1, {DATA_WIDTH-1{1'b0}}};
                    max_index <= 0;
                    min_index <= 0;
                    neg_count <= 0;
                    pos_count <= 0;
                end
            end

            FIND_MAX_MIN: begin
                if (i == 0) begin
                    for (integer j = 0; j < MAX_N; j = j + 1) begin
                        array_reg[j] <= array_in[DATA_WIDTH*j +: DATA_WIDTH];
                    end
                    if (N > 0) begin
                        max_val <= array_in[0 +: DATA_WIDTH];
                        min_val <= array_in[0 +: DATA_WIDTH];
                        if (array_in[0 +: DATA_WIDTH] < 0)
                            neg_count <= 1;
                        else if (array_in[0 +: DATA_WIDTH] > 0)
                            pos_count <= 1;
                    end
                    i <= 1;
                end else if (i < N) begin
                    if (array_reg[i] > max_val) begin
                        max_val <= array_reg[i];
                        max_index <= i;
                    end
                    if (array_reg[i] < min_val) begin
                        min_val <= array_reg[i];
                        min_index <= i;
                    end
                    if (array_reg[i] < 0)
                        neg_count <= neg_count + 1;
                    else if (array_reg[i] > 0)
                        pos_count <= pos_count + 1;
                    i <= i + 1;
                end else begin
                    state <= DECIDE;
                    i <= 0;
                end
            end

            DECIDE: begin
                if (min_val >= 0) begin
                    strategy <= 0;
                    total_count <= N - 1;
                end else if (max_val <= 0) begin
                    strategy <= 1;
                    total_count <= N - 1;
                end else if (max_val >= -min_val) begin
                    strategy <= 2;
                    total_count <= neg_count + (N - 1);
                end else begin
                    strategy <= 3;
                    total_count <= pos_count + (N - 1);
                end
                state <= OUTPUT_COUNT;
            end

            OUTPUT_COUNT: begin
                count_valid <= 1;
                count <= total_count;
                case (strategy)
                    0: state <= PREFIX;
                    1: state <= SUFFIX;
                    2, 3: state <= FIX;
                    default: state <= DONE;
                endcase
            end

            FIX: begin
                count_valid <= 0;
                if (i < N) begin
                    if ((strategy == 2 && array_reg[i] < 0) || (strategy == 3 && array_reg[i] > 0)) begin
                        op_valid <= 1;
                        if (strategy == 2) begin
                            op_x <= max_index + 1;
                            op_y <= i + 1;
                        end else begin
                            op_x <= min_index + 1;
                            op_y <= i + 1;
                        end
                    end else begin
                        op_valid <= 0;
                    end
                    i <= i + 1;
                end else begin
                    op_valid <= 0;
                    i <= 0;
                    if (strategy == 2)
                        state <= PREFIX;
                    else
                        state <= SUFFIX;
                end
            end

            PREFIX: begin
                if (i < N - 1) begin
                    op_valid <= 1;
                    op_x <= i + 1;
                    op_y <= i + 2;
                    i <= i + 1;
                end else begin
                    op_valid <= 0;
                    state <= DONE;
                end
            end

            SUFFIX: begin
                if (i < N - 1) begin
                    op_valid <= 1;
                    op_x <= (N - 1 - i) + 1;
                    op_y <= (N - 1 - i);
                    i <= i + 1;
                end else begin
                    op_valid <= 0;
                    state <= DONE;
                end
            end

            DONE: begin
                done <= 1;
                state <= IDLE;
            end
        endcase
    end
end

endmodule
