// Verilog module for combinatorial problem: number of ways to take K items from N types with M copies each.
// Computes modulo 1000007.
// Scaled to N, M, K <= 8 (4-bit inputs).
// Interface:
module combinatorial (
    input wire clk,          // Clock
    input wire rst_n,        // Active-low reset
    input wire start,        // Start pulse (1 cycle)
    input wire [3:0] N,      // Number of types (1-8)
    input wire [3:0] M,      // Max copies per type (1-8)
    input wire [3:0] K,      // Number of items to take (0-8)
    output reg [31:0] result, // Result modulo 1000007
    output reg done          // Done signal (1 cycle when result ready)
);

// Behavior:
// - On reset (rst_n=0), result=0, done=0, state machine idle.
// - On start=1, begins computation. The computation uses dynamic programming.
// - State machine states: IDLE, SETUP, COMPUTE, DONE.
// - In SETUP, initialize dp_prev[0] = 1, dp_prev[j>0] = 0.
// - In COMPUTE, iterate i from 1 to N:
//      For each j from 0 to K:
//          dp_curr[j] = sum_{c=0}^{min(M, j)} dp_prev[j-c] mod MOD
//      Then swap dp_prev and dp_curr.
// - After i = N, answer = dp_prev[K].
// - Transition to DONE, assert done for one cycle, then back to IDLE.
// - The computation takes at most N*(K+1)*(M+1) clock cycles.
// - MOD = 1000007.
// - All internal counters and registers should be sized appropriately.
// - The DP arrays dp_prev and dp_curr are of size K+1 (max 9), each element 32 bits.
// - The module should not use any external memory.

// Implementation details:
// - Use a 3-bit counter for i (max 8), 4-bit counter for j (max 8), 4-bit counter for c (max 8).
// - Use a temporary register to accumulate sum.
// - When done is asserted, result holds the answer.
// - If start is asserted while busy, it may be ignored (assumption: testbench ensures no conflict).

// Example: N=1, M=5, K=3 => result=1.
