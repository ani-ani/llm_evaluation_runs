module string_constructor (
    input clk,
    input rst_n,
    input start,
    input [15:0] a00,
    input [15:0] a01,
    input [15:0] a10,
    input [15:0] a11,
    output reg [7:0] char,
    output reg valid,
    output reg done
);

// Parameters for maximum constraints
parameter MAX_LEN = 16;
parameter MAX_N = 16;

// State definitions
parameter IDLE = 3'b000;
parameter COMPUTE_N = 3'b001;
parameter CHECK_COND = 3'b010;
parameter BUILD_STRING = 3'b011;
parameter OUTPUT_STRING = 3'b100;
parameter FINISHED = 3'b101;

// Internal registers
reg [2:0] state, next_state;
reg [4:0] n0, n1;  // Count of 0s and 1s
reg [4:0] i, j;    // Loop counters
reg [4:0] gap [0:MAX_N];  // Gap distribution
reg [7:0] string_buffer [0:MAX_LEN-1];  // Output string buffer
reg [4:0] buf_len;  // Actual string length
reg [4:0] out_idx;  // Output index

// Helper function: Check if value is triangular number
function [4:0] find_n;
    input [15:0] val;
    reg [15:0] tri;
    integer k;
    begin
        find_n = 0;
        for (k = 0; k <= MAX_N; k = k + 1) begin
            tri = k * (k - 1) / 2;
            if (tri == val) begin
                find_n = k;
            end
        end
    end
endfunction

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
    end else begin
        state <= next_state;
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: begin
            if (start) next_state = COMPUTE_N;
        end
        COMPUTE_N: begin
            next_state = CHECK_COND;
        end
        CHECK_COND: begin
            // If valid, go to build, else finish
            if ((n0 * n1 == a01 + a10) && (n0 + n1 <= MAX_LEN)) begin
                next_state = BUILD_STRING;
            end else begin
                next_state = FINISHED;
            end
        end
        BUILD_STRING: begin
            if (i >= buf_len) begin
                next_state = OUTPUT_STRING;
            end else begin
                next_state = BUILD_STRING;
            end
        end
        OUTPUT_STRING: begin
            if (out_idx >= buf_len) begin
                next_state = FINISHED;
            end else begin
                next_state = OUTPUT_STRING;
            end
        end
        FINISHED: next_state = FINISHED;
        default: next_state = IDLE;
    endcase
end

// Output logic and operations
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        char <= 8'b0;
        valid <= 1'b0;
        done <= 1'b0;
        n0 <= 0;
        n1 <= 0;
        i <= 0;
        j <= 0;
        buf_len <= 0;
        out_idx <= 0;
    end else begin
        case (state)
            IDLE: begin
                char <= 8'b0;
                valid <= 1'b0;
                done <= 1'b0;
            end
            
            COMPUTE_N: begin
                // Compute n0 and n1 from triangular numbers
                n0 <= find_n(a00);
                n1 <= find_n(a11);
                // Handle special case where all are zero
                if (a00 == 0 && a01 == 0 && a10 == 0 && a11 == 0) begin
                    n0 <= 1;
                    n1 <= 0;
                end
            end
            
            CHECK_COND: begin
                // Reset counters
                i <= 0;
                j <= 0;
                buf_len <= 0;
            end
            
            BUILD_STRING: begin
                if (i == 0) begin
                    // Initialize gap array for distribution
                    // Simple greedy distribution: place zeros in gaps
                    gap[0] <= n0;
                    for (integer k = 1; k <= MAX_N; k = k + 1) begin
                        gap[k] <= 0;
                    end
                    // Distribute zeros based on a01 value
                    if (n1 > 0 && n0 > 0) begin
                        gap[0] <= a01 / n1;
                        gap[1] <= n0 - (a01 / n1);
                    end
                end
                
                // Build string buffer
                if (i < n1) begin
                    // Add zeros for this gap
                    if (j < gap[i]) begin
                        string_buffer[buf_len] <= "0";
                        buf_len <= buf_len + 1;
                        j <= j + 1;
                    end else begin
                        // Add one after zeros
                        string_buffer[buf_len] <= "1";
                        buf_len <= buf_len + 1;
                        i <= i + 1;
                        j <= 0;
                    end
                end else if (i == n1 && j < gap[n1]) begin
                    // Add remaining zeros
                    string_buffer[buf_len] <= "0";
                    buf_len <= buf_len + 1;
                    j <= j + 1;
                end else if (i == n1) begin
                    // Add final one if needed (rare case)
                    if (n1 > 0 && buf_len > 0 && string_buffer[buf_len-1] != "1") begin
                        string_buffer[buf_len] <= "1";
                        buf_len <= buf_len + 1;
                    end
                    i <= i + 1;  // Mark complete
                end
                
                // Special case: all zeros
                if (n1 == 0 && n0 > 0 && i < n0) begin
                    string_buffer[buf_len] <= "0";
                    buf_len <= buf_len + 1;
                    i <= i + 1;
                end
                // Special case: all ones  
                if (n0 == 0 && n1 > 0 && i < n1) begin
                    string_buffer[buf_len] <= "1";
                    buf_len <= buf_len + 1;
                    i <= i + 1;
                end
            end
            
            OUTPUT_STRING: begin
                if (out_idx < buf_len) begin
                    char <= string_buffer[out_idx];
                    valid <= 1'b1;
                    out_idx <= out_idx + 1;
                end else begin
                    valid <= 1'b0;
                    done <= 1'b1;
                end
            end
            
            FINISHED: begin
                done <= 1'b1;
                valid <= 1'b0;
            end
        endcase
    end
end

endmodule
