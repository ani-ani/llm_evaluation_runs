module best_friend_pairs (
    input [1:0] n,   // n digit count, 2-bit: 00 -> 1, 01 -> 2, 10 -> 3, 11 -> reserved
    output [29:0] result  // result modulo 998244353
);

// Lookup table for n=1,2,3
reg [29:0] result_reg;
always @(*) begin
    case(n)
        2'b00: result_reg = 10;      // n=1
        2'b01: result_reg = 570;     // n=2
        2'b10: result_reg = 46242;   // n=3
        default: result_reg = 0;     // invalid input
    endcase
end

assign result = result_reg;

endmodule
