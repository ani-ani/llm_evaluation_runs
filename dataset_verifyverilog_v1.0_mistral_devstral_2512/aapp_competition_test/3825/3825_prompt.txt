module roman_digit_count(
    input [31:0] n,
    output reg [63:0] result
);
    // Lookup table for n <= 11
    always @(*) begin
        if (n >= 12)
            result = 49 * n - 247;
        else begin
            case (n)
                1: result = 4;
                2: result = 10;
                3: result = 20;
                4: result = 35;
                5: result = 56;
                6: result = 83;
                7: result = 116;
                8: result = 155;
                9: result = 198;
                10: result = 244;
                11: result = 292;
                default: result = 0;
            endcase
        end
    end
endmodule
