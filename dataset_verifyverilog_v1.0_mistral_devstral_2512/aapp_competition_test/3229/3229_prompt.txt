module WorstRank (
    input wire clk,
    input wire rst_n,
    input wire start,                // Assert to reset and start a new computation
    input wire [3:0] n_minus_one,    // Number of scores per contestant (n‑1)
    input wire scores_valid,         // High when scores_in is valid
    input wire [7:0] scores_in,      // Score value for current contestant
    input wire is_your_score,        // High if this score belongs to the first contestant (you)
    input wire end_of_contestants,   // High when this is the last score of the last contestant
    output reg [7:0] worst_rank,     // Result: worst possible rank
    output reg done                  // High when computation is complete
);

// Internal state and registers
reg [7:0] your_scores[0:9];         // Store your scores (max 9)
reg [7:0] other_scores[0:9];        // Store current other contestant's scores
reg [7:0] counter;                 // Count of contestants that can beat you
reg [3:0] score_index;             // Index for reading scores per contestant
reg [3:0] contestants_read;        // Number of other contestants processed
reg processing_you;                // Flag indicating we are reading your scores
reg compute_done;                  // Internal done flag

// Computed values for your aggregate
reg [15:0] your_aggregate;         // Sum of top 4 scores (or all if <4)

// Computed values for other contestant
reg [15:0] other_aggregate;        // Sum of top 4 scores (or all if <4)
reg [7:0] fourth_highest;          // 4th highest score (or 0 if <4)
reg [15:0] needed_score;           // Minimum score needed to beat you

// Helper task to compute top‑4 sum and fourth highest from a score array
function automatic void compute_top4(
    input reg [7:0] scores[0:9],
    input reg [3:0] len,
    output reg [15:0] sum,
    output reg [7:0] fourth
);
    integer i, j;
    reg [7:0] sorted[0:9];
    // Copy and sort descending using bubble sort (len ≤ 9)
    for (i = 0; i < len; i++) sorted[i] = scores[i];
    for (i = 0; i < len-1; i++) begin
        for (j = 0; j < len-i-1; j++) begin
            if (sorted[j] < sorted[j+1]) begin
                sorted[j] = sorted[j] ^ sorted[j+1];
                sorted[j+1] = sorted[j] ^ sorted[j+1];
                sorted[j] = sorted[j] ^ sorted[j+1];
            end
        end
    end
    // Compute sum of top min(4,len) scores
    sum = 0;
    for (i = 0; i < (len < 4 ? len : 4); i++) sum = sum + sorted[i];
    // Fourth highest is the 4th element if len ≥4, else 0
    fourth = (len >= 4) ? sorted[3] : 8'd0;
endfunction

// Sequential logic
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        // Reset all registers
        worst_rank <= 8'd0;
        done <= 1'b0;
        counter <= 8'd0;
        score_index <= 4'd0;
        contestants_read <= 4'd0;
        processing_you <= 1'b1;
        compute_done <= 1'b0;
        your_aggregate <= 16'd0;
        needed_score <= 16'd0;
    end else if (start) begin
        // Start a new computation: reset counters and flags
        worst_rank <= 8'd0;
        done <= 1'b0;
        counter <= 8'd0;
        score_index <= 4'd0;
        contestants_read <= 4'd0;
        processing_you <= 1'b1;
        compute_done <= 1'b0;
        your_aggregate <= 16'd0;
        needed_score <= 16'd0;
    end else if (scores_valid && !compute_done) begin
        // Store incoming score
        if (processing_you) begin
            your_scores[score_index] <= scores_in;
        end else begin
            other_scores[score_index] <= scores_in;
        end
        score_index <= score_index + 1;
        // When all scores for current contestant are read
        if (score_index + 1 == n_minus_one) begin
            score_index <= 0;
            if (processing_you) begin
                // Finished reading your scores: compute your aggregate
                compute_top4(your_scores, n_minus_one, your_aggregate, fourth_highest);
                processing_you <= 1'b0;
            end else begin
                // Finished reading other contestant's scores: compute needed score
                // Step 1: compute other_aggregate and fourth_highest
                compute_top4(other_scores, n_minus_one, other_aggregate, fourth_highest);
                // Step 2: calculate needed_score
                if (n_minus_one < 4) begin
                    // If fewer than 4 scores, aggregate = sum + X
                    // Need sum + X > your_aggregate => X > your_aggregate - sum
                    needed_score <= (your_aggregate >= other_aggregate) ? (your_aggregate - other_aggregate + 1) : 16'd0;
                end else begin
                    // If 4 or more scores
                    if (other_aggregate > your_aggregate) begin
                        needed_score <= 16'd0; // Already beats you
                    end else begin
                        // Needed X = max(fourth_highest + 1, your_aggregate - other_aggregate + fourth_highest + 1)
                        // Because X must be > fourth_highest to replace it, and must satisfy other_aggregate - fourth_highest + X > your_aggregate
                        reg [15:0] val1 = fourth_highest + 1;
                        reg [15:0] val2 = your_aggregate - other_aggregate + fourth_highest + 1;
                        needed_score <= (val1 > val2) ? val1 : val2;
                    end
                end
                // Step 3: increment counter if needed_score ≤ 101
                if (needed_score <= 16'd101) counter <= counter + 1;
                // Step 4: prepare for next contestant or finish
                contestants_read <= contestants_read + 1;
            end
        end
        // If end_of_contestants is asserted and we just finished a contestant
        if (end_of_contestants && score_index == n_minus_one - 1) begin
            // Wait for the next clock cycle to finalize the last contestant
        end
    end else if (end_of_contestants && !compute_done && score_index == 4'd0) begin
        // All contestants have been read; finalize result
        worst_rank <= counter + 1;
        done <= 1'b1;
        compute_done <= 1'b1;
    end
end

endmodule
