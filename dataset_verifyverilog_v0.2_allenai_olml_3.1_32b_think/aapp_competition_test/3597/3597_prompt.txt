module PoolShark (
    input wire clk,
    input wire rst_n,
    input wire start,
    // Inputs: w, l, r, x1, y1, x2, y2, x3, y3, h. Passed 2 per cycle for area optimization.
    input wire [15:0] data_in,
    input wire [1:0] data_valid, // 2-bit mask indicates which inputs are valid
    output reg [15:0] d_out,     // distance d in hundredths (rounded)
    output reg [15:0] theta_out, // angle theta in hundredths of a degree (rounded)
    output reg done,
    output reg impossible
);

    // --- Fixed Point Math (Q16.16) ---
    // Max value 120 * 65536 = 7,864,320 < 2^23. 32-bit signed is sufficient.
    // For area, we reuse the multiplier.
    
    reg [4:0] input_cnt;
    reg [5:0] state;
    
    // Storage Registers (Q16.16)
    reg signed [31:0] w_reg, l_reg, r_reg, h_reg;
    reg signed [31:0] x1, y1, x2, y2, x3, y3;
    
    // Intermediate Calculation Registers
    reg signed [31:0] in_A, in_B;   // Inputs to multiplier
    wire signed [63:0] mul_out;     // 64-bit result
    
    // Multiplier (Sequential logic implied, but since we are in one module, 
    // we will model a single-cycle multiplier for simplicity in this generated code,
    // assuming synthesis tools handle it. Or we can add a pipeline stage if needed.
    // To keep state count low, let's assume combinational multiplier block logic,
    // but in RTL we would register it. Here we simulate a registered output per state).
    
    assign mul_out = in_A * in_B;
    
    // Helper registers
    reg signed [31:0] val1, val2; // Temp storage
    reg signed [31:0] dx, dy;     // Current vector
    reg signed [31:0] t;          // Time/Distance parameter
    reg signed [31:0] x_start, y_start; // Start position for ray trace
    reg signed [31:0] x_target, y_target; // Target position
    reg signed [31:0] x_hole, y_hole;     // Hole position
    
    // Rounding constants
    localparam signed [31:0] HALF = 32'h00008000; // 0.5 in Q16.16

    // State Definitions
    localparam S_IDLE = 0;
    localparam S_LOAD = 1;
    localparam S_WAIT_CALC = 2;
    
    // Phase 1: Cue -> Ball 1
    localparam P1_START = 3;
    localparam P1_CHECK_DIST = 4;
    localparam P1_CALC_DIR = 5;
    localparam P1_CALC_DIR_WAIT = 6;
    localparam P1_REFLECT = 7;
    localparam P1_REFLECT_WAIT = 8;
    
    // Phase 2: Ball 1 -> Ball 3 (Sink 3)
    localparam P2_START = 9;
    localparam P2_CHECK_DIST = 10;
    localparam P2_SOLVE = 11;
    localparam P2_SOLVE_WAIT = 12;
    
    // Phase 3: Cue -> Ball 2 (Sink 2) - Checks if reflection angle works
    localparam P3_START = 13;
    localparam P3_CHECK_DIST = 14;
    localparam P3_SOLVE = 15;
    localparam P3_SOLVE_WAIT = 16;
    
    localparam S_DONE = 17;
    localparam S_IMPOSSIBLE = 18;

    // Integer conversion helpers (output rounding)
    reg signed [31:0] raw_val;
    reg signed [31:0] rounded;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= S_IDLE;
            input_cnt <= 0;
            done <= 0;
            impossible <= 0;
        end else begin
            case (state)
                S_IDLE: begin
                    done <= 0;
                    impossible <= 0;
                    if (start) begin
                        state <= S_LOAD;
                        input_cnt <= 0;
                    end
                end

                S_LOAD: begin
                    // Load 8 inputs (2 per clock) + h is separate in data stream
                    if (data_valid[0]) begin
                        case (input_cnt)
                            0: w_reg <= {16'h0, data_in};
                            1: l_reg <= {16'h0, data_in};
                            2: r_reg <= {16'h0, data_in};
                            3: x1 <= {16'h0, data_in};
                            4: y1 <= {16'h0, data_in};
                            5: x2 <= {16'h0, data_in};
                            6: y2 <= {16'h0, data_in};
                            7: x3 <= {16'h0, data_in};
                            8: y3 <= {16'h0, data_in};
                            9: h_reg <= {16'h0, data_in};
                        endcase
                        input_cnt <= input_cnt + 1;
                    end
                    if (input_cnt == 10 && data_valid[0]) begin
                        // All inputs loaded
                        // Start Phase 1
                        // d iterates from r to w-r with step 0.01. 
                        // To save area, we will scan d. 
                        // Since we need specific d, we can try to solve or scan.
                        // Given the complex reflection, we will search.
                        // Scan d. Set d = r.
                        val1 <= r_reg; // d = r
                        state <= P1_START;
                    end
                end

                // --- PHASE 1: Cue -> Ball 1 ---
                P1_START: begin
                    // Check if val1 (d) < w - r. 
                    in_A <= w_reg;
                    in_B <= 32'h00010000; // 1.0
                    state <= P1_CHECK_DIST;
                end
                
                P1_CHECK_DIST: begin
                    // w - r
                    val2 <= w_reg - r_reg;
                    // Check if d < w - r
                    // If val1 >= val2, fail.
                    if (val1 >= (w_reg - r_reg)) begin
                        state <= S_IMPOSSIBLE;
                    end else begin
                        // Calculate Cue Start Position: (val1, h_reg)
                        x_start <= val1;
                        y_start <= h_reg;
                        
                        // Calculate direction to Ball 1
                        // dx = x1 - x_start, dy = y1 - y_start
                        // We also need length for time t
                        dx <= x1 - val1;
                        dy <= y1 - h_reg;
                        state <= P1_CALC_DIR;
                    end
                end

                P1_CALC_DIR: begin
                    // Start calculation of (x1 - x_start)^2 + (y1 - y_start)^2
                    in_A <= dx;
                    in_B <= dx;
                    state <= P1_CALC_DIR_WAIT;
                end
                
                P1_CALC_DIR_WAIT: begin
                    // Awaiting dx^2. Store partial result.
                    val1 <= mul_out[47:16]; // dx^2
                    in_A <= dy;
                    in_B <= dy;
                    state <= P1_REFLECT;
                end
                
                P1_REFLECT: begin
                    // Add dy^2 to dx^2 to get dist_sq
                    val1 <= val1 + mul_out[47:16];
                    state <= P1_REFLECT_WAIT;
                end
                
                P1_REFLECT_WAIT: begin
                    // Now we have dist_sq. 
                    // Check collision: dist_sq < 4r^2
                    // We need 4r^2. Let's precompute or calculate.
                    // Since r is constant, let's compute 4r^2 once.
                    // But we need to handle multiple d's.
                    // Let's store 4r^2 in a temp reg during load.
                    // Actually, let's just compute 4r^2 here for safety or pass it.
                    // Optimization: Since inputs are fixed, we can compute 4r^2 in P1_START.
                    // Let's assume we computed 4r^2 in P1_START.
                    // Wait, we didn't. Let's do it now.
                    // Actually, let's just use sqrt if we had it, but we don't. 
                    // We use squared comparison.
                    // 4*r*r
                    // r is in 16.16.
                    in_A <= r_reg;
                    in_B <= r_reg;
                    state <= 19; // Special state to compute 4r^2
                end
                
                19: begin
                    val2 <= mul_out[47:16]; // r^2
                    state <= 20;
                end
                20: begin
                    in_A <= val2;
                    in_B <= 32'h00040000; // 4.0
                    state <= 21;
                end
                21: begin
                    val2 <= mul_out[47:16]; // 4*r^2
                    state <= 22;
                end
                22: begin
                    // Now compare val1 (dist_sq) with val2 (4r^2)
                    // If val1 >= val2, then no collision (or pass through). 
                    // We need to hit ball 1, so strictly < 4r^2? 
                    // In reality, touch = hit. <= 4r^2.
                    if (val1 > val2) begin
                        // No collision, move to next d
                        state <= P1_NEXT_D;
                    end else begin
                        // Collision at t = 1.0 (normalized).
                        // Now calculate impact point (x1, y1).
                        // Calculate normalized vector to Ball 1.
                        // But we need the reflection vector for the Cue ball.
                        // Reflection: V_new = V - 2 * (V . N) * N, where N is normal of Ball 1 at collision.
                        // N = (x1 - x_start, y1 - y_start) / dist.
                        // Impact at ball 1 center (idealized for collision logic).
                        
                        // Calculate V . N. 
                        // V = (dx, dy). N = (dx, dy) / dist.
                        // V . N = dist.
                        // Reflected vector: V - 2*dist*N = V - 2*(dx, dy).
                        // Wait, the problem says: 
                        // 1. Ball 1 moves in direction (B to A) = (x1 - x_cue, y1 - y_cue)
                        // 2. Ball B (Cue) reflects across A's vector.
                        // A's vector is (dx, dy). Normal is perp to (dx, dy).
                        // Let's stick to standard reflection: 
                        // N = (dx, dy) / dist.
                        // R = V - 2 (V . N) N = V - 2 (dist) (V/dist) = V - 2V = -V.
                        // Wait, V and N are parallel? Yes, V is vector to A. 
                        // So reflection is -V (straight back).
                        // But the problem says "reflected across A's new vector".
                        // A's vector is (dx, dy). Reflected across (dx, dy) means
                        // R = 2*proj_{A} V - V.
                        // Since V is parallel to A, proj is V. So R = 2V - V = V.
                        // So Cue keeps going in same direction? 
                        // "Ball B’s new velocity vector is B’s original vector reflected across A’s new vector."
                        // This implies the vector V is reflected across the line of A's motion.
                        // If V is ALONG that line, it stays the same.
                        // If V is PERPENDICULAR, it flips sign.
                        // But V connects centers. So V is normal to the surface.
                        // Wait. In 2D circles, collision is along the line connecting centers.
                        // So V is normal to the tangent.
                        // Standard reflection: bounce back along V. 
                        // Problem reflection: reflect V across V. Result is V.
                        // So Cue ball continues straight?
                        // Re-read: "Ball B’s new velocity vector is B’s original vector reflected across A’s new vector".
                        // Figure H.1 likely shows an angle. 
                        // If A moves in V, and B reflects across V:
                        // Let V be x-axis. B comes in at angle alpha. 
                        // Reflected across x-axis means (x, -y).
                        // So angle becomes -alpha. 
                        // But V is along the line of centers. 
                        // Okay, maybe the geometry is different. 
                        // Let's assume the problem implies a non-trivial reflection.
                        // HOWEVER, if the cue ball hits a stationary ball, standard physics says it stops (if masses equal) or deflects.
                        // The problem says: "B's is not (unless A is glued...)."
                        // If B reflects across A's vector:
                        // V_new = V - 2 * (V . N_A) * N_A.
                        // Where N_A is unit vector of A's new motion.
                        // A's motion is B->A vector (dx, dy).
                        // N_A = (dx, dy)/dist.
                        // V_new = V - 2 * (V . N_A) * N_A.
                        // V is parallel to N_A. V = |V| * N_A.
                        // V . N_A = |V|.
                        // V_new = |V|*N_A - 2|V|*N_A = -|V|*N_A = -V.
                        // So it bounces back.
                        // Let's assume it bounces back. (Straight line).
                        
                        // If it bounces back (x_start - dx, y_start - dy), 
                        // it goes away from ball 2 (which is generally ahead).
                        // The problem says: "ricochets into ball 2".
                        // This implies B does NOT go straight back.
                        // Look at Figure H.1. It is NOT head-on collision? 
                        // The problem says "vector from center of B to center of A at time of collision".
                        // This is the normal.
                        // The reflection is "across A's new vector".
                        // If A's new vector IS the normal (line of centers).
                        // Then reflection is just flipping the component parallel to the normal.
                        // Standard elastic collision: V_cue -> V_cue - 2(V_cue . N)N.
                        // Here B stops (masses equal). 
                        // Wait. "reflected across A's new vector". 
                        // This means the reflection line is the line of A's motion.
                        // So if V_cue is at 30 deg to A, V_cue_new is at -30 deg.
                        // If V_cue is 0 deg (parallel), it stays 0 deg.
                        // But V_cue is normal (90 deg to tangent). 
                        // So V_cue_new is -90 deg (backwards).
                        // This contradicts "ricochets into ball 2".
                        
                        // ALTERNATIVE INTERPRETATION:
                        // Maybe "reflected across A's new vector" means using A's new vector as the normal for reflection?
                        // No, "reflected across" implies mirror line.
                        // What if the mirror line is perpendicular to A's motion?
                        // Then B reflects off the tangent.
                        // If B hits at 30 deg, bounces off tangent at 30 deg -> 120 deg relative to horizontal.
                        // If B hits head-on (normal), it bounces back (180 deg).
                        // This is standard reflection off a wall moving with A.
                        // Wait. A has mass. B transfers momentum.
                        // Let's assume the problem intends a specific trick:
                        // If B hits A off-center, A moves on a line. B reflects across that line.
                        // If B hits A head on, A moves straight. B reflects straight back.
                        // How does B then hit B2? 
                        // It must be that B hits A off-center, A moves, B reflects.
                        // The "off-center" hit implies the line of centers is not parallel to B's incoming velocity?
                        // No, the line of centers IS the direction B->A.
                        // Wait. B moves. A is stationary.
                        // Collision happens when dist = 2r.
                        // At that instant, vector B->A is the normal.
                        // So the reflection is always along the normal.
                        // Unless... B is NOT moving directly at A's center?
                        // In billiards, you can hit the side of a ball.
                        // B moves in direction V.
                        // At collision, the contact point normal is N.
                        // N = (A - B_coll) / |A - B_coll|.
                        // This N is NOT necessarily parallel to V.
                        // It is parallel to the vector connecting centers AT collision.
                        // If B aims to graze A, V is not parallel to (A-B).
                        // Example: B moves parallel to X axis. A is at (X+OFFSET, Y).
                        // B touches A when B.x = A.x - 2r.
                        // Vector (A - B) = (2r, OFFSET).
                        // Normal N = (2r, OFFSET) / dist.
                        // V_new = V - 2(V.N)N.
                        // V = (1, 0).
                        // V_new = (1, 0) - 2 * (2r/dist) * (2r, OFFSET)/dist.
                        // This is the correct physics.
                        // The problem says:
                        // "A begins moving in the direction given by the vector from the center of B to the center of A at the time of the collision."
                        // (This matches physics).
                        // "Ball B’s new velocity vector is B’s original vector reflected across A’s new vector."
                        // "Across" means mirror line. If A's new vector is N.
                        // Reflect V across N. 
                        // If V is parallel to N, stays V. If perpendicular, flips.
                        // This is NOT the standard physics.
                        // Standard physics reflects across the tangent (perp to N).
                        // Here it reflects across N.
                        // V_new = 2*(V.N)*N - V. (Wait, that's reflection across tangent? No, that's V if parallel).
                        // Reflection across N: 
                        // Component parallel to N stays. Perpendicular flips.
                        // V_parallel = (V.N)*N.
                        // V_perp = V - V_parallel.
                        // V_new = V_parallel - V_perp = 2(V.N)N - V.
                        // Okay, this is a valid reflection formula across N.
                        // Standard physics is reflection across tangent: V_new = -V_parallel + V_perp = V - 2(V.N)N.
                        // So the problem swaps the sign of the perpendicular component.
                        // This means if B hits A head-on (V parallel to N), V_new = V (keeps going).
                        // If B hits A glancing (V perp to N), V_new = -V (bounces back).
                        // This is inverted from standard physics (where head-on bounces back, glancing stays).
                        // Okay. We will implement this "inverted" reflection:
                        // V_new = 2 * (V . N) * N - V.
                        // 
                        // Now we need to find a trajectory.
                        // 1. Cue at (d, h). Hits Ball 1. 
                        //    Need to find d and theta such that:
                        //    - Cue hits Ball 1.
                        //    - Cue (after reflection) hits Ball 2 (sinks 2 in top left hole).
                        //    - Ball 1 (after hit) hits Ball 3 (sinks 3 in top right hole).
                        
                        // Given d, we need to find theta.
                        // Let's try to solve for theta.
                        // Let Cue velocity be V.
                        // Contact at Ball 1. Center of B (Cue) is at Contact - r * N.
                        // Wait, collision occurs when dist = 2r.
                        // Center B = Center A - 2r * N.
                        // A moves in direction N (B->A vector).
                        // N = (A - B_coll) / |A - B_coll|.
                        // V_new = 2 * (V . N) * N - V.
                        // 
                        // We can iterate theta.
                        // Theta is on the dashed line? No, "place cue ball on dashed line". 
                        // So start Y = h. X = d.
                        // Theta is angle of shot.
                        // We need to find theta.
                        // Scan theta from -90 to +90 (or full 360).
                        // This is a search problem.
                        // Inputs are small (w, l <= 120). 
                        // We can iterate d (0.01 steps) and theta (0.01 deg).
                        // But that's 12000 * 36000 iterations. Too many.
                        // We need a smarter search.
                        // Or we can use the geometry to solve.
                        // 
                        // Let's try to solve for theta given d.
                        // 1. Cue moves to B1.
                        //    Ray: P(t) = P_start + V * t.
                        //    Intersect circle (B1 center, r=2r? No, collision radius is 2r).
                        //    Distance to B1 center = 2r.
                        //    Let B1 = (x1, y1).
                        //    (x_start + vx*t - x1)^2 + (y_start + vy*t - y1)^2 = (2r)^2.
                        //    This quadratic has solution if ray aims at it.
                        //    We can solve for t.
                        //    Let dx = x1 - x_start, dy = y1 - y_start.
                        //    (vx*t - dx)^2 + (vy*t - dy)^2 = 4r^2.
                        //    (vx^2+vy^2)t^2 - 2(vx*dx+vy*dy)t + (dx^2+dy^2-4r^2) = 0.
                        //    Since v is unit vector (or we can scale), let v be (cos, sin).
                        //    A = 1.
                        //    B = -2(dx cos + dy sin).
                        //    C = dx^2+dy^2-4r^2.
                        //    Discriminant D = B^2 - 4C >= 0.
                        //    We want smallest positive t.
                        //    t = (2(dx cos + dy sin) - sqrt(D)) / 2 = (dx cos + dy sin) - sqrt(D)/2.
                        //    Impact Point B_cue = (x_start + t*cos, y_start + t*sin).
                        //    Normal N = (x1 - B_cue.x, y1 - B_cue.y) / (2r). (Normalized).
                        //    
                        // 2. Ball 1 moves. 
                        //    V_b1 = N (speed arbitrary, assume 1).
                        //    We need V_b1 to hit Ball 3 at (x3, y3) to sink in (w, l).
                        //    Wait, hole is (w, l). 
                        //    Sink means pass through center (w, l).
                        //    Ray from (x1, y1) with direction N.
                        //    We need (x1 + lambda*Nx, y1 + lambda*Ny) = (w, l) for some lambda > 0.
                        //    This requires N to point exactly at the hole.
                        //    So N must be proportional to (w - x1, l - y1).
                        //    N is normalized. Target direction T3 = (w - x1, l - y1) / |T3|.
                        //    So N = T3.
                        //    This implies (x1 - B_cue.x, y1 - B_cue.y) is parallel to (w - x1, l - y1).

                        //    So we know the geometry of B_cue relative to B1.
                        //    Let r_B = r*2 (collision radius).

                        //    So the cue ball must arrive at (TargetX, TargetY) hitting Ball 1.
                        //    
                        // 3. Cue reflection to Ball 2.
                        //    We have V_cue_old (from start to impact).
                        //    We have N (impact normal).
                        //    V_cue_new = 2*(V_old.N)*N - V_old.

                        //    
                        //    So constraints:

                        //       Wait, B2 needs to go to (0, l). 
                        //       Impact point is B_cue.
                        //       V_new needs to point to (0, l).

                        //    This gives us an equation to solve for d.
                        //    Since we iterate d, we can check condition C.

                        //    BUT. We need to find d.
                        //    TargetX, TargetY depend on B1 and Hole 3. 
                        //    They are fixed points (given B1, r).
