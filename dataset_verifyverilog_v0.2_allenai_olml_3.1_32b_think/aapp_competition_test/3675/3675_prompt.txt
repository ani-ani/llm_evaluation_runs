module lcs_permutations(
    input clk,
    input rst_n,
    input start,
    input [2:0] n,           // number of strings (1-6)
    input [3:0] k,           // number of letters (1-8)
    input [31:0] strings_in, // serialized input: 8 chars packed into 32 bits (4 bits per char)
    input [2:0] str_idx,     // current string index for input
    output reg [3:0] result, // length of LCS
    output reg done,
    output reg ready_for_next // signal to send next string
);

    // Fixed-point format: Not needed for this integer-based problem
    // Characters are encoded as 4-bit values: A=1, B=2, ..., H=8
    
    // State definitions
    localparam IDLE = 3'd0;
    localparam INPUT_STRINGS = 3'd1;
    localparam BUILD_ORDERING = 3'd2;
    localparam COMPUTE_LCS = 3'd3;
    localparam OUTPUT_RESULT = 3'd4;
    
    reg [2:0] state;
    reg [2:0] input_counter;  // tracks which string we're inputting
    reg [2:0] current_str;    // current string being processed
    reg [3:0] i, j, idx;      // loop counters
    
    // Storage for strings: max 6 strings, 8 characters each
    // Each character is 4 bits, so 8 chars = 32 bits per string
    reg [31:0] strings [0:5];  // 6 strings of 8 chars each
    
    // Position arrays: pos[str_idx][char_value] = position (1-based)
    // We need to store positions of each character in each string
    reg [3:0] pos [0:5][0:8];  // [string_index][char_value] -> position
    
    // DAG adjacency matrix: can_precede[i][j] = 1 if char i appears before char j in ALL strings
    reg [0:8] can_precede [0:8]; // can_precede[i] is a bitmask of characters that can follow i
    
    // DP for longest path: dp[char] = longest chain ending at this char
    reg [3:0] dp [0:8];
    reg [3:0] max_lcs;
    
    // Temporary registers for comparison
    reg [3:0] pos_i_current, pos_j_current;
    reg can_precede_ij;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            result <= 0;
            ready_for_next <= 0;
            input_counter <= 0;
            current_str <= 0;
            i <= 0;
            j <= 0;
            idx <= 0;
            max_lcs <= 0;
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        state <= INPUT_STRINGS;
                        input_counter <= 0;
                        ready_for_next <= 1;
                    end
                end
                
                INPUT_STRINGS: begin
                    // When new string arrives
                    ready_for_next <= 0;
                    if (input_counter < n) begin
                        strings[input_counter] <= strings_in;
                        // Parse and store positions
                        // Extract each 4-bit character and record its position
                        for (idx = 0; idx < 8; idx = idx + 1) begin
                            if (idx < k) begin
                                pos[input_counter][strings_in[idx*4 +: 4]] <= idx + 1;
                            end
                        end
                        input_counter <= input_counter + 1;
                        ready_for_next <= 1;
                        if (input_counter + 1 >= n) begin
                            ready_for_next <= 0;
                            state <= BUILD_ORDERING;
                            current_str <= 0;
                            i <= 1;  // start from character 1
                            j <= 1;
                        end
                    end
                end
                
                BUILD_ORDERING: begin
                    // Build can_precede matrix
                    // For each pair (i,j), check if i appears before j in ALL strings
                    if (current_str < n) begin
                        if (i <= k && j <= k && i != j) begin
                            if (pos[current_str][i] < pos[current_str][j]) begin
                                // i appears before j in this string, check other strings
                                // For now, just mark - we'll validate after checking all strings
                                can_precede[i][j] <= can_precede[i][j] && 1'b1;
                            end else begin
                                can_precede[i][j] <= 0;
                            end
                        end
                        // Advance counters
                        if (j > k) begin
                            j <= 1;
                            if (i > k) begin
                                i <= 1;
                                current_str <= current_str + 1;
                            end else begin
                                i <= i + 1;
                            end
                        end else begin
                            j <= j + 1;
                        end
                    end else begin
                        // All strings processed, now compute LCS
                        state <= COMPUTE_LCS;
                        i <= 1;
                        j <= 1;
                        // Initialize dp
                        for (idx = 1; idx <= 8; idx = idx + 1) begin
                            dp[idx] <= 0;
                        end
                    end
                end
                
                COMPUTE_LCS: begin
                    // Find longest path in DAG using DP
                    // dp[i] = max(dp[j] + 1) for all j where can_precede[j][i] = 1
                    if (i <= k) begin
                        // Compute dp[i]
                        if (j <= k) begin
                            if (can_precede[j][i] && (dp[j] + 1 > dp[i])) begin
                                dp[i] <= dp[j] + 1;
                            end
                            j <= j + 1;
                        end else begin
                            // Done with this i
                            if (dp[i] == 0) dp[i] <= 1;  // single character
                            i <= i + 1;
                            j <= 1;
                        end
                    end else begin
                        // Find maximum
                        max_lcs <= 0;
                        for (i = 1; i <= k; i = i + 1) begin
                            if (dp[i] > max_lcs) max_lcs <= dp[i];
                        end
                        state <= OUTPUT_RESULT;
                    end
                end
                
                OUTPUT_RESULT: begin
                    result <= max_lcs;
                    done <= 1;
                    state <= IDLE;
                end
            endcase
        end
    end

endmodule

// Note: The above sequential implementation is complex.
// For actual Verilog, a simpler combinational approach:

module lcs_permutations_comb(
    input [31:0] str1,  // 8 chars, 4 bits each
    input [31:0] str2,
    input [31:0] str3,
    input [31:0] str4,
    input [31:0] str5,
    input [31:0] str6,
    input [2:0] n,      // number of strings (1-6)
    output [3:0] result // LCS length
);
    // This module computes LCS of up to 6 permutations of 8 letters
    // Character encoding: A=1, B=2, C=3, D=4, E=5, F=6, G=7, H=8
    
    // Step 1: Build position arrays for each string
    // pos[str][char] = position of char in string (1-8)
    reg [3:0] pos1 [1:8];
    reg [3:0] pos2 [1:8];
    reg [3:0] pos3 [1:8];
    reg [3:0] pos4 [1:8];
    reg [3:0] pos5 [1:8];
    reg [3:0] pos6 [1:8];
    
    integer c, p;
    always @(*) begin
        // Initialize positions
        for (c = 1; c <= 8; c = c + 1) begin
            pos1[c] = 0;
            pos2[c] = 0;
            pos3[c] = 0;
            pos4[c] = 0;
            pos5[c] = 0;
            pos6[c] = 0;
        end
        // Extract positions from bit strings
        // str1[31:28] = char at pos 1, str1[27:24] = char at pos 2, etc.
        for (p = 0; p < 8; p = p + 1) begin
            if (str1[4*(7-p)+:4] != 0) pos1[str1[4*(7-p)+:4]] = p + 1;
            if (str2[4*(7-p)+:4] != 0) pos2[str2[4*(7-p)+:4]] = p + 1;
            if (str3[4*(7-p)+:4] != 0) pos3[str3[4*(7-p)+:4]] = p + 1;
            if (str4[4*(7-p)+:4] != 0) pos4[str4[4*(7-p)+:4]] = p + 1;
            if (str5[4*(7-p)+:4] != 0) pos5[str5[4*(7-p)+:4]] = p + 1;
            if (str6[4*(7-p)+:4] != 0) pos6[str6[4*(7-p)+:4]] = p + 1;
        end
    end
    
    // Step 2: Compute can_precede matrix
    // can_precede[i][j] = 1 if char i appears before char j in ALL strings
    reg can_precede [1:8][1:8];
    
    always @(*) begin
        for (i = 1; i <= 8; i = i + 1) begin
            for (j = 1; j <= 8; j = j + 1) begin
                can_precede[i][j] = 1;
                if (i == j) can_precede[i][j] = 0;
                else begin
                    if (n >= 1 && pos1[i] >= pos1[j]) can_precede[i][j] = 0;
                    if (n >= 2 && pos2[i] >= pos2[j]) can_precede[i][j] = 0;
                    if (n >= 3 && pos3[i] >= pos3[j]) can_precede[i][j] = 0;
                    if (n >= 4 && pos4[i] >= pos4[j]) can_precede[i][j] = 0;
                    if (n >= 5 && pos5[i] >= pos5[j]) can_precede[i][j] = 0;
                    if (n >= 6 && pos6[i] >= pos6[j]) can_precede[i][j] = 0;
                end
            end
        end
    end
    
    // Step 3: Longest path in DAG (DP)
    // dp[i] = longest chain ending at character i
    reg [3:0] dp [1:8];
    reg [3:0] max_val;
    integer k;
    
    always @(*) begin
        // Initialize dp
        for (c = 1; c <= 8; c = c + 1) begin
            dp[c] = 1;  // Each character by itself has length 1
        end
        
        // Compute longest path - repeat to handle all dependencies
        // Since graph is small, we can iterate k times (k=8)
        for (k = 0; k < 8; k = k + 1) begin
            for (i = 1; i <= 8; i = i + 1) begin
                for (j = 1; j <= 8; j = j + 1) begin
                    if (can_precede[i][j]) begin
                        if (dp[i] + 1 > dp[j]) begin
                            dp[j] = dp[i] + 1;
                        end
                    end
                end
            end
        end
        
        // Find maximum
        max_val = dp[1];
        for (c = 2; c <= 8; c = c + 1) begin
            if (dp[c] > max_val) max_val = dp[c];
        end
    end
    
    assign result = max_val;

endmodule
