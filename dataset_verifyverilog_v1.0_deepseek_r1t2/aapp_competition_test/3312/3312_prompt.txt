module oil_cans (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] start_energy,
    input wire [1:0] start_x,
    input wire [1:0] start_y,
    input wire [1:0] can_x0, can_y0,
    input wire [4:0] can_t0,
    input wire can_valid0,
    input wire [1:0] can_x1, can_y1,
    input wire [4:0] can_t1,
    input wire can_valid1,
    input wire [1:0] can_x2, can_y2,
    input wire [4:0] can_t2,
    input wire can_valid2,
    input wire [1:0] can_x3, can_y3,
    input wire [4:0] can_t3,
    input wire can_valid3,
    input wire [1:0] can_x4, can_y4,
    input wire [4:0] can_t4,
    input wire can_valid4,
    input wire [1:0] can_x5, can_y5,
    input wire [4:0] can_t5,
    input wire can_valid5,
    input wire [1:0] can_x6, can_y6,
    input wire [4:0] can_t6,
    input wire can_valid6,
    input wire [1:0] can_x7, can_y7,
    input wire [4:0] can_t7,
    input wire can_valid7,
    output reg [3:0] max_points,
    output reg done
);

// Fixed parameters
localparam GRID_SIZE = 4;
localparam MAX_ENERGY = 10;
localparam MAX_TIME = 20;
localparam NUM_STATES = 176; // (MAX_ENERGY+1) * GRID_SIZE * GRID_SIZE

// State definitions
localparam IDLE = 0;
localparam LOAD = 1;
localparam COMPUTE_INIT = 2;
localparam PRECOMPUTE = 3;
localparam UPDATE_STATE = 4;
localparam MAX_STATE = 5;

// Registers
reg [2:0] state, next_state;
reg [3:0] can_count [0:20][0:3][0:3]; // [time][x][y]
reg [3:0] dp_current [0:10][0:3][0:3]; // [energy][x][y]
reg [3:0] dp_next [0:10][0:3][0:3];
reg [3:0] points_gain_table [0:3][0:3];
reg [3:0] energy_gain_table [0:3][0:3];
reg [4:0] time_step;
reg [7:0] counter; // Multi-purpose counter
reg [3:0] max_value;
reg [2:0] load_counter;

// Helper signals
wire [3:0] new_energy_stay;
wire [3:0] new_energy_move;
wire [3:0] new_points_stay;
wire [3:0] new_points_move;
wire [1:0] x_index, y_index;
wire [3:0] energy_index;

// Combinational logic for state transitions and computations
always @(*) begin
    next_state = state;
    case (state)
        IDLE: if (start) next_state = LOAD;
        LOAD: if (load_counter == 3'd7) next_state = COMPUTE_INIT;
        COMPUTE_INIT: next_state = PRECOMPUTE;
        PRECOMPUTE: if (counter >= 8'd191) next_state = UPDATE_STATE; // 176 + 16 cycles
        UPDATE_STATE: if (counter >= 8'd175) next_state = PRECOMPUTE;
        MAX_STATE: if (counter >= 8'd175) next_state = IDLE;
        default: next_state = IDLE;
    endcase
end

// Sequential state update
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        max_points <= 0;
        // Reset arrays
        for (integer t = 0; t <= 20; t = t + 1) begin
            for (integer i = 0; i < 4; i = i + 1) begin
                for (integer j = 0; j < 4; j = j + 1) begin
                    can_count[t][i][j] <= 0;
                end
            end
        end
        for (integer e = 0; e <= 10; e = e + 1) begin
            for (integer i = 0; i < 4; i = i + 1) begin
                for (integer j = 0; j < 4; j = j + 1) begin
                    dp_current[e][i][j] <= 4'b1111; // 15 = unreachable
                    dp_next[e][i][j] <= 4'b1111;
                end
            end
        end
        counter <= 0;
        time_step <= 0;
        load_counter <= 0;
        max_value <= 0;
    end else begin
        state <= next_state;
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    // Reset can_count for loading
                    for (integer t = 0; t <= 20; t = t + 1) begin
                        for (integer i = 0; i < 4; i = i + 1) begin
                            for (integer j = 0; j < 4; j = j + 1) begin
                                can_count[t][i][j] <= 0;
                            end
                        end
                    end
                    load_counter <= 0;
                    counter <= 0;
                    time_step <= 0;
                    max_value <= 0;
                end
            end
            
            LOAD: begin
                // Load each can event
                case (load_counter)
                    0: if (can_valid0 && can_t0 <= 20) can_count[can_t0][can_x0][can_y0] <= can_count[can_t0][can_x0][can_y0] + 1;
                    1: if (can_valid1 && can_t1 <= 20) can_count[can_t1][can_x1][can_y1] <= can_count[can_t1][can_x1][can_y1] + 1;
                    2: if (can_valid2 && can_t2 <= 20) can_count[can_t2][can_x2][can_y2] <= can_count[can_t2][can_x2][can_y2] + 1;
                    3: if (can_valid3 && can_t3 <= 20) can_count[can_t3][can_x3][can_y3] <= can_count[can_t3][can_x3][can_y3] + 1;
                    4: if (can_valid4 && can_t4 <= 20) can_count[can_t4][can_x4][can_y4] <= can_count[can_t4][can_x4][can_y4] + 1;
                    5: if (can_valid5 && can_t5 <= 20) can_count[can_t5][can_x5][can_y5] <= can_count[can_t5][can_x5][can_y5] + 1;
                    6: if (can_valid6 && can_t6 <= 20) can_count[can_t6][can_x6][can_y6] <= can_count[can_t6][can_x6][can_y6] + 1;
                    7: if (can_valid7 && can_t7 <= 20) can_count[can_t7][can_x7][can_y7] <= can_count[can_t7][can_x7][can_y7] + 1;
                endcase
                load_counter <= load_counter + 1;
            end
            
            COMPUTE_INIT: begin
                // Initialize DP table
                for (integer e = 0; e <= 10; e = e + 1) begin
                    for (integer i = 0; i < 4; i = i + 1) begin
                        for (integer j = 0; j < 4; j = j + 1) begin
                            dp_current[e][i][j] <= 4'b1111;
                            dp_next[e][i][j] <= 4'b1111;
                        end
                    end
                end
                // Set initial state
                if (start_energy <= MAX_ENERGY && start_x < GRID_SIZE && start_y < GRID_SIZE) begin
                    dp_current[start_energy][start_x][start_y] <= 0;
                end
                time_step <= 0;
                counter <= 0;
            end
            
            PRECOMPUTE: begin
                counter <= counter + 1;
                if (counter < 8'd176) begin
                    // Reset dp_next (first 176 cycles)
                    for (integer e = 0; e <= 10; e = e + 1) begin
                        for (integer i = 0; i < 4; i = i + 1) begin
                            for (integer j = 0; j < 4; j = j + 1) begin
                                dp_next[e][i][j] <= 4'b1111;
                            end
                        end
                    end
                end else if (counter < 8'd192) begin
                    // Compute gain tables for time_step+1 (cycles 176-191)
                    integer idx = counter - 8'd176;
                    integer x = idx / 4;
                    integer y = idx % 4;
                    integer t = time_step + 1;
                    
                    if (t <= MAX_TIME) begin
                        // Points gain: cans at current cell
                        points_gain_table[x][y] <= (t <= 20) ? can_count[t][x][y] : 0;
                        
                        // Energy gain: cans in adjacent cells
                        energy_gain_table[x][y] <= 0;
                        if (x+1 < GRID_SIZE) energy_gain_table[x][y] <= energy_gain_table[x][y] + ((t <= 20) ? can_count[t][x+1][y] : 0);
                        if (x-1 >= 0) energy_gain_table[x][y] <= energy_gain_table[x][y] + ((t <= 20) ? can_count[t][x-1][y] : 0);
                        if (y+1 < GRID_SIZE) energy_gain_table[x][y] <= energy_gain_table[x][y] + ((t <= 20) ? can_count[t][x][y+1] : 0);
                        if (y-1 >= 0) energy_gain_table[x][y] <= energy_gain_table[x][y] + ((t <= 20) ? can_count[t][x][y-1] : 0);
                    end else begin
                        points_gain_table[x][y] <= 0;
                        energy_gain_table[x][y] <= 0;
                    end
                end
            end
            
            UPDATE_STATE: begin
                counter <= counter + 1;
                if (counter < 8'd176) begin
                    // Decode state index
                    integer state_idx = counter;
                    integer e = state_idx / 16; // (GRID_SIZE * GRID_SIZE = 16)
                    integer rem = state_idx % 16;
                    integer x = rem / 4;
                    integer y = rem % 4;
                    
                    if (dp_current[e][x][y] != 4'b1111 && time_step < MAX_TIME) begin
                        // Stay option
                        reg [3:0] new_e_stay = (e + energy_gain_table[x][y] > MAX_ENERGY) ? MAX_ENERGY : e + energy_gain_table[x][y];
                        reg [3:0] new_p_stay = dp_current[e][x][y] + points_gain_table[x][y];
                        
                        if (dp_next[new_e_stay][x][y] == 4'b1111 || dp_next[new_e_stay][x][y] < new_p_stay)
                            dp_next[new_e_stay][x][y] <= new_p_stay;
                        
                        // Move options (if energy >= 1)
                        if (e >= 1) begin
                            // Right
                            if (x+1 < GRID_SIZE) begin
                                reg [3:0] new_e_right = (e-1 + energy_gain_table[x+1][y] > MAX_ENERGY) ? MAX_ENERGY : e-1 + energy_gain_table[x+1][y];
                                reg [3:0] new_p_right = dp_current[e][x][y] + points_gain_table[x+1][y];
                                if (dp_next[new_e_right][x+1][y] == 4'b1111 || dp_next[new_e_right][x+1][y] < new_p_right)
                                    dp_next[new_e_right][x+1][y] <= new_p_right;
                            end
                            // Left
                            if (x-1 >= 0) begin
                                reg [3:0] new_e_left = (e-1 + energy_gain_table[x-1][y] > MAX_ENERGY) ? MAX_ENERGY : e-1 + energy_gain_table[x-1][y];
                                reg [3:0] new_p_left = dp_current[e][x][y] + points_gain_table[x-1][y];
                                if (dp_next[new_e_left][x-1][y] == 4'b1111 || dp_next[new_e_left][x-1][y] < new_p_left)
                                    dp_next[new_e_left][x-1][y] <= new_p_left;
                            end
                            // Down
                            if (y+1 < GRID_SIZE) begin
                                reg [3:0] new_e_down = (e-1 + energy_gain_table[x][y+1] > MAX_ENERGY) ? MAX_ENERGY : e-1 + energy_gain_table[x][y+1];
                                reg [3:0] new_p_down = dp_current[e][x][y] + points_gain_table[x][y+1];
                                if (dp_next[new_e_down][x][y+1] == 4'b1111 || dp_next[new_e_down][x][y+1] < new_p_down)
                                    dp_next[new_e_down][x][y+1] <= new_p_down;
                            end
                            // Up
                            if (y-1 >= 0) begin
                                reg [3:0] new_e_up = (e-1 + energy_gain_table[x][y-1] > MAX_ENERGY) ? MAX_ENERGY : e-1 + energy_gain_table[x][y-1];
                                reg [3:0] new_p_up = dp_current[e][x][y] + points_gain_table[x][y-1];
                                if (dp_next[new_e_up][x][y-1] == 4'b1111 || dp_next[new_e_up][x][y-1] < new_p_up)
                                    dp_next[new_e_up][x][y-1] <= new_p_up;
                            end
                        end
                    end
                end else if (counter == 8'd175) begin
                    // End of UPDATE_STATE: swap dp_current and dp_next, increment time
                    for (integer e = 0; e <= 10; e = e + 1) begin
                        for (integer i = 0; i < 4; i = i + 1) begin
                            for (integer j = 0; j < 4; j = j + 1) begin
                                dp_current[e][i][j] <= dp_next[e][i][j];
                                dp_next[e][i][j] <= 4'b1111;
                            end
                        end
                    end
                    time_step <= time_step + 1;
                    counter <= 0;
                end
            end
            
            MAX_STATE: begin
                counter <= counter + 1;
                if (counter < 8'd176) begin
                    integer state_idx = counter;
                    integer e = state_idx / 16;
                    integer rem = state_idx % 16;
                    integer x = rem / 4;
                    integer y = rem % 4;
                    
                    if (dp_current[e][x][y] != 4'b1111) begin
                        if (dp_current[e][x][y] > max_value) begin
                            max_value <= dp_current[e][x][y];
                        end
                    end
                end else begin
                    max_points <= max_value;
                    done <= 1;
                end
            end
        endcase
    end
end

endmodule
