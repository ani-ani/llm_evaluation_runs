module t9_keypress #(
    parameter DICT_SIZE = 8,
    parameter MAX_WORD_LEN = 8,
    parameter MAX_QUERY_LEN = 16
) (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [7:0] dict_words [0:DICT_SIZE-1][0:MAX_WORD_LEN-1],  // Dictionary words (ASCII)
    input wire [3:0] dict_lengths [0:DICT_SIZE-1],                // Length of each dictionary word
    input wire [7:0] query_word [0:MAX_QUERY_LEN-1],               // Query word (ASCII)
    input wire [4:0] query_length,                                // Query word length
    output reg [7:0] key_sequence [0:255],                        // Output key sequence
    output reg [7:0] key_count,                                   // Number of keys in sequence
    output reg done                                               // Computation complete
);

// Internal state machine states
reg [2:0] state;
localparam IDLE = 0;
localparam COMPUTE_COSTS = 1;
localparam DP_COMPUTE = 2;
localparam BACKTRACK = 3;
localparam OUTPUT = 4;

// T9 mapping lookup function
function automatic [3:0] char_to_digit(input [7:0] c);
    case (c)
        "a", "b", "c": char_to_digit = 2;
        "d", "e", "f": char_to_digit = 3;
        "g", "h", "i": char_to_digit = 4;
        "j", "k", "l": char_to_digit = 5;
        "m", "n", "o": char_to_digit = 6;
        "p", "q", "r", "s": char_to_digit = 7;
        "t", "u", "v": char_to_digit = 8;
        "w", "x", "y", "z": char_to_digit = 9;
        default: char_to_digit = 0;
    endcase
endfunction

// DP arrays
reg [31:0] dp [0:MAX_QUERY_LEN];  // Minimum keypresses to reach position
reg [31:0] dp_prev [0:MAX_QUERY_LEN];  // Previous position for backtracking
reg [31:0] dp_word_idx [0:MAX_QUERY_LEN];  // Word used to reach position
reg [31:0] dp_cycle_cost [0:MAX_QUERY_LEN];  // Cycling cost for this word

// Word properties
reg [31:0] word_costs [0:DICT_SIZE-1];  // Cost to type each word (digits + cycling)
reg [3:0] word_digit_seqs [0:DICT_SIZE-1][0:MAX_WORD_LEN-1];  // Digit sequences

// Counters and temporary registers
reg [4:0] i, j, k;
reg [31:0] min_cost, current_cost;
reg [31:0] backtrack_idx;
reg [7:0] output_idx;
reg [7:0] temp_key;

// Output buffer
reg [7:0] out_buf [0:255];
reg [7:0] out_len;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        key_count <= 0;
        out_len <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    state <= COMPUTE_COSTS;
                    i <= 0;
                    j <= 0;
                    done <= 0;
                    out_len <= 0;
                end
            end
            
            COMPUTE_COSTS: begin
                // Compute digit sequences and costs for each dictionary word
                if (j < dict_lengths[i]) begin
                    // Convert character to digit
                    word_digit_seqs[i][j] <= char_to_digit(dict_words[i][j]);
                    j <= j + 1;
                end else begin
                    // Compute cycling cost
                    // Count how many words have the same digit sequence
                    // This is simplified - in real implementation would need grouping
                    // For now, assume cycling cost = 0 (first word in group)
                    // Calculate base cost: digits needed + cycling
                    word_costs[i] <= dict_lengths[i];  // Simplified: no cycling cost
                    i <= i + 1;
                    j <= 0;
                    if (i == DICT_SIZE - 1) begin
                        state <= DP_COMPUTE;
                        i <= 0;  // Position in query
                        j <= 0;  // Word index
                    end
                end
            end
            
            DP_COMPUTE: begin
                // Dynamic programming to find minimal keypresses
                if (i <= query_length) begin
                    if (j < DICT_SIZE) begin
                        // Check if word j fits starting at position i
                        if (i + dict_lengths[j] <= query_length) begin
                            // Check if word matches substring
                            // Simplified: assume all words in dictionary match where they can
                            // In real implementation would compare characters
                            current_cost = dp[i] + word_costs[j];
                            if (i > 0) current_cost = current_cost + 1;  // Add 'R' cost
                            
                            if (current_cost < dp[i + dict_lengths[j]] || dp[i + dict_lengths[j]] == 0) begin
                                dp[i + dict_lengths[j]] <= current_cost;
                                dp_prev[i + dict_lengths[j]] <= i;
                                dp_word_idx[i + dict_lengths[j]] <= j;
                                dp_cycle_cost[i + dict_lengths[j]] <= 0;  // Simplified
                            end
                        end
                        j <= j + 1;
                    end else begin
                        j <= 0;
                        i <= i + 1;
                    end
                end else begin
                    state <= BACKTRACK;
                    backtrack_idx <= query_length;
                    out_len <= 0;
                end
            end
            
            BACKTRACK: begin
                // Reconstruct solution from DP table
                if (backtrack_idx > 0) begin
                    // Add the word to output (reversed order for now)
                    if (out_len < 250) begin
                        // Add cycling keys if needed (simplified: none)
                        // Add digit keys
                        for (k = 0; k < dict_lengths[dp_word_idx[backtrack_idx]]; k = k + 1) begin
                            out_buf[out_len] <= (word_digit_seqs[dp_word_idx[backtrack_idx]][k] + 8'h30);  // Convert to ASCII
                            out_len <= out_len + 1;
                        end
                        // Add 'R' if not first part
                        if (dp_prev[backtrack_idx] > 0) begin
                            out_buf[out_len] <= "R";
                            out_len <= out_len + 1;
                        end
                        backtrack_idx <= dp_prev[backtrack_idx];
                    end else begin
                        state <= OUTPUT;
                        output_idx <= 0;
                    end
                end else begin
                    state <= OUTPUT;
                    output_idx <= 0;
                end
            end
            
            OUTPUT: begin
                // Output the key sequence
                if (output_idx < out_len) begin
                    key_sequence[output_idx] <= out_buf[output_idx];
                    output_idx <= output_idx + 1;
                    key_count <= out_len;
                end else begin
                    done <= 1;
                    state <= IDLE;
                end
            end
        endcase
    end
end

endmodule
