module sequence_in_gcd_table (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [63:0] n,         // 64-bit for n up to 10^12
    input wire [63:0] m,         // 64-bit for m up to 10^12  
    input wire [3:0] k,          // max 8 elements
    input wire [63:0] a [0:7],   // input sequence (max 8 elements)
    output reg result,            // 1 if sequence exists
    output reg done              // computation complete
);

    // State definitions
    localparam IDLE = 4'b0000;
    localparam COMPUTE_LCM = 4'b0001;
    localparam SOLVE_CRT = 4'b0010;
    localparam CHECK_GCD = 4'b0011;
    localparam VERIFY_RESULT = 4'b0100;
    localparam FINISH = 4'b0101;

    reg [3:0] state;
    reg [63:0] lcm_val;
    reg [63:0] crt_R, crt_M;
    reg [3:0] iter_idx;
    reg [63:0] gcd_a, gcd_b;
    reg gcd_start;
    wire [63:0] gcd_result;
    wire gcd_done;
    reg [63:0] j0;
    reg [63:0] temp_result;
    
    // GCD computation unit
    gcd_unit gcd_inst (
        .clk(clk),
        .rst_n(rst_n),
        .start(gcd_start),
        .a(gcd_a),
        .b(gcd_b),
        .result(gcd_result),
        .done(gcd_done)
    );

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            result <= 0;
            done <= 0;
            lcm_val <= 64'd1;
            crt_R <= 64'd0;
            crt_M <= 64'd1;
            iter_idx <= 4'd0;
            j0 <= 64'd0;
            gcd_start <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        state <= COMPUTE_LCM;
                        iter_idx <= 4'd0;
                        lcm_val <= 64'd1;
                    end
                end

                COMPUTE_LCM: begin
                    if (iter_idx < k) begin
                        // Compute GCD(lcm, a[i])
                        if (!gcd_start) begin
                            gcd_a <= lcm_val;
                            gcd_b <= a[iter_idx];
                            gcd_start <= 1;
                        end else if (gcd_done) begin
                            gcd_start <= 0;
                            if (gcd_result != 0) begin
                                temp_result <= (lcm_val * a[iter_idx]) / gcd_result;
                                // Check overflow in next cycle
                                state <= COMPUTE_LCM;
                            end else begin
                                state <= FINISH;
                                result <= 0;
                            end
                            iter_idx <= iter_idx + 1;
                        end
                    end else begin
                        // Check LCM overflow
                        if (lcm_val > n) begin
                            result <= 0;
                            state <= FINISH;
                        end else begin
                            state <= SOLVE_CRT;
                            iter_idx <= 4'd0;
                            crt_R <= 64'd0;
                            crt_M <= 64'd1;
                        end
                    end
                end

                SOLVE_CRT: begin
                    if (iter_idx < k) begin
                        // Simplified CRT: just set j0 to 0 and check conditions
                        // In full implementation, would use extended Euclidean algorithm
                        state <= SOLVE_CRT;
                        iter_idx <= iter_idx + 1;
                    end else begin
                        j0 <= crt_R;
                        state <= CHECK_GCD;
                        iter_idx <= 4'd0;
                    end
                end

                CHECK_GCD: begin
                    if (iter_idx < k) begin
                        // Check GCD(lcm_val, j0 + iter_idx) == a[iter_idx]
                        if (!gcd_start) begin
                            gcd_a <= lcm_val;
                            gcd_b <= j0 + iter_idx;
                            gcd_start <= 1;
                        end else if (gcd_done) begin
                            gcd_start <= 0;
                            if (gcd_result != a[iter_idx]) begin
                                result <= 0;
                                state <= FINISH;
                            end else if (iter_idx == k - 1) begin
                                state <= VERIFY_RESULT;
                            end
                            iter_idx <= iter_idx + 1;
                        end
                    end
                end

                VERIFY_RESULT: begin
                    // Check if j0 + k - 1 <= m
                    if (j0 + k - 1 <= m && j0 >= 1) begin
                        result <= 1;
                    end else begin
                        result <= 0;
                    end
                    state <= FINISH;
                end

                FINISH: begin
                    done <= 1;
                    if (!start) begin
                        state <= IDLE;
                        done <= 0;
                    end
                end
            endcase
        end
    end

endmodule

// GCD computation unit with iterative Euclidean algorithm
module gcd_unit (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [63:0] a,
    input wire [63:0] b,
    output reg [63:0] result,
    output reg done
);

    localparam IDLE = 2'b00;
    localparam COMPUTE = 2'b01;
    localparam FINISH = 2'b10;

    reg [1:0] state;
    reg [63:0] x, y;
    reg [63:0] a_reg, b_reg;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            result <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        a_reg <= a;
                        b_reg <= b;
                        state <= COMPUTE;
                        done <= 0;
                    end
                end

                COMPUTE: begin
                    if (b_reg == 0) begin
                        result <= a_reg;
                        state <= FINISH;
                    end else begin
                        x <= a_reg;
                        y <= b_reg;
                        a_reg <= y;
                        b_reg <= x % y;
                    end
                end

                FINISH: begin
                    done <= 1;
                    state <= IDLE;
                end
            endcase
        end
    end

endmodule
