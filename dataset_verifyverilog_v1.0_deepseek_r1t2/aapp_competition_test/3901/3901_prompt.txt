module gcd #(
    parameter WIDTH = 16
)(
    input clk,
    input rst_n,
    input start,
    input [WIDTH-1:0] a,
    input [WIDTH-1:0] b,
    output reg [WIDTH-1:0] gcd,
    output reg done
);
    reg busy;
    reg [WIDTH-1:0] x, y;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            busy <= 0;
            done <= 0;
            gcd <= 0;
        end else begin
            if (start && !busy) begin
                x <= a;
                y <= b;
                busy <= 1;
                done <= 0;
            end else if (busy) begin
                if (y != 0) begin
                    x <= y;
                    y <= x % y;
                end else begin
                    gcd <= x;
                    done <= 1;
                    busy <= 0;
                end
            end else begin
                done <= 0;
            end
        end
    end
endmodule
