// Module for Intergalactic Surgery Puzzle Solver

// E (empty) is represented as 0

parameter GRID_WIDTH = 31; // 2*15+1 = 31

// State machine states
typedef enum logic [3:0] {
    IDLE,
    RESET,
    FIND_BLANK,
    MOVE_TO_CORNER,
    SORT_ORGANS,
    CHECK_RESULT,
    OUTPUT_SEQUENCE,
    DONE_STATE
} state_t;

state_t current_state, next_state;

// Internal registers

reg [7:0] iteration_counter; // For bounded loops

// Helper: Check if value is defined
function logic is_valid(input [7:0] val);
    begin
        is_valid = (val !== 8'hxx && val !== 8'hzz);
    end
endfunction

// Helper: Find blank position in grid
function void find_blank();
    begin
        for (int i = 0; i < 2; i++) begin
            for (int j = 0; j < GRID_WIDTH; j++) begin
                if (grid[i][j] == 8'h00) begin
                    blank_row = i;
                    blank_col = j;
                    return;
                end
            end
        end
    end
endfunction

// Helper: Generate move sequence for moving blank to corner
function void generate_corner_moves();
    logic [7:0] moves [0:31];
    logic [7:0] move_idx = 0;
    
    // Move right to column 2k
    for (int i = 0; i < GRID_WIDTH - blank_col - 1; i++) begin
        moves[move_idx] = "r";
        move_idx++;
    end
    
    // Move down to row 1
    if (blank_row == 0) begin
        moves[move_idx] = "d";
        move_idx++;
    end
    
    // Copy to output
    for (int i = 0; i < move_idx; i++) begin
        if (move_count < SEQUENCE_LENGTH) begin
            move_sequence[move_count] = moves[i];
            move_count++;
        end
    end
endfunction

                // Hold done signal

                    current_state <= IDLE;
                    done <= 0;
                    valid <= 0;
                end
            end
            
            default: current_state <= IDLE;
        endcase
    end
