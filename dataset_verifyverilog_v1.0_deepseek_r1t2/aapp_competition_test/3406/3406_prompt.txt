module castle_danger (
    input wire clk,
    input wire rst_n,
    input wire start,
    
    // Four Nazi troop positions (8-bit coordinates for scaling)
    input wire [7:0] p0_x, p0_y,
    input wire [7:0] p1_x, p1_y,
    input wire [7:0] p2_x, p2_y,
    input wire [7:0] p3_x, p3_y,
    
    // Castle position
    input wire [7:0] castle_x, castle_y,
    
    output reg danger,
    output reg done
);

// State machine states
localparam IDLE = 2'd0;
localparam CHECK = 2'd1;
localparam DONE = 2'd2;

// Internal registers
reg [1:0] state;
reg [1:0] edge_idx;      // 0-3 for four edges
reg [15:0] cross;        // Cross product result
reg is_inside;           // Inside flag
reg first_sign;          // Sign of first edge's cross product
reg degenerate;          // Degeneracy flag

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        danger <= 1'b0;
        done <= 1'b0;
        is_inside <= 1'b0;
        degenerate <= 1'b0;
        edge_idx <= 2'd0;
    end else begin
        case (state)
            IDLE: begin
                done <= 1'b0;
                danger <= 1'b0;
                if (start) begin
                    state <= CHECK;
                    edge_idx <= 2'd0;
                    is_inside <= 1'b1;
                    degenerate <= 1'b0;
                end
            end
            
            CHECK: begin
                // Compute cross product for current edge and castle
                case (edge_idx)
                    2'd0: cross <= ( (p1_x - p0_x) * (castle_y - p0_y) ) - ( (p1_y - p0_y) * (castle_x - p0_x) );
                    2'd1: cross <= ( (p2_x - p1_x) * (castle_y - p1_y) ) - ( (p2_y - p1_y) * (castle_x - p1_x) );
                    2'd2: cross <= ( (p3_x - p2_x) * (castle_y - p2_y) ) - ( (p3_y - p2_y) * (castle_x - p2_x) );
                    2'd3: cross <= ( (p0_x - p3_x) * (castle_y - p3_y) ) - ( (p0_y - p3_y) * (castle_x - p3_x) );
                endcase
                state <= CHECK + 1;
            end
            
            CHECK + 1: begin
                // Check degeneracy on first iteration
                if (edge_idx == 2'd0) begin
                    if (((p1_x - p0_x) * (p2_y - p0_y) - (p1_y - p0_y) * (p2_x - p0_x) == 16'd0) ||
                        ((p2_x - p1_x) * (p3_y - p1_y) - (p2_y - p1_y) * (p3_x - p1_x) == 16'd0) ||
                        ((p3_x - p2_x) * (p0_y - p2_y) - (p3_y - p2_y) * (p0_x - p2_x) == 16'd0) ||
                        ((p0_x - p3_x) * (p1_y - p3_y) - (p0_y - p3_y) * (p1_x - p3_x) == 16'd0)) begin
                        degenerate <= 1'b1;
                    end
                    if (cross != 16'd0) first_sign <= cross[15];
                end else begin
                    // Subsequent edges: sign must match first (or be zero)
                    if (cross != 16'd0 && cross[15] != first_sign) begin
                        is_inside <= 1'b0;
                    end
                end
                
                // Check if castle is exactly on edge
                if (cross == 16'd0) begin
                    // Still counts as inside
                end
                
                if (edge_idx < 2'd3) begin
                    edge_idx <= edge_idx + 1;
                    state <= CHECK;
                end else begin
                    // Final result
                    if (degenerate) begin
                        danger <= 1'b0;
                    end else begin
                        danger <= is_inside;
                    end
                    state <= DONE;
                end
            end
            
            DONE: begin
                done <= 1'b1;
                if (!start) state <= IDLE;
            end
            
            default: state <= IDLE;
        endcase
    end
end

endmodule
