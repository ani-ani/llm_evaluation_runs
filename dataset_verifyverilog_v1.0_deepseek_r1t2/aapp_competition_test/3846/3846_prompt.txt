module leader_determination #(
    parameter n = 8,
    parameter m = 16
) (
    input clk,
    input rst_n,
    input [m-1:0] event_type,  // 0 for '+', 1 for '-'
    input [4*m-1:0] event_id,  // Packed event IDs, each 4 bits
    output reg [n-1:0] possible_leaders,
    output reg done
);

    // State definitions
    localparam S_INIT = 0;
    localparam S_CHECK_S0 = 1;
    localparam S_SIM_START = 2;
    localparam S_CHECK_INITIAL = 3;
    localparam S_SIM_EVENT = 4;
    localparam S_CHECK_EVENT = 5;
    localparam S_CHECK_FINAL = 6;
    localparam S_UPDATE_CANDIDATE = 7;
    localparam S_DONE = 8;

    // Internal registers
    reg [3:0] state;
    reg [3:0] current_candidate;
    reg [7:0] current_S0;  // 2^n possibilities, 8 bits for n=8
    reg [7:0] current_state;
    reg [3:0] current_event;
    reg found_valid_S0;

    // Event storage
    reg event_type_reg [0:m-1];
    reg [3:0] event_id_reg [0:m-1];

    integer i;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= S_INIT;
            possible_leaders <= 0;
            done <= 0;
        end else begin
            case (state)
                S_INIT: begin
                    // Initialize and unpack events
                    current_candidate <= 1;
                    current_S0 <= 0;
                    found_valid_S0 <= 0;
                    for (i = 0; i < m; i = i + 1) begin
                        event_type_reg[i] <= event_type[i];
                        event_id_reg[i] <= event_id[4*i+:4];
                    end
                    state <= S_CHECK_S0;
                end

                S_CHECK_S0: begin
                    // Check if current_S0 satisfies first event constraint
                    if (current_S0 >= (1 << n)) begin
                        // All S0 checked for current candidate
                        if (!found_valid_S0) begin
                            possible_leaders[current_candidate-1] <= 0;
                        end
                        state <= S_UPDATE_CANDIDATE;
                    end else begin
                        // Check first event constraint
                        if (event_type_reg[0] == 0) begin  // '+' login
                            if (current_S0[event_id_reg[0]-1] == 0) begin
                                state <= S_SIM_START;
                            end else begin
                                current_S0 <= current_S0 + 1;
                            end
                        end else begin  // '-' logout
                            if (current_S0[event_id_reg[0]-1] == 1) begin
                                state <= S_SIM_START;
                            end else begin
                                current_S0 <= current_S0 + 1;
                            end
                        end
                    end
                end

                S_SIM_START: begin
                    // Start simulation for this S0
                    current_state <= current_S0;
                    current_event <= 0;
                    state <= S_CHECK_INITIAL;
                end

                S_CHECK_INITIAL: begin
                    // Check initial state condition
                    if (current_state != 0 && (current_state & (1 << (current_candidate-1)) == 0) begin
                        // Fail for this S0
                        current_S0 <= current_S0 + 1;
                        state <= S_CHECK_S0;
                    end else begin
                        state <= S_SIM_EVENT;
                    end
                end

                S_SIM_EVENT: begin
                    if (current_event >= m) begin
                        state <= S_CHECK_FINAL;
                    end else begin
                        // Apply event
                        if (event_type_reg[current_event] == 0) begin
                            // Login
                            current_state <= current_state | (1 << (event_id_reg[current_event]-1));
                        end else begin
                            // Logout
                            current_state <= current_state & ~(1 << (event_id_reg[current_event]-1));
                        end
                        state <= S_CHECK_EVENT;
                    end
                end

                S_CHECK_EVENT: begin
                    // Check state after event
                    if (current_state != 0 && (current_state & (1 << (current_candidate-1)) == 0) begin
                        // Fail for this S0
                        current_S0 <= current_S0 + 1;
                        state <= S_CHECK_S0;
                    end else begin
                        current_event <= current_event + 1;
                        state <= S_SIM_EVENT;
                    end
                end

                S_CHECK_FINAL: begin
                    // Check final state after all events
                    if (current_state != 0 && (current_state & (1 << (current_candidate-1)) == 0) begin
                        // Fail for this S0
                        current_S0 <= current_S0 + 1;
                        state <= S_CHECK_S0;
                    end else begin
                        // Found valid S0 for this candidate
                        found_valid_S0 <= 1;
                        possible_leaders[current_candidate-1] <= 1;
                        // We can stop after finding one valid S0 for this candidate
                        current_S0 <= (1 << n);  // Force to max to skip remaining S0
                        state <= S_CHECK_S0;
                    end
                end

                S_UPDATE_CANDIDATE: begin
                    if (current_candidate == n) begin
                        state <= S_DONE;
                    end else begin
                        current_candidate <= current_candidate + 1;
                        current_S0 <= 0;
                        found_valid_S0 <= 0;
                        state <= S_CHECK_S0;
                    end
                end

                S_DONE: begin
                    done <= 1;
                    state <= S_DONE;
                end

                default: state <= S_INIT;
            endcase
        end
    end
endmodule
