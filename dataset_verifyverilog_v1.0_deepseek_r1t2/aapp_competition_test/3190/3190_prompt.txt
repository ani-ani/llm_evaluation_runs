module voodoo_counter (
    input wire clk,              // Clock, rising edge
    input wire rst_n,            // Active-low synchronous reset
    input wire start,            // Start pulse, assert for 1 cycle
    input wire [2:0] N,          // Number of valid array elements (1..8)
    input wire [7:0] P,          // Threshold price
    input wire [7:0] arr_0, arr_1, arr_2, arr_3,
                     arr_4, arr_5, arr_6, arr_7, // Array values (N valid)
    output reg [5:0] result,     // Count of subarrays with average >= P
    output reg done              // Asserted high for 1 cycle when result is valid
);

// Behavioral description:
// The module computes the number of consecutive subsequences (subarrays)
// whose average is >= P, for a small array of up to 8 elements.
//
// Algorithm:
// 1. Compute b_i = arr_i - P (signed 9-bit).
// 2. Compute prefix sums: prefix[0] = 0; for i=0..N-1: prefix[i+1] = prefix[i] + b_i.
// 3. Count pairs (i,j) with 0 <= i < j <= N and prefix[i] <= prefix[j].
// 4. Result is the count.
//
// Implementation uses a state machine with the following states:
//   IDLE: wait for start
//   INIT: set prefix[0]=0, i=0
//   PREFIX: compute prefix[i+1] for i=0..N-1
//   COUNT_INIT: set i=0, j=1, result=0
//   COUNT_LOOP: compare prefix[i] and prefix[j]; increment result if condition true; update j and i
//   DONE: assert done for one cycle
//
// Internal registers:
//   prefix[0:8] : 12-bit signed, store prefix sums
//   i_cnt, j_cnt: 4-bit counters
//   state: 3-bit state register
//   temp_result: 6-bit accumulator
//
// Timing: For N=8, computation takes about 44 clock cycles.
//
// Reset: On rst_n=0, all state is cleared, result=0, done=0.
//
endmodule
