module triple_correlation #(
    parameter P_MAX = 20,
    parameter N_MAX = 3,
    parameter M_MAX = 3
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [7:0] p,
    input wire [3:0] seq [0:P_MAX-1],
    output reg found,
    output reg [3:0] a_out,
    output reg [3:0] b_out,
    output reg [3:0] c_out,
    output reg [7:0] n_out,
    output reg [7:0] m_out,
    output reg done
);

    // Internal registers
    reg [3:0] seq_reg [0:P_MAX-1];
    reg [7:0] p_reg;
    reg [7:0] T;
    reg [7:0] n_reg, m_reg;
    reg [3:0] a_reg, b_reg, c_reg;
    reg [7:0] i;
    reg [7:0] triple_count;
    reg [7:0] first_i;
    reg candidate_valid;
    reg [3:0] best_a, best_b, best_c;
    reg [7:0] best_n, best_m, best_first_i;

    // State definition
    localparam IDLE = 0;
    localparam LOAD = 1;
    localparam SEARCH = 2;
    localparam SCANNING = 3;
    localparam EVALUATE = 4;
    localparam NEXT_C = 5;
    localparam NEXT_B = 6;
    localparam NEXT_A = 7;
    localparam NEXT_M = 8;
    localparam NEXT_N = 9;
    localparam DONE = 10;
    reg [3:0] state;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            found <= 0;
            // Other registers reset if needed
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        state <= LOAD;
                    end
                end
                LOAD: begin
                    // Load sequence from inputs
                    for (integer idx = 0; idx < P_MAX; idx = idx + 1) begin
                        seq_reg[idx] <= seq[idx];
                    end
                    p_reg <= p;
                    T <= (p + 39) / 40 + 1;  // ceil(p/40)+1
                    best_first_i <= P_MAX + 1;
                    found <= 0;
                    state <= SEARCH;
                end
                SEARCH: begin
                    n_reg <= 1;
                    m_reg <= 1;
                    a_reg <= 0;
                    b_reg <= 0;
                    c_reg <= 0;
                    state <= SCANNING;
                    i <= 0;
                    triple_count <= 0;
                    first_i <= P_MAX + 1;
                    candidate_valid <= 1;
                end
                SCANNING: begin
                    if (i < p_reg) begin
                        // Check correlation conditions and triple occurrence
                        if ( (
                             (seq_reg[i] == a_reg && i + n_reg < p_reg && seq_reg[i + n_reg] == b_reg && i + n_reg + m_reg < p_reg && seq_reg[i + n_reg + m_reg] != c_reg) ||
                             (seq_reg[i] == b_reg && i + m_reg < p_reg && seq_reg[i + m_reg] == c_reg && i >= n_reg && seq_reg[i - n_reg] != a_reg) ||
                             (seq_reg[i] == a_reg && i + n_reg + m_reg < p_reg && seq_reg[i + n_reg + m_reg] == c_reg && i + n_reg < p_reg && seq_reg[i + n_reg] != b_reg)
                           ) ) begin
                            candidate_valid <= 0;
                            state <= NEXT_C;
                        end else begin
                            // Count triple occurrences
                            if (i + n_reg < p_reg && i + n_reg + m_reg < p_reg &&
                                seq_reg[i] == a_reg && seq_reg[i + n_reg] == b_reg && seq_reg[i + n_reg + m_reg] == c_reg) begin
                                triple_count <= triple_count + 1;
                                if (first_i > P_MAX) first_i <= i;
                            end
                            i <= i + 1;
                            if (i + 1 < p_reg) begin
                                state <= SCANNING;
                            end else begin
                                state <= EVALUATE;
                            end
                        end
                    end else begin
                        state <= EVALUATE;
                    end
                end
                EVALUATE: begin
                    if (candidate_valid && triple_count >= T) begin
                        if (first_i < best_first_i ||
                            (first_i == best_first_i && n_reg < best_n) ||
                            (first_i == best_first_i && n_reg == best_n && m_reg < best_m)) begin
                            best_a <= a_reg;
                            best_b <= b_reg;
                            best_c <= c_reg;
                            best_n <= n_reg;
                            best_m <= m_reg;
                            best_first_i <= first_i;
                        end
                    end
                    state <= NEXT_C;
                end
                NEXT_C: begin
                    if (c_reg == 9) begin
                        state <= NEXT_B;
                    end else begin
                        c_reg <= c_reg + 1;
                        state <= SCANNING;
                        i <= 0;
                        triple_count <= 0;
                        first_i <= P_MAX + 1;
                        candidate_valid <= 1;
                    end
                end
                NEXT_B: begin
                    if (b_reg == 9) begin
                        state <= NEXT_A;
                    end else begin
                        b_reg <= b_reg + 1;
                        c_reg <= 0;
                        state <= SCANNING;
                        i <= 0;
                        triple_count <= 0;
                        first_i <= P_MAX + 1;
                        candidate_valid <= 1;
                    end
                end
                NEXT_A: begin
                    if (a_reg == 9) begin
                        state <= NEXT_M;
                    end else begin
                        a_reg <= a_reg + 1;
                        b_reg <= 0;
                        c_reg <= 0;
                        state <= SCANNING;
                        i <= 0;
                        triple_count <= 0;
                        first_i <= P_MAX + 1;
                        candidate_valid <= 1;
                    end
                end
                NEXT_M: begin
                    if (m_reg == M_MAX) begin
                        state <= NEXT_N;
                    end else begin
                        m_reg <= m_reg + 1;
                        a_reg <= 0;
                        b_reg <= 0;
                        c_reg <= 0;
                        state <= SCANNING;
                        i <= 0;
                        triple_count <= 0;
                        first_i <= P_MAX + 1;
                        candidate_valid <= 1;
                    end
                end
                NEXT_N: begin
                    if (n_reg == N_MAX) begin
                        state <= DONE;
                    end else begin
                        n_reg <= n_reg + 1;
                        m_reg <= 1;
                        a_reg <= 0;
                        b_reg <= 0;
                        c_reg <= 0;
                        state <= SCANNING;
                        i <= 0;
                        triple_count <= 0;
                        first_i <= P_MAX + 1;
                        candidate_valid <= 1;
                    end
                end
                DONE: begin
                    done <= 1;
                    if (best_first_i <= P_MAX) begin
                        found <= 1;
                        a_out <= best_a;
                        b_out <= best_b;
                        c_out <= best_c;
                        n_out <= best_n;
                        m_out <= best_m;
                    end else begin
                        found <= 0;
                    end
                    state <= IDLE;
                end
                default: state <= IDLE;
            endcase
        end
    end
endmodule
