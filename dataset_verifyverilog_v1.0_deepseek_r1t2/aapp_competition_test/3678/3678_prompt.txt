module loop_checker (
    input [7:0] x0, y0, x1, y1, x2, y2, x3, y3,
    input [7:0] x4, y4, x5, y5, x6, y6, x7, y7,
    input [3:0] n,
    output result
);

// Check 1: n must be even and >= 4
wire n_valid = (n % 2 == 0) && (n >= 4);

// Check 2: All x-coordinates appear in even counts
wire x_parity_ok;

// Check 3: All y-coordinates appear in even counts  
wire y_parity_ok;

// Check 4: Points form connected graph via horizontal/vertical edges
wire graph_connected;

// Final result
assign result = n_valid & x_parity_ok & y_parity_ok & graph_connected;

// - Verify all counts are even numbers
// - Build adjacency graph from shared x/y coordinates
// - Check graph connectivity

endmodule
