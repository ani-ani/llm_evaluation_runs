module battery_allocator (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] n,
    input wire [3:0] k,
    input wire [7:0] batteries [0:15],
    output reg [7:0] result,
    output reg done
);
    parameter MAX_BATTERIES = 16;
    localparam IDLE = 3'b000, COPY = 3'b001, SORT = 3'b010, COMPUTE = 3'b011, RESULT = 3'b100, DONE = 3'b101;

    reg [2:0] state;
    reg [7:0] mem [0:15];
    reg [3:0] copy_cnt, outer_cnt, inner_cnt, compute_cnt;
    reg [7:0] max_diff;
    wire [7:0] diff;

    assign diff = mem[2*compute_cnt + 1] - mem[2*compute_cnt];

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            result <= 0;
            copy_cnt <= 0;
            outer_cnt <= 0;
            inner_cnt <= 0;
            compute_cnt <= 0;
            max_diff <= 0;
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        copy_cnt <= 0;
                        state <= COPY;
                    end
                end

                COPY: begin
                    mem[copy_cnt] <= batteries[copy_cnt];
                    if (copy_cnt == MAX_BATTERIES-1) begin
                        outer_cnt <= 0;
                        inner_cnt <= 0;
                        state <= SORT;
                    end else begin
                        copy_cnt <= copy_cnt + 1;
                    end
                end

                SORT: begin
                    if (inner_cnt < MAX_BATTERIES - 1 - outer_cnt) begin
                        if (mem[inner_cnt] > mem[inner_cnt+1]) begin
                            mem[inner_cnt] <= mem[inner_cnt+1];
                            mem[inner_cnt+1] <= mem[inner_cnt];
                        end
                        inner_cnt <= inner_cnt + 1;
                    end else begin
                        outer_cnt <= outer_cnt + 1;
                        inner_cnt <= 0;
                        if (outer_cnt == MAX_BATTERIES - 2) begin
                            compute_cnt <= 0;
                            max_diff <= 0;
                            state <= COMPUTE;
                        end
                    end
                end

                COMPUTE: begin
                    if (diff > max_diff) max_diff <= diff;
                    if (compute_cnt < n - 1) begin
                        compute_cnt <= compute_cnt + 1;
                    end else begin
                        state <= RESULT;
                    end
                end

                RESULT: begin
                    result <= max_diff;
                    done <= 1;
                    state <= DONE;
                end

                DONE: begin
                    if (start) begin
                        done <= 0;
                        copy_cnt <= 0;
                        state <= COPY;
                    end
                end

                default: state <= IDLE;
            endcase
        end
    end
endmodule
