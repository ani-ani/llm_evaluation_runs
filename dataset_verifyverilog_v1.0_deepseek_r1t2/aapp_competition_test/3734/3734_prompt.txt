module day_checker (
    input clk,
    input rst_n,
    input start,
    input [2:0] day1,  // 0=Monday, 1=Tuesday, ..., 6=Sunday
    input [2:0] day2,
    output reg result,
    output reg done
);

// States
reg [1:0] state;

// Internal registers
reg [2:0] diff;

// State definitions
localparam IDLE = 2'b00;
localparam COMPUTE = 2'b01;
localparam COMPLETE = 2'b10;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        result <= 0;
        done <= 0;
        diff <= 0;
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    state <= COMPUTE;
                    // Calculate difference modulo 7
                    diff <= (day2 - day1 + 7) % 7;
                end
            end
            COMPUTE: begin
                // Check if difference is 0, 2, or 3
                if (diff == 3'd0 || diff == 3'd2 || diff == 3'd3) begin
                    result <= 1;
                end else begin
                    result <= 0;
                end
                state <= COMPLETE;
            end
            COMPLETE: begin
                done <= 1;
                state <= IDLE;
            end
            default: state <= IDLE;
        endcase
    end
end

endmodule
