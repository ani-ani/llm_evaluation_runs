module SubsequenceHash (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] N,        // Number of valid elements (1-8)
    input wire [7:0] K,        // Number of hashes to output (1-255)
    input wire [15:0] B,       // Base for hash
    input wire [15:0] M,       // Modulus for hash
    input wire [7:0] a0, a1, a2, a3, a4, a5, a6, a7, // Array elements
    output reg [15:0] hash,    // Output hash value
    output reg valid,          // Valid output (high for 1 cycle per hash)
    output reg done            // Done signal (high for 1 cycle after all outputs)
);

// Behavior:
// - On rising edge of clk, if rst_n is low, reset internal state (hash=0, valid=0, done=0)
// - On rising edge of clk, if start is high and the module is idle, start processing.
// - The module computes all non-empty subsequences of a[0:N-1].
// - For each subsequence, compute hash as: h = (v1 * B^{p-1} + ... + vp) mod M.
//   Implemented as: h = 0; for each element v in subsequence: h = (h * B + v) % M.
// - Store all subsequences and hashes in internal memory (up to 255 entries).
// - Sort the subsequences in lexicographic order using bubble sort.
// - After sorting, output the first K hashes sequentially, one per cycle, with valid high.
// - When all K hashes are output, assert done high for one cycle.
// - The module remains idle until next start.

endmodule
