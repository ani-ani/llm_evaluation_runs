module envelope_optimizer #(
    parameter N = 8,  // Max number of card types
    parameter K = 4,  // Max envelope types
    parameter DATA_W = 16,  // Data width for w,h,q
    parameter RESULT_W = 32  // Result width
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [DATA_W-1:0] card_width [N-1:0],
    input wire [DATA_W-1:0] card_height [N-1:0],
    input wire [DATA_W-1:0] card_qty [N-1:0],
    input wire [3:0] num_env_types,  // k (1-4)
    output reg [RESULT_W-1:0] result,
    output reg done
);

// States
localparam IDLE = 3'd0;
localparam COMPUTE_SUBSET = 3'd1;
localparam DP_INIT = 3'd2;
localparam DP_COMPUTE = 3'd3;
localparam FINISH = 3'd4;

reg [2:0] state;
reg [2:0] next_state;

// Memory for subset waste calculation (2^N = 256 entries)
reg [RESULT_W-1:0] subset_waste [0:255];
reg [7:0] subset_idx;  // Current subset being computed

// DP arrays: dp[mask][t]
reg [RESULT_W-1:0] dp [0:255];  // Current dp layer (t-1)
reg [RESULT_W-1:0] next_dp [0:255];  // Next dp layer (t)

// Helper variables
reg [7:0] mask;
reg [7:0] submask;
reg [DATA_W-1:0] max_w, max_h;
reg [DATA_W*2-1:0] env_area;
reg [RESULT_W-1:0] sum_qty;
reg [RESULT_W-1:0] sum_area;
reg [3:0] t;  // Current envelope type count
reg [7:0] i, j;
reg [RESULT_W-1:0] candidate;

// Combinational logic for subset computation
always @(*) begin
    // Calculate max width and height for current subset
    max_w = 0;
    max_h = 0;
    sum_qty = 0;
    sum_area = 0;
    
    for (i = 0; i < N; i = i + 1) begin
        if (subset_idx[i]) begin
            if (card_width[i] > max_w) max_w = card_width[i];
            if (card_height[i] > max_h) max_h = card_height[i];
            sum_qty = sum_qty + card_qty[i];
            sum_area = sum_area + (card_qty[i] * (card_width[i] * card_height[i]));
        end
    end
    
    env_area = max_w * max_h;
end

// Sequential state transitions
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        subset_idx <= 0;
        t <= 0;
        done <= 0;
        result <= 0;
    end else begin
        state <= next_state;
        
        case (state)
            IDLE: begin
                if (start) begin
                    subset_idx <= 1;  // Start from subset {0} (but we'll process from 0)
                    t <= 1;
                    done <= 0;
                end
            end
            
            COMPUTE_SUBSET: begin
                // Calculate waste for current subset
                if (sum_qty > 0) begin
                    subset_waste[subset_idx] <= (env_area * sum_qty) - sum_area;
                end else begin
                    subset_waste[subset_idx] <= 0;
                end
                
                if (subset_idx == 255) begin
                    subset_idx <= 0;
                end else begin
                    subset_idx <= subset_idx + 1;
                end
            end
            
            DP_INIT: begin
                // Initialize dp[0][*] = 0, dp[mask][0] = INF for mask != 0
                if (subset_idx == 0) begin
                    dp[0] <= 0;
                    subset_idx <= 1;
                end else if (subset_idx < 255) begin
                    dp[subset_idx] <= 32'h7FFFFFFF;  // INF
                    subset_idx <= subset_idx + 1;
                end else begin
                    dp[255] <= 32'h7FFFFFFF;
                    subset_idx <= 0;
                    t <= 1;
                end
            end
            
            DP_COMPUTE: begin
                if (t <= num_env_types) begin
                    // For each mask, compute min over submask
                    if (mask == 0) begin
                        mask <= 1;
                        next_dp[0] <= dp[0];  // dp[0][t] = dp[0][t-1] = 0
                    end else if (mask < 255) begin
                        // Compute for current mask
                        if (submask == 0) begin
                            // Initialize with large value
                            next_dp[mask] <= 32'h7FFFFFFF;
                            submask <= mask;
                        end else if (submask > 0) begin
                            // submask is subset of mask
                            if (subset_waste[submask] < 32'h7FFFFFFF && dp[mask ^ submask] < 32'h7FFFFFFF) begin
                                candidate <= subset_waste[submask] + dp[mask ^ submask];
                            end else begin
                                candidate <= 32'h7FFFFFFF;
                            end
                            
                            if (submask == mask) begin
                                submask <= 0;
                                mask <= mask + 1;
                            end else begin
                                // Find next submask using trick: (submask - 1) & mask
                                submask <= (submask - 1) & mask;
                            end
                        end else begin
                            // Done with this mask
                            if (next_dp[mask] > candidate) begin
                                next_dp[mask] <= candidate;
                            end
                        end
                    end else begin
                        // All masks computed for current t
                        // Copy next_dp to dp
                        if (subset_idx < 255) begin
                            dp[subset_idx] <= next_dp[subset_idx];
                            subset_idx <= subset_idx + 1;
                        end else begin
                            dp[255] <= next_dp[255];
                            subset_idx <= 0;
                            mask <= 0;
                            submask <= 0;
                            t <= t + 1;
                        end
                    end
                end else begin
                    // Finished all DP iterations
                    result <= dp[255];
                    done <= 1;
                end
            end
            
            FINISH: begin
                done <= 0;
            end
        endcase
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: begin
            if (start) next_state = COMPUTE_SUBSET;
        end
        
        COMPUTE_SUBSET: begin
            if (subset_idx == 255) next_state = DP_INIT;
        end
        
        DP_INIT: begin
            if (subset_idx == 255 && t == 1) next_state = DP_COMPUTE;
        end
        
        DP_COMPUTE: begin
            if (t > num_env_types) next_state = FINISH;
        end
        
        FINISH: begin
            if (!done) next_state = IDLE;
        end
        
        default: next_state = IDLE;
    endcase
end

endmodule
