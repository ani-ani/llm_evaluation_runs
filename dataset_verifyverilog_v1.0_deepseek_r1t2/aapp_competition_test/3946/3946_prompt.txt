module nauuo_visited (
  input clk,
  input rst_n,
  input start,
  input [2:0] m,          // Number of visits (max 8)
  input [2:0] n,          // Number of pictures (max 8)
  input [7:0] a [0:7],    // Liked/disliked flags (0/1)
  input [15:0] w [0:7],   // Initial weights (Q8.8 fixed-point)
  output reg [15:0] result [0:7], // Expected weights (Q8.8)
  output reg done         // Computation complete
);

  // Parameters
  parameter MOD = 998244353;
  parameter FRAC_BITS = 8;
  
  // State machine states
  typedef enum logic [2:0] {
    IDLE,
    ACCUMULATE,
    DP_INIT,
    DP_COMPUTE,
    FINALIZE,
    OUTPUT
  } state_t;
  
  state_t state;
  
  // Internal registers
  reg [15:0] li, di;      // Sum of liked/disliked weights
  reg [15:0] S;           // Total weight
  reg [3:0] t;            // DP step counter
  reg [3:0] i;            // DP state index
  reg [15:0] F [0:8];     // DP array (max m+1 states)
  reg [15:0] F_next [0:8];
  reg [15:0] po, ne;      // Final accumulators
  reg [2:0] out_idx;      // Output index
  
  // Helper function: modular inverse
  function [15:0] mod_inv(input [15:0] x);
    // Simplified: assumes x is small and invertible
    mod_inv = 1; // Placeholder - real implementation needs exponentiation
  endfunction
  
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state <= IDLE;
      done <= 0;
      // Reset outputs
      for (int k=0; k<8; k++) result[k] <= 0;
    end else begin
      case (state)
        IDLE: begin
          if (start) begin
            li <= 0;
            di <= 0;
            t <= 0;
            i <= 0;
            out_idx <= 0;
            state <= ACCUMULATE;
          end
        end
        
        ACCUMULATE: begin
          // Accumulate li/di in fixed-point
          for (int k=0; k<8; k++) begin
            if (k < n) begin
              if (a[k]) li <= li + w[k];
              else di <= di + w[k];
            end
          end
          S <= li + di;
          state <= DP_INIT;
        end
        
        DP_INIT: begin
          // Initialize DP array
          F[0] <= 16'h0100; // 1.0 in Q8.8
          for (int k=1; k<9; k++) F[k] <= 0;
          t <= 0;
          state <= (t < m) ? DP_COMPUTE : FINALIZE;
        end
        
        DP_COMPUTE: begin
          if (i <= t) begin
            // Calculate denominator (fixed-point)
            reg [15:0] den = S + (i << 1) - t;
            reg [15:0] inv_den = mod_inv(den);
            
            // Liked transition
            reg [15:0] prob_like = ((li + i) * inv_den) >> 8;
            F_next[i+1] <= F[i] * prob_like;
            
            // Disliked transition
            if (di > t - i) begin
              reg [15:0] prob_dislike = ((di - t + i) * inv_den) >> 8;
              F_next[i] <= F[i] * prob_dislike;
            end
            
            i <= i + 1;
          end else begin
            // Update F for next step
            for (int k=0; k<9; k++) F[k] <= F_next[k];
            t <= t + 1;
            i <= 0;
            state <= (t < m) ? DP_COMPUTE : FINALIZE;
          end
        end
        
        FINALIZE: begin
          // Compute po and ne
          po <= 0;
          ne <= 0;
          for (int k=0; k<9; k++) begin
            if (k <= m) begin
              po <= po + F[k] * (li + k);
              ne <= ne + F[k] * (di - m + k);
            end
          end
          state <= OUTPUT;
        end
        
        OUTPUT: begin
          if (out_idx < n) begin
            if (a[out_idx]) begin
              result[out_idx] <= (w[out_idx] * po) / li;
            end else begin
              result[out_idx] <= (w[out_idx] * ne) / di;
            end
            out_idx <= out_idx + 1;
          end else begin
            done <= 1;
            state <= IDLE;
          end
        end
        
        default: state <= IDLE;
      endcase
    end
  end
endmodule
