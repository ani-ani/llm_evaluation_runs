module tape_art_solver #(
    parameter MAX_N = 16,
    parameter MAX_COLOR = 16,
    parameter DATA_WIDTH = 4,
    parameter ADDR_WIDTH = 4
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [ADDR_WIDTH-1:0] n,
    input wire [DATA_WIDTH-1:0] colors [0:MAX_N-1],
    
    output reg done,
    output reg impossible,
    output reg [ADDR_WIDTH-1:0] num_instructions,
    output reg [ADDR_WIDTH-1:0] inst_l [0:MAX_COLOR-1],
    output reg [ADDR_WIDTH-1:0] inst_r [0:MAX_COLOR-1],
    output reg [DATA_WIDTH-1:0] inst_c [0:MAX_COLOR-1],
    output reg inst_valid [0:MAX_COLOR-1]
);
    localparam S_IDLE = 3'b000, S_SCAN = 3'b001, S_CHECK = 3'b010,
               S_SORT = 3'b011, S_OUTPUT = 3'b100, S_DONE = 3'b101,
               S_IMPOSSIBLE = 3'b110;
    
    reg [2:0] state, next_state;
    reg [ADDR_WIDTH-1:0] L [0:MAX_COLOR-1];
    reg [ADDR_WIDTH-1:0] R [0:MAX_COLOR-1];
    reg present [0:MAX_COLOR-1];
    reg [ADDR_WIDTH-1:0] idx, i, j, sort_i, sort_j, min_idx;
    reg [ADDR_WIDTH-1:0] temp_L, temp_R;
    reg [DATA_WIDTH-1:0] temp_C;
    reg temp_valid;
    
    always @(posedge clk or negedge rst_n)
        if (!rst_n) state <= S_IDLE; else state <= next_state;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            done <= 0; impossible <= 0; num_instructions <= 0;
            idx <= 0; i <= 0; j <= 0; sort_i <= 0; sort_j <= 0;
            for (integer k = 0; k < MAX_COLOR; k = k + 1) begin
                L[k] <= 0; R[k] <= 0; present[k] <= 0;
                inst_l[k] <= 0; inst_r[k] <= 0; inst_c[k] <= 0; inst_valid[k] <= 0;
            end
            next_state <= S_IDLE;
        end else begin
            case (state)
                S_IDLE: if (start) begin
                    idx <= 0;
                    for (integer k = 0; k < MAX_COLOR; k = k + 1) begin
                        L[k] <= 0; R[k] <= 0; present[k] <= 0;
                    end
                    next_state <= S_SCAN;
                end else next_state <= S_IDLE;
                
                S_SCAN: if (idx < n) begin
                    if (!present[colors[idx]]) begin
                        present[colors[idx]] <= 1;
                        L[colors[idx]] <= idx;
                        R[colors[idx]] <= idx;
                    end else R[colors[idx]] <= idx;
                    idx <= idx + 1;
                    next_state <= S_SCAN;
                end else begin
                    i <= 0; j <= 1;
                    next_state <= S_CHECK;
                end
                
                S_CHECK: if (i >= MAX_COLOR) begin
                    sort_i <= 0; sort_j <= 1; min_idx <= 0;
                    next_state <= S_SORT;
                end else if (j >= MAX_COLOR) begin
                    i <= i + 1; j <= i + 2;
                end else if (!present[i] || !present[j]) begin
                    j <= j + 1;
                end else if (R[i] >= L[j] && L[i] <= R[j] &&
                           !(L[i] <= L[j] && R[j] <= R[i]) &&
                           !(L[j] <= L[i] && R[i] <= R[j])) begin
                    impossible <= 1; next_state <= S_IMPOSSIBLE;
                end else j <= j + 1;
                
                S_SORT: if (sort_i >= MAX_COLOR - 1) begin
                    i <= 0; next_state <= S_OUTPUT;
                end else if (sort_j >= MAX_COLOR) begin
                    if (min_idx != sort_i) begin
                        temp_L <= L[sort_i]; L[sort_i] <= L[min_idx]; L[min_idx] <= temp_L;
                        temp_R <= R[sort_i]; R[sort_i] <= R[min_idx]; R[min_idx] <= temp_R;
                        temp_valid <= present[sort_i]; present[sort_i] <= present[min_idx]; present[min_idx] <= temp_valid;
                    end
                    sort_i <= sort_i + 1; sort_j <= sort_i + 1; min_idx <= sort_i + 1;
                end else if (!present[sort_j]) sort_j <= sort_j + 1;
                else if (sort_j == sort_i) begin
                    min_idx <= sort_j; sort_j <= sort_j + 1;
                end else begin
                    if ((R[sort_j] - L[sort_j]) > (R[min_idx] - L[min_idx])) min_idx <= sort_j;
                    sort_j <= sort_j + 1;
                end
                
                S_OUTPUT: if (i < MAX_COLOR) begin
                    if (present[i]) begin
                        inst_l[i] <= L[i]; inst_r[i] <= R[i]; inst_c[i] <= i;
                        inst_valid[i] <= 1; num_instructions <= num_instructions + 1;
                    end else inst_valid[i] <= 0;
                    i <= i + 1;
                end else begin done <= 1; next_state <= S_DONE; end
                
                S_DONE: next_state <= S_DONE;
                S_IMPOSSIBLE: begin done <= 1; next_state <= S_IMPOSSIBLE; end
                default: next_state <= S_IDLE;
            endcase
        end
    end
endmodule
