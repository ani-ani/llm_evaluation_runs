module max_queens_4x4 (
    input clk,
    input rst_n,
    input start,
    input [15:0] broken,  // 1=broken cell, 0=usable
    output reg [3:0] max_queens,
    output reg [15:0] num_ways,
    output reg done
);
    // Parameters
    parameter BOARD_SIZE = 16;
    parameter MAX_TRIANGLES = 560; // Number of triangles in 4x4 board

    // State definitions
    localparam IDLE = 3'b000;
    localparam INIT = 3'b001;
    localparam CHECK_BROKEN = 3'b010;
    localparam POPCOUNT = 3'b011;
    localparam CHECK_TRIANGLE = 3'b100;
    localparam UPDATE_MAX = 3'b101;
    localparam NEXT_SUBSET = 3'b110;
    localparam DONE = 3'b111;

    // Registers
    reg [2:0] state;
    reg [15:0] subset;
    reg [3:0] max_q;
    reg [15:0] ways;
    reg [3:0] popcnt;
    reg [9:0] tri_idx; // 0 to 559
    reg triangle_found;
    reg [15:0] current_mask;
    reg [3:0] bit_idx; // for popcount loop
    reg [15:0] temp_mask; // for triangle check

    // Triangle table: precomputed for 4x4 board
    // Each entry is a 16-bit mask of three cells that form a triangle
    reg [15:0] triangle_mask [0:MAX_TRIANGLES-1];

    // Initialize triangle table (generated by Python script)
    initial begin
        // Example entries (replace with actual generated values)
        triangle_mask[0] = 16'h0000; // placeholder
        triangle_mask[1] = 16'h0000;
        // ... up to triangle_mask[559]
    end

    // State machine
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            subset <= 16'b0;
            max_q <= 4'b0;
            ways <= 16'b0;
            popcnt <= 4'b0;
            tri_idx <= 10'b0;
            triangle_found <= 1'b0;
            done <= 1'b0;
            bit_idx <= 4'b0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        state <= INIT;
                        done <= 1'b0;
                    end
                end

                INIT: begin
                    subset <= 16'b0;
                    max_q <= 4'b0;
                    ways <= 16'b0;
                    state <= CHECK_BROKEN;
                end

                CHECK_BROKEN: begin
                    // Check if current subset intersects broken cells
                    if (|(subset & broken)) begin
                        // Skip this subset
                        state <= NEXT_SUBSET;
                    end else begin
                        // Initialize popcount variables
                        popcnt <= 4'b0;
                        bit_idx <= 4'b0;
                        state <= POPCOUNT;
                    end
                end

                POPCOUNT: begin
                    if (bit_idx < 4'd16) begin
                        if (subset[bit_idx]) begin
                            popcnt <= popcnt + 1'b1;
                        end
                        bit_idx <= bit_idx + 1'b1;
                    end else begin
                        // Popcount done, initialize triangle check
                        tri_idx <= 10'b0;
                        triangle_found <= 1'b0;
                        state <= CHECK_TRIANGLE;
                    end
                end

                CHECK_TRIANGLE: begin
                    if (tri_idx < MAX_TRIANGLES) begin
                        current_mask <= triangle_mask[tri_idx];
                        // Check if triangle_mask[tri_idx] is subset of subset
                        if ((subset & current_mask) == current_mask) begin
                            triangle_found <= 1'b1;
                            // No need to check further
                            state <= NEXT_SUBSET;
                        end else begin
                            tri_idx <= tri_idx + 1'b1;
                        end
                    end else begin
                        // No triangle found, update max and ways
                        if (triangle_found == 1'b0) begin
                            state <= UPDATE_MAX;
                        end else begin
                            state <= NEXT_SUBSET;
                        end
                    end
                end

                UPDATE_MAX: begin
                    if (popcnt > max_q) begin
                        max_q <= popcnt;
                        ways <= 16'd1;
                    end else if (popcnt == max_q) begin
                        ways <= ways + 1'b1;
                    end
                    state <= NEXT_SUBSET;
                end

                NEXT_SUBSET: begin
                    if (subset < 16'hFFFF) begin
                        subset <= subset + 1'b1;
                        state <= CHECK_BROKEN;
                    end else begin
                        state <= DONE;
                    end
                end

                DONE: begin
                    max_queens <= max_q;
                    num_ways <= ways;
                    done <= 1'b1;
                    state <= IDLE;
                end

                default: state <= IDLE;
            endcase
        end
    end

endmodule
