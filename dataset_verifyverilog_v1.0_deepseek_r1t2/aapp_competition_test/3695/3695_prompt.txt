module dog_show (
    input clk,
    input rst_n,
    input start,
    input [2:0] n,        // number of bowls (1-8)
    input [7:0] T,        // time limit
    input [7:0] t_i [0:7], // array of t_i values
    output reg [3:0] result,
    output reg done
);

// State definitions
localparam IDLE = 4'd0;
localparam PREPROCESS = 4'd1;
localparam SORT_EVENTS = 4'd2;
localparam SWEEP = 4'd3;
localparam DONE = 4'd4;

reg [3:0] state;
reg [2:0] i;          // bowl index
reg [4:0] event_count; // number of events (0-16)
reg [7:0] events_time [0:15];
reg events_is_end [0:15]; // 0=start, 1=end

reg [7:0] counter, max_count;
reg [3:0] j, k;       // sorting indices

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
        event_count <= 0;
        counter <= 0;
        max_count <= 0;
        i <= 0;
        j <= 0;
        k <= 0;
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    state <= PREPROCESS;
                    i <= 0;
                    event_count <= 0;
                end
            end

            PREPROCESS: begin
                if (i < n) begin
                    if (T >= (i + 2)) begin
                        // Compute m = max(0, t_i[i] - i - 1)
                        reg [7:0] m_val;
                        if (t_i[i] > (i + 1)) 
                            m_val = t_i[i] - i - 1;
                        else 
                            m_val = 0;
                        // Compute M = T - i - 2
                        reg [7:0] M_val = T - i - 2;
                        
                        if (m_val <= M_val) begin
                            events_time[event_count] <= m_val;
                            events_is_end[event_count] <= 0;
                            events_time[event_count+1] <= M_val;
                            events_is_end[event_count+1] <= 1;
                            event_count <= event_count + 2;
                        end
                    end
                    i <= i + 1;
                end else begin
                    if (event_count > 0) begin
                        state <= SORT_EVENTS;
                        j <= 0;
                        k <= 0;
                    end else begin
                        state <= DONE;
                        result <= 0;
                    end
                end
            end

            SORT_EVENTS: begin
                if (j < event_count - 1) begin
                    if (k < event_count - j - 1) begin
                        // Compare events[k] and events[k+1]
                        if (events_time[k] > events_time[k+1]) begin
                            // Swap events
                            events_time[k] <= events_time[k+1];
                            events_time[k+1] <= events_time[k];
                            events_is_end[k] <= events_is_end[k+1];
                            events_is_end[k+1] <= events_is_end[k];
                        end else if (events_time[k] == events_time[k+1]) begin
                            if (events_is_end[k] > events_is_end[k+1]) begin
                                events_time[k] <= events_time[k+1];
                                events_time[k+1] <= events_time[k];
                                events_is_end[k] <= events_is_end[k+1];
                                events_is_end[k+1] <= events_is_end[k];
                            end
                        end
                        k <= k + 1;
                    end else begin
                        k <= 0;
                        j <= j + 1;
                    end
                end else begin
                    state <= SWEEP;
                    i <= 0;
                    counter <= 0;
                    max_count <= 0;
                end
            end

            SWEEP: begin
                if (i < event_count) begin
                    if (events_is_end[i] == 0) begin
                        counter <= counter + 1;
                        if (counter + 1 > max_count)
                            max_count <= counter + 1;
                    end else begin
                        counter <= counter - 1;
                    end
                    i <= i + 1;
                end else begin
                    state <= DONE;
                    result <= max_count;
                end
            end

            DONE: begin
                done <= 1;
                state <= IDLE;
            end

            default: state <= IDLE;
        endcase
    end
end

endmodule
