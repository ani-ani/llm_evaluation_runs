module ranger_strength (
  input clk,
  input rst_n,
  input start,
  input [3:0] n,          // number of rangers (1 to 8)
  input [3:0] k,          // k up to 8
  input [9:0] x,          // x up to 1000
  input [9:0] arr_0, arr_1, arr_2, arr_3, arr_4, arr_5, arr_6, arr_7,
  output reg [9:0] max_out,
  output reg [9:0] min_out,
  output reg done
);

// Parameters
parameter N = 8;
parameter DATA_WIDTH = 10;
parameter K_WIDTH = 4;

// State definitions
localparam IDLE = 3'b000;
localparam SORT = 3'b001;
localparam XOR = 3'b010;
localparam CHECK_K = 3'b011;
localparam FINISH = 3'b100;

// Registers
reg [2:0] state, next_state;
reg [K_WIDTH-1:0] k_count;
reg [2:0] sort_step; // 0 to 7
reg [DATA_WIDTH-1:0] arr [0:N-1]; // current array
reg [DATA_WIDTH-1:0] next_arr [0:N-1]; // next array

// Combinational logic for next array (sorting and XOR)
integer i;
always @(*) begin
  // Default: next_arr = arr
  for (i=0; i<N; i=i+1) next_arr[i] = arr[i];

  if (state == SORT) begin
    // Odd-even transposition sort step
    if (sort_step[0] == 0) begin
      // Even step: compare (0,1), (2,3), (4,5), (6,7) if within n
      if (n > 1 && 0 < n && 1 < n && arr[0] > arr[1]) begin
        next_arr[0] = arr[1];
        next_arr[1] = arr[0];
      end
      if (n > 3 && 2 < n && 3 < n && arr[2] > arr[3]) begin
        next_arr[2] = arr[3];
        next_arr[3] = arr[2];
      end
      if (n > 5 && 4 < n && 5 < n && arr[4] > arr[5]) begin
        next_arr[4] = arr[5];
        next_arr[5] = arr[4];
      end
      if (n > 7 && 6 < n && 7 < n && arr[6] > arr[7]) begin
        next_arr[6] = arr[7];
        next_arr[7] = arr[6];
      end
    end else begin
      // Odd step: compare (1,2), (3,4), (5,6) if within n
      if (n > 2 && 1 < n && 2 < n && arr[1] > arr[2]) begin
        next_arr[1] = arr[2];
        next_arr[2] = arr[1];
      end
      if (n > 4 && 3 < n && 4 < n && arr[3] > arr[4]) begin
        next_arr[3] = arr[4];
        next_arr[4] = arr[3];
      end
      if (n > 6 && 5 < n && 6 < n && arr[5] > arr[6]) begin
        next_arr[5] = arr[6];
        next_arr[6] = arr[5];
      end
    end
  end else if (state == XOR) begin
    // XOR even indices that are less than n
    for (i=0; i<N; i=i+1) begin
      if (i < n && i[0] == 0) // even index and within n
        next_arr[i] = arr[i] ^ x;
      else
        next_arr[i] = arr[i];
    end
  end
end

// Min and max calculation (combinational) for indices < n
reg [9:0] max_val, min_val;
always @(*) begin
  // Initialize with the first element if n>0, else 0
  if (n > 0) begin
    max_val = arr[0];
    min_val = arr[0];
  end else begin
    max_val = 0;
    min_val = 0;
  end
  for (i=1; i<N; i=i+1) begin
    if (i < n) begin
      if (arr[i] > max_val) max_val = arr[i];
      if (arr[i] < min_val) min_val = arr[i];
    end
  end
end

// State transition and register update
always @(posedge clk or negedge rst_n) begin
  if (!rst_n) begin
    state <= IDLE;
    k_count <= 0;
    sort_step <= 0;
    done <= 0;
    max_out <= 0;
    min_out <= 0;
    // Reset array
    for (i=0; i<N; i=i+1) begin
      arr[i] <= 0;
    end
  end else begin
    case (state)
      IDLE: begin
        done <= 0;
        if (start) begin
          // Load input array
          arr[0] <= arr_0;
          arr[1] <= arr_1;
          arr[2] <= arr_2;
          arr[3] <= arr_3;
          arr[4] <= arr_4;
          arr[5] <= arr_5;
          arr[6] <= arr_6;
          arr[7] <= arr_7;
          k_count <= k;
          sort_step <= 0;
          state <= SORT;
        end
      end

      SORT: begin
        // Update array with next_arr
        for (i=0; i<N; i=i+1) arr[i] <= next_arr[i];
        sort_step <= sort_step + 1;
        if (sort_step == 3'd7) begin
          // After 8 cycles of sorting (0 to 7), we have sorted array
          sort_step <= 0;
          state <= XOR;
        end
      end

      XOR: begin
        // Update array with next_arr (which includes XOR)
        for (i=0; i<N; i=i+1) arr[i] <= next_arr[i];
        state <= CHECK_K;
      end

      CHECK_K: begin
        if (k_count > 0) begin
          k_count <= k_count - 1;
          state <= SORT;
        end else begin
          state <= FINISH;
        end
      end

      FINISH: begin
        // Compute min and max
        max_out <= max_val;
        min_out <= min_val;
        done <= 1;
        state <= IDLE;
      end

      default: state <= IDLE;
    endcase
  end
end

endmodule
