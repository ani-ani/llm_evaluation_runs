module CycleBreaker #(
    parameter N_MAX = 8,
    parameter M_MAX = 16
)(
    input clk, rst_n, start,
    input [3:0] n,
    input [4:0] m,
    input [3:0] src [M_MAX-1:0],
    input [3:0] dst [M_MAX-1:0],
    output reg [4:0] r,
    output reg [3:0] remove_list [M_MAX/2-1:0],
    output reg done
);

// Description:
// Implements greedy algorithm to remove at most half of the edges
// while eliminating all cycles. Input src/dst are 0‑based.
// Algorithm:
// 1. Build adjacency matrix from src/dst.
// 2. While vertices remain:
//    - Compute outdegree of each active vertex.
//    - Select vertex with maximum outdegree (ties by index).
//    - Append it to ordering (assign position pos).
//    - Mark it inactive.
//    - For each active vertex u that has an edge to the selected vertex,
//      decrement outdegree[u] (edge u->selected removed).
// 3. After ordering, for each edge (u,v) if position[u] > position[v]
//    it is backward and must be removed.
// 4. Output list of edge indices to remove (1‑based).
// State machine: IDLE, INIT, FIND_MAX, APPEND, UPDATE, CHECK_DONE, COMPUTE_REMOVAL, OUTPUT, DONE.
// All loops bounded by N_MAX and M_MAX.
endmodule
