module spike_cave (
    input clk,
    input rst_n,
    input start,
    input [2:0] N,   // number of chambers (1-8)
    input [3:0] M,   // number of edges (1-16)
    input [M_MAX-1:0][39:0] edges, // each edge: [39:36]=A, [35:32]=B, [31:0]=L
    output reg [31:0] danger [0:N_MAX-1],
    output reg done
);

parameter N_MAX = 8;
parameter M_MAX = 16;
parameter MOD = 1000000007;
parameter INF = 64'hFFFFFFFF_FFFFFFFF;

// State encoding
localparam IDLE = 0, INIT = 1, LOAD_EDGES = 2, FLOYD = 3, SUM = 4, DONE = 5;

reg [2:0] state, next_state;

// Captured N and M
reg [2:0] n_reg;
reg [3:0] m_reg;

// Dist matrix - 64-bit for large distances
reg [63:0] dist [0:N_MAX-1][0:N_MAX-1];

// Counters and registers for INIT
reg [2:0] init_i, init_j;

// Counter for LOAD_EDGES
reg [3:0] edge_idx;

// Counters for FLOYD
reg [2:0] k_cnt, i_cnt, j_cnt;

// Edge decoding
wire [3:0] A = edges[edge_idx][39:36];
wire [3:0] B = edges[edge_idx][35:32];
wire [31:0] L = edges[edge_idx][31:0];

// Combinational block for sum calculation
wire [63:0] temp_sum [0:N_MAX-1];

integer i, j;
always @(*) begin
    for (i=0; i<N_MAX; i=i+1) begin
        temp_sum[i] = 0;
        for (j=0; j<N_MAX; j=j+1) begin
            if (j < n_reg) begin
                temp_sum[i] = temp_sum[i] + dist[i][j];
            end
        end
    end
end

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
    end else begin
        state <= next_state;
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: begin
            if (start) next_state = INIT;
        end
        INIT: begin
            if (init_i == n_reg-1 && init_j == n_reg-1) next_state = LOAD_EDGES;
            else next_state = INIT;
        end
        LOAD_EDGES: begin
            if (edge_idx == m_reg-1) next_state = FLOYD;
            else next_state = LOAD_EDGES;
        end
        FLOYD: begin
            if (k_cnt == n_reg-1 && i_cnt == n_reg-1 && j_cnt == n_reg-1) next_state = SUM;
            else next_state = FLOYD;
        end
        SUM: begin
            next_state = DONE;
        end
        DONE: begin
            next_state = DONE;
        end
        default: next_state = IDLE;
    endcase
end

// Datapath
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        done <= 0;
        for (i=0; i<N_MAX; i=i+1) danger[i] <= 0;
        init_i <= 0;
        init_j <= 0;
        edge_idx <= 0;
        k_cnt <= 0;
        i_cnt <= 0;
        j_cnt <= 0;
        n_reg <= 0;
        m_reg <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    n_reg <= N;
                    m_reg <= M;
                    init_i <= 0;
                    init_j <= 0;
                    edge_idx <= 0;
                    k_cnt <= 0;
                    i_cnt <= 0;
                    j_cnt <= 0;
                    done <= 0;
                end
            end

            INIT: begin
                // Initialize dist matrix
                if (init_i == init_j) begin
                    dist[init_i][init_j] <= 0;
                end else begin
                    dist[init_i][init_j] <= INF;
                end
                // Increment counters
                if (init_j == n_reg-1) begin
                    init_j <= 0;
                    if (init_i == n_reg-1) begin
                        // Done
                    end else begin
                        init_i <= init_i + 1;
                    end
                end else begin
                    init_j <= init_j + 1;
                end
            end

            LOAD_EDGES: begin
                // Process edge if within valid range
                if (A != 0 && B != 0 && A <= n_reg && B <= n_reg) begin
                    // Convert to 0-based index and assign
                    dist[A-1][B-1] <= {32'b0, L};
                    dist[B-1][A-1] <= {32'b0, L};
                end
                // Increment edge_idx
                if (edge_idx == m_reg-1) begin
                    // Done
                end else begin
                    edge_idx <= edge_idx + 1;
                end
            end

            FLOYD: begin
                // Update dist[i_cnt][j_cnt] if shorter path exists
                if (dist[i_cnt][k_cnt] < INF && dist[k_cnt][j_cnt] < INF) begin
                    if (dist[i_cnt][k_cnt] + dist[k_cnt][j_cnt] < dist[i_cnt][j_cnt]) begin
                        dist[i_cnt][j_cnt] <= dist[i_cnt][k_cnt] + dist[k_cnt][j_cnt];
                    end
                end
                // Increment counters
                if (j_cnt == n_reg-1) begin
                    j_cnt <= 0;
                    if (i_cnt == n_reg-1) begin
                        i_cnt <= 0;
                        if (k_cnt == n_reg-1) begin
                            // Done
                        end else begin
                            k_cnt <= k_cnt + 1;
                        end
                    end else begin
                        i_cnt <= i_cnt + 1;
                    end
                end else begin
                    j_cnt <= j_cnt + 1;
                end
            end

            SUM: begin
                // Calculate danger levels with modulo
                for (i=0; i<N_MAX; i=i+1) begin
                    if (i < n_reg) begin
                        danger[i] <= temp_sum[i] % MOD;
                    end else begin
                        danger[i] <= 0;
                    end
                end
                done <= 1;
            end

            DONE: begin
                done <= 1;
            end
        endcase
    end
end

endmodule
