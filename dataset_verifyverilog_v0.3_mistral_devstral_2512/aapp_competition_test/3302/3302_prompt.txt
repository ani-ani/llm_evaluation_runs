module color_code_generator (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [1:0] n,           // n value (max 4)
    input wire [3:0] p_mask,      // bit i = 1 means distance i+1 is allowed
    output reg [3:0] output_value,
    output reg output_valid,
    output reg done
);

// Parameters
parameter MAX_N = 4;
parameter MAX_STATES = 16; // 2^MAX_N

// State encoding
parameter IDLE = 3'b000;
parameter INIT = 3'b001;
parameter SEARCH = 3'b010;
parameter OUTPUT = 3'b011;
parameter COMPLETE = 3'b100;

// Internal registers
reg [2:0] state, next_state;
reg [3:0] current_value;      // Current sequence element
reg [3:0] counter;            // Number of elements generated
reg [15:0] visited;           // Visited states bitmask
reg [3:0] search_idx;         // Index for searching next value
reg found_next;               // Flag when next value found
reg [3:0] next_candidate;     // Candidate for next value

// Combinational logic for Hamming distance
function integer hamming;
    input [3:0] a;
    input [3:0] b;
    integer i;
    begin
        hamming = 0;
        for (i = 0; i < MAX_N; i = i + 1) begin
            if (a[i] != b[i]) hamming = hamming + 1;
        end
    end
endfunction

// Check if distance is allowed
function integer is_allowed;
    input [3:0] dist;
    input [3:0] mask;
    begin
        if (dist == 0 || dist > 4) begin
            is_allowed = 0;
        end else begin
            is_allowed = mask[dist-1];
        end
    end
endfunction

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        current_value <= 0;
        counter <= 0;
        visited <= 0;
        search_idx <= 0;
        found_next <= 0;
        next_candidate <= 0;
        output_value <= 0;
        output_valid <= 0;
        done <= 0;
    end else begin
        case (state)
            IDLE: begin
                output_valid <= 0;
                done <= 0;
                if (start) begin
                    state <= INIT;
                end
            end
            
            INIT: begin
                // Initialize for new sequence
                current_value <= 0;
                counter <= 0;
                visited <= 16'b1;  // Mark 0 as visited
                search_idx <= 0;
                found_next <= 0;
                // Special case: if n=0 or invalid, go to COMPLETE
                if (n == 0 || n > MAX_N) begin
                    state <= COMPLETE;
                end else begin
                    state <= OUTPUT;
                end
            end
            
            OUTPUT: begin
                // Output current value
                output_value <= current_value;
                output_valid <= 1;
                counter <= counter + 1;
                
                // Check if sequence complete
                if (counter == (1 << n) - 1) begin
                    state <= COMPLETE;
                end else begin
                    state <= SEARCH;
                    search_idx <= 0;
                    found_next <= 0;
                end
            end
            
            SEARCH: begin
                output_valid <= 0;
                
                // Search for next valid value
                if (search_idx < (1 << n)) begin
                    // Check if candidate is valid
                    if (!visited[search_idx]) begin
                        // Check Hamming distance
                        if (is_allowed(hamming(current_value, search_idx), p_mask)) begin
                            found_next <= 1;
                            next_candidate <= search_idx;
                            state <= OUTPUT;
                            // Mark as visited
                            visited[search_idx] <= 1;
                            current_value <= search_idx;
                        end else begin
                            search_idx <= search_idx + 1;
                        end
                    end else begin
                        search_idx <= search_idx + 1;
                    end
                end else begin
                    // No valid next state found
                    state <= COMPLETE;
                end
            end
            
            COMPLETE: begin
                output_valid <= 0;
                done <= 1;
                if (!start) state <= IDLE;
            end
            
            default: state <= IDLE;
        endcase
    end
end

endmodule
