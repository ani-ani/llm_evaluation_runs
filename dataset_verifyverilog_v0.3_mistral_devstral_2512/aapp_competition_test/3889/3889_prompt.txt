module puppy_standardization (
    input [127:0] string,   // 16 characters packed as 8-bit ASCII values (string[7:0]=char0)
    input [3:0] length,     // Number of valid characters (1-16)
    output reg result        // 1 if possible, 0 otherwise
);
    integer i, j;
    reg duplicate_found;

    always @(*) begin
        duplicate_found = 0;
        // Check all unique pairs for duplicates
        for (i = 0; i < 16; i = i + 1) begin
            for (j = i+1; j < 16; j = j + 1) begin
                if (i < length && j < length) begin
                    if (string[i*8 +: 8] == string[j*8 +: 8]) begin
                        duplicate_found = 1;
                    end
                end
            end
        end
        // Final result: 1 if length=1 OR duplicate found
        result = (length == 1) || duplicate_found;
    end
endmodule
