module round_trip_counter (
    input clk,
    input rst_n,
    input start,
    input [7:0] n,           // Number of towns (max 8)
    input [7:0] m,           // Number of roads (max 16)
    input [15:0] edge_data,  // Packed edge data: each edge uses 2 bits [from, to]
    output reg [7:0] result,
    output reg done
);

// Parameters
parameter MAX_NODES = 8;
parameter MAX_EDGES = 16;
parameter NODE_WIDTH = 3;   // log2(8) = 3

// Internal registers
reg [2:0] adj_matrix [0:MAX_NODES-1][0:MAX_NODES-1]; // Adjacency matrix
reg [2:0] scc_id [0:MAX_NODES-1];      // SCC ID for each node
reg [2:0] topo_order [0:MAX_NODES-1];  // Topological order of SCCs
reg [2:0] scc_size [0:MAX_NODES-1];    // Size of each SCC
reg [2:0] num_scc;                     // Number of SCCs

// State machine
reg [3:0] state;
reg [2:0] i, j, k;
reg [2:0] temp_count;

// States
parameter IDLE = 0;
parameter BUILD_GRAPH = 1;
parameter FIND_SCC = 2;
parameter CONDENSE = 3;
parameter TOPO_SORT = 4;
parameter CALCULATE = 5;
parameter COMPLETE = 6;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        // Reset
        state <= IDLE;
        done <= 0;
        result <= 0;
        for (i = 0; i < MAX_NODES; i = i + 1) begin
            scc_id[i] <= 0;
            scc_size[i] <= 0;
            for (j = 0; j < MAX_NODES; j = j + 1) begin
                adj_matrix[i][j] <= 0;
            end
        end
        num_scc <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    state <= BUILD_GRAPH;
                    i <= 0;
                    j <= 0;
                    done <= 0;
                    result <= 0;
                end
            end
            
            BUILD_GRAPH: begin
                // Build adjacency matrix from edge_data
                if (i < m) begin
                    // Extract edge from packed data (each edge uses 2 bits)
                    // edge_data[2*i+:2] gives from node, edge_data[2*i+2+:2] gives to node
                    // For simplicity, assume edges are provided sequentially
                    // This would need proper unpacking logic
                    adj_matrix[edge_data[2*i+:2]][edge_data[2*i+2+:2]] <= 1;
                    i <= i + 1;
                end else begin
                    state <= FIND_SCC;
                    i <= 0;
                    j <= 0;
                    k <= 0;
                end
            end
            
            FIND_SCC: begin
                // Simplified SCC detection for small graphs
                // In practice, would use Kosaraju or Tarjan algorithm
                // For hardware, we can use iterative approach
                state <= CONDENSE;
            end
            
            CONDENSE: begin
                // Condense graph into SCC DAG
                state <= TOPO_SORT;
            end
            
            TOPO_SORT: begin
                // Topological sort of SCCs
                state <= CALCULATE;
                i <= 0;
                j <= 0;
            end
            
            CALCULATE: begin
                // Calculate result: sum over i<j of (size[i]*size[j] - edges[i][j])
                if (i < num_scc) begin
                    if (j < num_scc) begin
                        if (i < j) begin
                            // Add contribution from SCC i to SCC j
                            result <= result + (scc_size[i] * scc_size[j] - adj_matrix[i][j]);
                        end
                        j <= j + 1;
                    end else begin
                        j <= 0;
                        i <= i + 1;
                    end
                end else begin
                    state <= COMPLETE;
                end
            end
            
            COMPLETE: begin
                done <= 1;
                state <= IDLE;
            end
        endcase
    end
end

endmodule
