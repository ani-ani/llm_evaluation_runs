module gerald_coins (
    input clk,
    input rst_n,
    input start,
    input [63:0] n,
    output [63:0] result,
    output done
);

    // States
    localparam IDLE = 2'b00;
    localparam DIVIDE = 2'b01;
    localparam COMPUTE = 2'b10;
    localparam FINISH = 2'b11;

    reg [1:0] state, next_state;
    reg [63:0] current_n;
    reg [63:0] quotient;
    reg [1:0] remainder;
    reg divider_start;
    reg [5:0] counter; // For 64-bit division
    reg [63:0] result_reg;
    reg done_reg;

    // Divider module for division by 3
    divider_3 divider (
        .clk(clk),
        .rst_n(rst_n),
        .start(divider_start),
        .dividend(current_n),
        .quotient(quotient),
        .remainder(remainder),
        .done(divider_done)
    );

    // State transition
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
        end else begin
            state <= next_state;
        end
    end

    // Next state logic
    always @(*) begin
        case (state)
            IDLE: next_state = start ? DIVIDE : IDLE;
            DIVIDE: next_state = divider_done ? COMPUTE : DIVIDE;
            COMPUTE: begin
                if (remainder == 0) begin
                    next_state = DIVIDE; // Continue dividing
                end else begin
                    next_state = FINISH;
                end
            end
            FINISH: next_state = IDLE;
            default: next_state = IDLE;
        endcase
    end

    // Output and datapath logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            current_n <= 64'd0;
            divider_start <= 1'b0;
            result_reg <= 64'd0;
            done_reg <= 1'b0;
        end else begin
            done_reg <= 1'b0;
            divider_start <= 1'b0;

            case (state)
                IDLE: begin
                    if (start) begin
                        current_n <= n;
                    end
                end
                DIVIDE: begin
                    divider_start <= 1'b1;
                end
                COMPUTE: begin
                    if (remainder == 0) begin
                        current_n <= quotient; // Continue with n/3
                    end else begin
                        result_reg <= quotient + 1; // n/3 + 1
                        done_reg <= 1'b1;
                    end
                end
                FINISH: begin
                    // Wait for next start
                end
            endcase
        end
    end

    assign result = result_reg;
    assign done = done_reg;

endmodule

// Divider by 3 module
module divider_3 (
    input clk,
    input rst_n,
    input start,
    input [63:0] dividend,
    output [63:0] quotient,
    output [1:0] remainder,
    output done
);

    reg [63:0] quotient_reg;
    reg [1:0] remainder_reg;
    reg running;
    reg [5:0] counter;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            quotient_reg <= 64'd0;
            remainder_reg <= 2'd0;
            running <= 1'b0;
            counter <= 6'd0;
        end else begin
            if (start && !running) begin
                // Start new division
                running <= 1'b1;
                counter <= 6'd0;
                quotient_reg <= 64'd0;
                remainder_reg <= 2'd0;
            end else if (running) begin
                if (counter < 64) begin
                    // Shift remainder left by 1 and bring in next bit
                    remainder_reg <= (remainder_reg << 1) | (dividend[63-counter]);
                    
                    // If remainder >= 3, subtract 3 and set quotient bit
                    if ({(remainder_reg << 1) | (dividend[63-counter])} >= 2'd3) begin
                        remainder_reg <= ((remainder_reg << 1) | (dividend[63-counter])) - 2'd3;
                        quotient_reg <= {quotient_reg[62:0], 1'b1};
                    end else begin
                        quotient_reg <= {quotient_reg[62:0], 1'b0};
                    end
                    
                    counter <= counter + 1;
                end else begin
                    running <= 1'b0;
                end
            end
        end
    end

    assign quotient = quotient_reg;
    assign remainder = remainder_reg;
    assign done = !running && (counter == 64) && (start || running);

endmodule
