module laser_tag #(
    parameter WIDTH = 32,
    parameter FRAC_BITS = 16
) (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [WIDTH-1:0] x1, y1, x2, y2, x3, y3,
    output reg [WIDTH-1:0] low_y,
    output reg [WIDTH-1:0] high_y,
    output reg low_inf,
    output reg high_inf,
    output reg can_hit,
    output reg done
);
    // State definitions
    localparam IDLE = 0;
    localparam COMPUTE_A_B = 1;
    localparam COMPUTE_C = 2;
    localparam COMPUTE_D = 3;
    localparam COMPUTE_DENOM = 4;
    localparam COMPUTE_S_PRIME = 5;
    localparam CHECK_CAN_HIT = 6;
    localparam COMPUTE_ENDPOINT1 = 7;
    localparam COMPUTE_ENDPOINT2 = 8;
    localparam DETERMINE_RESULT = 9;
    localparam DONE = 10;

    reg [4:0] state;
    reg [WIDTH-1:0] A, B, C, D, denom;
    reg [WIDTH-1:0] Sx, Sy;  // Virtual shooter coordinates
    reg [WIDTH-1:0] x_min, x_max;
    reg [WIDTH-1:0] dx, dy, t, y_wall;
    reg [1:0] endpoint_count;
    reg [WIDTH-1:0] y_values[0:1];
    reg inf_flags[0:1];  // 0=finite, 1=infinite
    reg signed_flag[0:1]; // For infinity direction: 0=negative, 1=positive
    
    // Helper: Fixed-point multiplication
    function [WIDTH-1:0] mult;
        input [WIDTH-1:0] a, b;
        begin
            mult = (a * b) >>> FRAC_BITS;
        end
    endfunction
    
    // Helper: Fixed-point division (simplified - assumes denominator is power of 2)
    // Real implementation would need sequential divider
    function [WIDTH-1:0] divide;
        input [WIDTH-1:0] num, denom;
        begin
            // For simulation, use integer division
            divide = num / denom;
        end
    endfunction
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            low_inf <= 0;
            high_inf <= 0;
            can_hit <= 0;
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        // Latch inputs
                        state <= COMPUTE_A_B;
                        x_min <= (x1 < x2) ? x1 : x2;
                        x_max <= (x1 > x2) ? x1 : x2;
                        endpoint_count <= 0;
                    end
                end
                
                COMPUTE_A_B: begin
                    // A = y2 - y1, B = x1 - x2
                    A <= y2 - y1;
                    B <= x1 - x2;
                    state <= COMPUTE_C;
                end
                
                COMPUTE_C: begin
                    // C = -(A*x1 + B*y1)
                    C <= -(mult(A, x1) + mult(B, y1));
                    state <= COMPUTE_D;
                end
                
                COMPUTE_D: begin
                    // D = A*x3 + B*y3 + C
                    D <= mult(A, x3) + mult(B, y3) + C;
                    state <= COMPUTE_DENOM;
                end
                
                COMPUTE_DENOM: begin
                    // denom = A*A + B*B
                    denom <= mult(A, A) + mult(B, B);
                    state <= COMPUTE_S_PRIME;
                end
                
                COMPUTE_S_PRIME: begin
                    // Sx = x3 - (2*A*D)/denom
                    // Sy = y3 - (2*B*D)/denom
                    if (denom != 0) begin
                        Sx <= x3 - (2 * mult(A, D)) / denom;
                        Sy <= y3 - (2 * mult(B, D)) / denom;
                    end else begin
                        Sx <= 0;
                        Sy <= 0;
                    end
                    state <= CHECK_CAN_HIT;
                end
                
                CHECK_CAN_HIT: begin
                    // Check if mirror intersects interval between 0 and Sx
                    if ((Sx > 0 && x_min >= Sx) || (Sx < 0 && x_max <= Sx) || (Sx == 0)) begin
                        can_hit <= 0;
                        state <= DONE;
                    end else begin
                        can_hit <= 1;
                        state <= COMPUTE_ENDPOINT1;
                    end
                end
                
                COMPUTE_ENDPOINT1, COMPUTE_ENDPOINT2: begin
                    if (can_hit) begin
                        // Get current endpoint
                        if (state == COMPUTE_ENDPOINT1) begin
                            dx <= x1 - Sx;
                            dy <= y1 - Sy;
                        end else begin
                            dx <= x2 - Sx;
                            dy <= y2 - Sy;
                        end
                        // Check for infinite case (dx == 0)
                        if (dx == 0) begin
                            inf_flags[endpoint_count] <= 1;
                            // Determine infinity direction
                            if (dy > 0) begin
                                signed_flag[endpoint_count] <= 1; // positive infinity
                            end else begin
                                signed_flag[endpoint_count] <= 0; // negative infinity
                            end
                        end else begin
                            inf_flags[endpoint_count] <= 0;
                            // Compute t = -Sx / dx
                            t <= (-Sx) / dx;  // Integer division, need proper handling
                            // Compute y = Sy + t * dy
                            y_values[endpoint_count] <= Sy + mult(t, dy);
                        end
                        endpoint_count <= endpoint_count + 1;
                        if (endpoint_count == 0) begin
                            state <= COMPUTE_ENDPOINT2;
                        end else begin
                            state <= DETERMINE_RESULT;
                        end
                    end
                end
                
                DETERMINE_RESULT: begin
                    // Determine low and high from y_values and inf_flags
                    if (inf_flags[0] && inf_flags[1]) begin
                        low_inf <= signed_flag[0] ? 0 : 1;
                        high_inf <= signed_flag[1] ? 1 : 0;
                        low_y <= 0;
                        high_y <= 0;
                    end else if (inf_flags[0]) begin
                        low_inf <= 1;
                        high_inf <= 0;
                        low_y <= 0;
                        high_y <= y_values[1];
                    end else if (inf_flags[1]) begin
                        low_inf <= 0;
                        high_inf <= 1;
                        low_y <= y_values[0];
                        high_y <= 0;
                    end else begin
                        low_inf <= 0;
                        high_inf <= 0;
                        low_y <= (y_values[0] < y_values[1]) ? y_values[0] : y_values[1];
                        high_y <= (y_values[0] > y_values[1]) ? y_values[0] : y_values[1];
                    end
                    state <= DONE;
                end
                
                DONE: begin
                    done <= 1;
                    if (!start) begin
                        state <= IDLE;
                        done <= 0;
                    end
                end
            endcase
        end
    end
endmodule
