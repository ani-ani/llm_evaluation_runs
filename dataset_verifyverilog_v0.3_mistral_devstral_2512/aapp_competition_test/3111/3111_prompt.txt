module dial_game (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] A,  // 1-indexed, max 8
    input wire [3:0] B,
    output reg [6:0] result,
    output reg done
);
    // Parameters for easy scaling
    parameter N = 8;          // Number of dials
    parameter WIDTH = 4;      // Bits per dial (0-9 fits in 4 bits)

    // Storage for dial values
    reg [WIDTH-1:0] dials [0:N-1];

    // State definitions
    localparam IDLE = 2'b00;
    localparam CALC_SUM = 2'b01;
    localparam UPDATE = 2'b10;
    localparam DONE = 2'b11;

    reg [1:0] state;
    reg [3:0] idx;           // Iteration index
    reg [6:0] sum;           // Accumulator for sum (max 8*9=72)
    reg [3:0] a_reg, b_reg;  // Store A and B (0-indexed)

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            result <= 0;
            // Dials are not reset here; testbench will initialize them.
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        // Convert 1-indexed to 0-indexed and store
                        a_reg <= A - 4'd1;
                        b_reg <= B - 4'd1;
                        sum <= 0;
                        idx <= A - 4'd1;
                        state <= CALC_SUM;
                    end
                end

                CALC_SUM: begin
                    if (idx <= b_reg) begin
                        sum <= sum + dials[idx];
                        idx <= idx + 1;
                    end else begin
                        result <= sum;
                        idx <= a_reg;
                        state <= UPDATE;
                    end
                end

                UPDATE: begin
                    if (idx <= b_reg) begin
                        // Increment modulo 10
                        if (dials[idx] == 4'd9)
                            dials[idx] <= 4'd0;
                        else
                            dials[idx] <= dials[idx] + 4'd1;
                        idx <= idx + 1;
                    end else begin
                        state <= DONE;
                        done <= 1;
                    end
                end

                DONE: begin
                    done <= 0;
                    state <= IDLE;
                end

                default: state <= IDLE;
            endcase
        end
    end
endmodule
