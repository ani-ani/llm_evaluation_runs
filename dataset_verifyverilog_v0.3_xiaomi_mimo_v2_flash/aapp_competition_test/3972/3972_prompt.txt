module sequence_counter (
    input [3:0] n,  // n from 1 to 16
    output reg [31:0] result
);

localparam MOD = 1000000007;

reg [31:0] dp [0:15];  // dp[0] to dp[15]
integer i;
reg [31:0] dp_sum;
reg [31:0] term;

always @(*) begin
    // Initialize dp array
    for (i = 0; i < 16; i = i + 1) begin
        dp[i] = 0;
    end
    
    if (n == 1) begin
        result = 1;
    end else begin
        // Initialize base cases
        dp[0] = 1;
        dp[1] = 1;
        dp_sum = 2;
        
        // Compute dp[2] to dp[n-1]
        for (i = 2; i < n; i = i + 1) begin
            dp[i] = (dp_sum - dp[i-2] + MOD) % MOD;
            dp_sum = (dp_sum + dp[i]) % MOD;
        end
        
        // Compute the final answer
        result = 0;
        for (i = 0; i < n-1; i = i + 1) begin
            if (i >= n-3) begin
                term = ((n-1) * (n-1) + (n-1)) % MOD;
            end else begin
                term = ((n-1) * (n-1) + (i+2)) % MOD;
            end
            result = (result + dp[i] * term) % MOD;
        end
        result = (result + dp[n-1] * n) % MOD;
    end
end

endmodule
