module kernel_area (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [2:0] n,
    input wire signed [31:0] x [0:7],
    input wire signed [31:0] y [0:7],
    output reg [63:0] area,
    output reg done
);
    // Combinational calculation of sum of cross products
    reg signed [63:0] sum_comb;
    integer i;
    always @(*) begin
        sum_comb = 0;
        for (i = 0; i < 8; i = i + 1) begin
            if (i < n) begin
                if (i < n-1) begin
                    sum_comb = sum_comb + (x[i] * y[i+1] - x[i+1] * y[i]);
                end else begin
                    sum_comb = sum_comb + (x[i] * y[0] - x[0] * y[i]);
                end
            end
        end
    end

    // Sequential logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            area <= 0;
            done <= 0;
        end else begin
            if (start) begin
                // Compute absolute value and divide by 2
                if (sum_comb < 0)
                    area <= (-sum_comb) >> 1;
                else
                    area <= sum_comb >> 1;
                done <= 1;
            end else begin
                done <= 0;
            end
        end
    end
endmodule
