module aladin_device (
    input clk,
    input rst_n,
    input start,
    input [1:0] cmd_type,   // 1: update, 2: query
    input [15:0] L, R,      // 16-bit, but we only use 1..16
    input [19:0] A, B,      // 20-bit
    output reg [31:0] result,
    output reg done
);

parameter BOXES = 16;

// Box storage: 1-indexed, boxes 1 to 16
reg [31:0] box [1:BOXES];

// State machine
reg [1:0] state;
localparam IDLE = 2'd0, UPDATE = 2'd1, QUERY = 2'd2;

reg [15:0] current_index, step;
reg [31:0] sum;
integer i;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        // Reset all boxes to 0
        for (i = 1; i <= BOXES; i = i + 1) begin
            box[i] <= 0;
        end
        state <= IDLE;
        result <= 0;
        done <= 0;
        current_index <= 0;
        step <= 0;
        sum <= 0;
    end else begin
        done <= 0; // default

        case (state)
            IDLE: begin
                if (start) begin
                    if (cmd_type == 1) begin
                        // Update command
                        current_index <= L;
                        step <= 1;
                        state <= UPDATE;
                    end else if (cmd_type == 2) begin
                        // Query command
                        current_index <= L;
                        sum <= 0;
                        state <= QUERY;
                    end
                end
            end

            UPDATE: begin
                if (current_index <= R && current_index <= BOXES) begin
                    // Compute value for current box
                    box[current_index] <= (step * A) % B;
                    current_index <= current_index + 1;
                    step <= step + 1;
                end else begin
                    // Finished update
                    state <= IDLE;
                    done <= 1;
                end
            end

            QUERY: begin
                if (current_index <= R && current_index <= BOXES) begin
                    sum <= sum + box[current_index];
                    current_index <= current_index + 1;
                end else begin
                    state <= IDLE;
                    result <= sum;
                    done <= 1;
                end
            end
        endcase
    end
end

endmodule
