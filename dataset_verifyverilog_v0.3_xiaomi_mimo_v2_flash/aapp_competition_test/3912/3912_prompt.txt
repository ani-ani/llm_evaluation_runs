module palindrome_cutter (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [7:0] char_in,   // ASCII character input
    input wire [3:0] len,        // Actual string length (1-15)
    output reg [3:0] k,          // Number of palindromes
    output reg [7:0] palindrome_mem [0:15][0:15], // 16 palindromes, max 16 chars each
    output reg [3:0] palindrome_len [0:15],       // Length of each palindrome
    output reg done
);

    // States
    localparam IDLE = 3'b000;
    localparam READ = 3'b001;
    localparam COMPUTE_K = 3'b010;
    localparam BUILD = 3'b011;
    localparam OUTPUT = 3'b100;

    reg [2:0] state;
    reg [3:0] idx;               // Character index during read/build
    reg [3:0] ctr [0:3];         // Counters for 'a' to 'd'
    reg [3:0] odd_count;         // Number of characters with odd frequency
    reg [7:0] temp_palindrome [0:15]; // Temporary storage for current palindrome
    reg [3:0] temp_idx;          // Index for temp_palindrome
    reg [1:0] char_idx;          // 0-3 for 'a'-'d'

    // Helper: Map ASCII to index
    function [1:0] char_to_idx(input [7:0] c);
        case(c)
            8'h61: char_to_idx = 0; // 'a'
            8'h62: char_to_idx = 1; // 'b'
            8'h63: char_to_idx = 2; // 'c'
            8'h64: char_to_idx = 3; // 'd'
            default: char_to_idx = 0;
        endcase
    endfunction

    // Helper: Get ASCII character from index
    function [7:0] idx_to_char(input [1:0] i);
        case(i)
            0: idx_to_char = 8'h61; // 'a'
            1: idx_to_char = 8'h62; // 'b'
            2: idx_to_char = 8'h63; // 'c'
            3: idx_to_char = 8'h64; // 'd'
            default: idx_to_char = 8'h20; // space
        endcase
    endfunction

    integer i, j;

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            k <= 0;
            for (i = 0; i < 4; i = i + 1) ctr[i] <= 0;
            for (i = 0; i < 16; i = i + 1) palindrome_len[i] <= 0;
            idx <= 0;
            temp_idx <= 0;
            odd_count <= 0;
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        state <= READ;
                        idx <= 0;
                        for (i = 0; i < 4; i = i + 1) ctr[i] <= 0;
                    end
                end

                READ: begin
                    if (idx < len) begin
                        char_idx <= char_to_idx(char_in);
                        ctr[char_to_idx(char_in)] <= ctr[char_to_idx(char_in)] + 1;
                        idx <= idx + 1;
                    end else begin
                        state <= COMPUTE_K;
                        idx <= 0;
                    end
                end

                COMPUTE_K: begin
                    // Count odd frequencies
                    odd_count <= 0;
                    for (i = 0; i < 4; i = i + 1) begin
                        if (ctr[i][0]) odd_count <= odd_count + 1;
                    end
                    // Determine k
                    if (odd_count <= 1) k <= 1;
                    else k <= len;
                    state <= BUILD;
                    idx <= 0;
                    temp_idx <= 0;
                    // Reset palindrome memory lengths
                    for (i = 0; i < 16; i = i + 1) palindrome_len[i] <= 0;
                end

                BUILD: begin
                    if (k == 1) begin
                        // Build single palindrome
                        if (idx < 4) begin
                            if (ctr[idx] > 0) begin
                                if (ctr[idx][0]) begin
                                    // Odd: place one in center later
                                    for (j = 0; j < (ctr[idx] - 1) / 2; j = j + 1) begin
                                        if (temp_idx < 16) begin
                                            temp_palindrome[temp_idx] <= idx_to_char(idx);
                                            temp_idx <= temp_idx + 1;
                                        end
                                    end
                                end else begin
                                    // Even: split equally
                                    for (j = 0; j < ctr[idx] / 2; j = j + 1) begin
                                        if (temp_idx < 16) begin
                                            temp_palindrome[temp_idx] <= idx_to_char(idx);
                                            temp_idx <= temp_idx + 1;
                                        end
                                    end
                                end
                            end
                            idx <= idx + 1;
                        end else if (idx == 4) begin
                            // Add center character if any odd
                            for (i = 0; i < 4; i = i + 1) begin
                                if (ctr[i][0]) begin
                                    if (temp_idx < 16) begin
                                        temp_palindrome[temp_idx] <= idx_to_char(i);
                                        temp_idx <= temp_idx + 1;
                                    end
                                end
                            end
                            idx <= idx + 1;
                        end else begin
                            // Add reversed part
                            if (temp_idx > 0) begin
                                temp_idx <= temp_idx - 1;
                                if (temp_idx - 1 >= 0) begin
                                    palindrome_mem[0][16 - (idx - 5)] <= temp_palindrome[temp_idx - 1];
                                end
                                idx <= idx + 1;
                            end else begin
                                palindrome_len[0] <= 16 - (idx - 5);
                                state <= OUTPUT;
                            end
                        end
                    end else begin // k == len
                        if (idx < len) begin
                            // Each character is a single-character palindrome
                            palindrome_mem[idx][0] <= char_in;
                            palindrome_len[idx] <= 1;
                            idx <= idx + 1;
                        end else begin
                            state <= OUTPUT;
                        end
                    end
                end

                OUTPUT: begin
                    done <= 1;
                    state <= IDLE;
                end
            endcase
        end
    end
endmodule
