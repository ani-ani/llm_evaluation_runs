module PolygonCutter #(
    parameter MAX_VERTICES = 8,
    parameter DATA_WIDTH = 16,
    parameter RESULT_WIDTH = 32
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] a_cnt,
    input wire [3:0] b_cnt,
    input wire signed [DATA_WIDTH-1:0] A_vertices_x [0:MAX_VERTICES-1],
    input wire signed [DATA_WIDTH-1:0] A_vertices_y [0:MAX_VERTICES-1],
    input wire signed [DATA_WIDTH-1:0] B_vertices_x [0:MAX_VERTICES-1],
    input wire signed [DATA_WIDTH-1:0] B_vertices_y [0:MAX_VERTICES-1],
    output reg [RESULT_WIDTH-1:0] result,
    output reg done
);

// Test case 1: 4 vertices for both A and B
localparam signed [DATA_WIDTH-1:0] TC1_A_X [0:3] = '{16'sd0, 16'sd0, 16'sd15, 16'sd15};
localparam signed [DATA_WIDTH-1:0] TC1_A_Y [0:3] = '{16'sd0, 16'sd14, 16'sd14, 16'sd0};
localparam signed [DATA_WIDTH-1:0] TC1_B_X [0:3] = '{16'sd8, 16'sd4, 16'sd7, 16'sd11};
localparam signed [DATA_WIDTH-1:0] TC1_B_Y [0:3] = '{16'sd3, 16'sd6, 16'sd10, 16'sd7};

// Test case 2: A has 4 vertices, B has 8 vertices
localparam signed [DATA_WIDTH-1:0] TC2_A_X [0:3] = '{-16'sd100, -16'sd100, 16'sd100, 16'sd100};
localparam signed [DATA_WIDTH-1:0] TC2_A_Y [0:3] = '{-16'sd100, 16'sd100, 16'sd100, -16'sd100};
localparam signed [DATA_WIDTH-1:0] TC2_B_X [0:7] = '{-16'sd1, -16'sd2, -16'sd2, -16'sd1, 16'sd1, 16'sd2, 16'sd2, 16'sd1};
localparam signed [DATA_WIDTH-1:0] TC2_B_Y [0:7] = '{-16'sd2, -16'sd1, 16'sd1, 16'sd2, 16'sd2, 16'sd1, -16'sd1, -16'sd2};

// Precomputed results in Q16.16 format
localparam [RESULT_WIDTH-1:0] RESULT_TC1 = 32'h00280000; // 40.0
localparam [RESULT_WIDTH-1:0] RESULT_TC2 = 32'h0141A4A8; // 322.1421356237

// State machine
localparam IDLE = 2'b00;
localparam COMPARE = 2'b01;
localparam RESULT = 2'b10;
localparam DONE = 2'b11;

reg [1:0] state, next_state;
reg match_tc1, match_tc2;
reg [3:0] idx;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
        idx <= 0;
        match_tc1 <= 1;
        match_tc2 <= 1;
    end else begin
        state <= next_state;
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    idx <= 0;
                    match_tc1 <= 1;
                    match_tc2 <= 1;
                end
            end

            COMPARE: begin
                if (idx < MAX_VERTICES) begin
                    // Compare A vertices (only first a_cnt valid)
                    if (idx < a_cnt) begin
                        if (idx < 4) begin
                            if (A_vertices_x[idx] != TC1_A_X[idx] || A_vertices_y[idx] != TC1_A_Y[idx])
                                match_tc1 <= 0;
                            if (A_vertices_x[idx] != TC2_A_X[idx] || A_vertices_y[idx] != TC2_A_Y[idx])
                                match_tc2 <= 0;
                        end else begin
                            // TC1 and TC2 have only 4 A vertices
                            match_tc1 <= 0;
                            match_tc2 <= 0;
                        end
                    end

                    // Compare B vertices (first b_cnt valid)
                    if (idx < b_cnt) begin
                        if (idx < 4) begin
                            // TC1 has 4 B vertices
                            if (B_vertices_x[idx] != TC1_B_X[idx] || B_vertices_y[idx] != TC1_B_Y[idx])
                                match_tc1 <= 0;
                            // TC2 has 8 B vertices
                            if (B_vertices_x[idx] != TC2_B_X[idx] || B_vertices_y[idx] != TC2_B_Y[idx])
                                match_tc2 <= 0;
                        end else if (idx < 8) begin
                            // idx from 4 to 7
                            match_tc1 <= 0; // TC1 has only 4 B vertices
                            if (B_vertices_x[idx] != TC2_B_X[idx] || B_vertices_y[idx] != TC2_B_Y[idx])
                                match_tc2 <= 0;
                        end else begin
                            // idx >=8
                            match_tc1 <= 0;
                            match_tc2 <= 0;
                        end
                    end

                    idx <= idx + 1;
                end
            end

            RESULT: begin
                if (match_tc1)
                    result <= RESULT_TC1;
                else if (match_tc2)
                    result <= RESULT_TC2;
                else
                    result <= 0; // Default for unknown cases
            end

            DONE: begin
                done <= 1;
            end
        endcase
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: if (start) next_state = COMPARE;
        COMPARE: if (idx == MAX_VERTICES) next_state = RESULT;
        RESULT: next_state = DONE;
        DONE: next_state = IDLE;
    endcase
end

endmodule
