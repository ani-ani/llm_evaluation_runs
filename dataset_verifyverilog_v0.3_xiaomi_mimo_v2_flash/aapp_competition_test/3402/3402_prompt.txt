// Password Recovery Module
// Finds characters in f^K(S) where f expands each character via substitution strings T[a-z]
// Scaled constraints: |S|<=16, |T_i|<=8, K<=256, M<=8 queries
module password_recovery (
    input clk,
    input rst_n,
    input start,
    
    // Configuration
    input [7:0] K,          // Iterations (1-256)
    input [3:0] M,          // Queries (1-8)
    
    // Data loading interface
    input load_en,
    input [4:0] load_addr,  // 0-15=S, 16-41=T[0-25]
    input [7:0] load_data,
    input [3:0] load_len,   // Length for current entity
    
    // Query loading
    input query_load_en,
    input [2:0] query_addr,
    input [31:0] query_pos,
    
    // Results
    output reg [7:0] result_char,
    output reg [2:0] result_addr,
    output reg result_valid,
    output reg done,
    output reg error
);

// Memory arrays
reg [7:0] S [0:15];
reg [3:0] S_len;
reg [7:0] T [0:25][0:7];
reg [3:0] T_len [0:25];
reg [31:0] query_pos_arr [0:7];
reg [7:0] results [0:7];

// Length table: length_table[c][k] = length of expanding char c k times
reg [31:0] length_table [0:25][0:256];

// State machine
reg [3:0] state;
localparam IDLE = 0, PRECOMP = 1, PRECOMP_LOOP = 2, PRECOMP_SUM = 3;
localparam COMPUTE = 4, FIND_CHAR = 5, UPDATE = 6, OUTPUT = 7, DONE = 8;

// Processing registers
reg [7:0] cur_q, cur_level, cur_char;
reg [31:0] cur_pos, cumulative;
reg [3:0] cur_len, idx;
reg [7:0] cur_str [0:7];
reg [7:0] pc, pl, pi;  // Precompute counters
reg [31:0] temp_sum;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE; done <= 0; error <= 0; result_valid <= 0; result_addr <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (load_en) begin
                    if (load_addr < 16) begin
                        S[load_addr] <= load_data;
                        if (load_addr == 15) S_len <= load_len;
                    end else if (load_addr >= 16 && load_addr < 42) begin
                        // Simplified: assume sequential loading of T chars
                        integer char_idx = load_addr - 16;
                        integer char_pos = load_len[2:0]; // Use load_len low bits as position
                        T[char_idx][char_pos] <= load_data;
                        T_len[char_idx] <= load_len[7:4]; // High bits for length
                    end
                end
                if (query_load_en) query_pos_arr[query_addr] <= query_pos;
                if (start && load_addr == 41) begin // All data loaded
                    if (S_len == 0 || K == 0 || K > 256 || M == 0 || M > 8) begin
                        error <= 1; state <= DONE;
                    end else begin
                        cur_q <= 0; pc <= 0; pl <= 0; state <= PRECOMP;
                    end
                end
            end
            
            PRECOMP: begin
                if (pc < 26) begin
                    length_table[pc][0] <= 1; pc <= pc + 1;
                end else begin
                    pc <= 0; pl <= 1; state <= (K > 0) ? PRECOMP_LOOP : COMPUTE;
                end
            end
            
            PRECOMP_LOOP: begin
                if (pl <= K) begin
                    if (pc < 26) begin
                        temp_sum <= 0; pi <= 0; state <= PRECOMP_SUM;
                    end else begin
                        pc <= 0; pl <= pl + 1;
                    end
                end else state <= COMPUTE;
            end
            
            PRECOMP_SUM: begin
                if (pi < T_len[pc]) begin
                    temp_sum <= temp_sum + length_table[T[pc][pi]][pl-1]; pi <= pi + 1;
                end else begin
                    length_table[pc][pl] <= temp_sum; pc <= pc + 1; state <= PRECOMP_LOOP;
                end
            end
            
            COMPUTE: begin
                if (cur_q < M) begin
                    cur_level <= K; cur_len <= S_len; cur_pos <= query_pos_arr[cur_q];
                    for (integer i = 0; i < 8; i = i + 1) cur_str[i] <= (i < S_len) ? S[i] : 0;
                    state <= FIND_CHAR;
                end else state <= OUTPUT;
            end
            
            FIND_CHAR: begin
                if (cur_level == 0) begin
                    if (cur_pos > 0 && cur_pos <= cur_len) begin
                        results[cur_q] <= cur_str[cur_pos-1]; cur_q <= cur_q + 1; state <= COMPUTE;
                    end else begin error <= 1; state <= DONE; end
                end else begin
                    cumulative <= 0; idx <= 0; state <= UPDATE;
                end
            end
            
            UPDATE: begin
                if (idx < cur_len) begin
                    cur_char <= cur_str[idx];
                    if (cur_pos <= cumulative + length_table[cur_str[idx]][cur_level]) begin
                        cur_pos <= cur_pos - cumulative; cur_len <= T_len[cur_str[idx]];
                        for (integer i = 0; i < 8; i = i + 1) cur_str[i] <= (i < T_len[cur_str[idx]]) ? T[cur_str[idx]][i] : 0;
                        cur_level <= cur_level - 1; state <= FIND_CHAR;
                    end else begin
                        cumulative <= cumulative + length_table[cur_str[idx]][cur_level]; idx <= idx + 1;
                    end
                end else begin error <= 1; state <= DONE; end
            end
            
            OUTPUT: begin
                if (result_addr < M) begin
                    result_char <= results[result_addr]; result_addr <= result_addr + 1; result_valid <= 1;
                end else begin result_valid <= 0; done <= 1; state <= DONE; end
            end
            
            DONE: begin
                if (!start) begin state <= IDLE; done <= 0; result_addr <= 0; end
            end
        endcase
    end
end

endmodule
