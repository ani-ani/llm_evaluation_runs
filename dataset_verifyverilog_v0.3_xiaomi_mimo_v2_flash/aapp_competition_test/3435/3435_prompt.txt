module SpyIDCounter #(
    parameter MAX_N = 8,
    parameter MAX_M = 6
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] n,              // ID length (1 to MAX_N)
    input wire [3:0] pattern_len,    // Pattern length (1 to MAX_M)
    input wire [MAX_M-1:0] pattern_mask, // 1= fixed 1, 0= wildcard
    output reg [15:0] result,
    output reg done
);

// Internal state for DP
reg [7:0] dp_vector [0:(2*2**(MAX_M-1)-1)]; // 64 counters, 8 bits each
reg [3:0] current_pos;
reg computing;
reg [15:0] result_reg;
reg start_delayed;

// Helper signals
wire [MAX_M-1:0] mask_eff;
assign mask_eff = pattern_mask & ((1 << pattern_len) - 1);

// State encoding: {has_match, suffix}
// suffix width = pattern_len - 1 bits (stored in MAX_M-1 bits)

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        // Reset
        result <= 16'd0;
        done <= 1'b0;
        computing <= 1'b0;
        current_pos <= 4'd0;
        start_delayed <= 1'b0;
        // Clear dp_vector
        for (integer i = 0; i < (2*2**(MAX_M-1)); i = i + 1) begin
            dp_vector[i] <= 8'd0;
        end
    end else begin
        done <= 1'b0;
        start_delayed <= start;
        
        if (start && !start_delayed) begin
            // Start pulse detected
            current_pos <= 4'd0;
            
            if (pattern_len > n) begin
                // Case 1: pattern longer than ID
                result <= 16'd0;
                done <= 1'b1;
                computing <= 1'b0;
            end else if (pattern_len == n) begin
                // Case 2: pattern length equals ID length
                // Count wildcards = n - popcount(pattern_mask[0:n-1])
                // Result = 2^wildcards
                integer wildcards = n - popcount(mask_eff);
                result <= (wildcards >= 0 && wildcards <= 16) ? (16'd1 << wildcards) : 16'd0;
                done <= 1'b1;
                computing <= 1'b0;
            end else begin
                // Case 3: pattern length < ID length
                // Initialize DP vector
                for (integer i = 0; i < (2*2**(MAX_M-1)); i = i + 1) begin
                    dp_vector[i] <= 8'd0;
                end
                // Initial state: has_match=0, suffix=0, count=1
                dp_vector[0] <= 8'd1; // State index 0 = {0, 0}
                computing <= 1'b1;
            end
        end else if (computing) begin
            if (current_pos < n) begin
                // Perform DP transition for this position
                // Compute next_dp_vector from current dp_vector
                // This is combinational logic, but we implement it sequentially here for clarity
                
                // For each state in dp_vector
                for (integer state_idx = 0; state_idx < (2*2**(MAX_M-1)); state_idx = state_idx + 1) begin
                    if (dp_vector[state_idx] != 8'd0) begin
                        // Extract has_match and suffix from state_idx
                        // state_idx = {has_match, suffix} where suffix is pattern_len-1 bits
                        wire has_match = state_idx[MAX_M-1];
                        wire [MAX_M-2:0] suffix = state_idx[MAX_M-2:0];
                        
                        // For bit = 0 and 1
                        for (integer bit_val = 0; bit_val < 2; bit_val = bit_val + 1) begin
                            // Compute new suffix
                            wire [MAX_M-2:0] new_suffix = ((suffix << 1) | bit_val) & ((1 << (pattern_len-1)) - 1);
                            
                            // Check for match
                            wire match;
                            if (current_pos >= pattern_len - 1) begin
                                // We have enough bits to check a full substring
                                wire [MAX_M-1:0] full_substring = (suffix << 1) | bit_val;
                                match = ((full_substring & mask_eff) == mask_eff);
                            end else begin
                                match = 1'b0;
                            end
                            
                            wire new_has_match = has_match | match;
                            
                            // Compute new state index
                            wire [MAX_M-1:0] new_state_idx = {new_has_match, new_suffix};
                            
                            // Add count to next_dp_vector[new_state_idx]
                            // This needs to be done in a separate always block or using temporary storage
                            // For simplicity, we'll use a separate combinational block
                        end
                    end
                end
                
                current_pos <= current_pos + 1'b1;
            end else begin
                // Finished DP: compute result
                // Sum all states where has_match = 1
                result_reg <= 16'd0;
                for (integer state_idx = 0; state_idx < (2*2**(MAX_M-1)); state_idx = state_idx + 1) begin
                    if (state_idx[MAX_M-1] == 1'b1) begin // has_match bit
                        result_reg <= result_reg + dp_vector[state_idx];
                    end
                end
                result <= result_reg;
                done <= 1'b1;
                computing <= 1'b0;
            end
        end
    end
end

// Helper function to count set bits (popcount)
function integer popcount(input [MAX_M-1:0] val);
    integer i;
    popcount = 0;
    for (i = 0; i < MAX_M; i = i + 1) begin
        if (val[i]) popcount = popcount + 1;
    end
endfunction

endmodule
