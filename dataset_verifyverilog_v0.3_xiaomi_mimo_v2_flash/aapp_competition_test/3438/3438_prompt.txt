module optimal_cache (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] c,
    input wire [4:0] a,
    input wire [4:0] obj_addr,
    input wire [3:0] obj_data,
    input wire obj_write,
    output reg [7:0] miss_count,
    output reg done
);
    // Parameters
    parameter MAX_CACHE = 4;
    parameter MAX_SEQ = 16;

    // State encoding
    localparam IDLE = 0,
               INIT = 1,
               CHECK = 2,
               MISS = 3,
               ADD = 4,
               EVICT = 5,
               NEXT = 6,
               DONE = 7;

    // Sequence memory
    reg [3:0] seq [0:MAX_SEQ-1];
    // Cache
    reg [3:0] cache [0:MAX_CACHE-1];
    reg valid [0:MAX_CACHE-1];
    // Registers
    reg [4:0] current_index;
    reg [2:0] occupied;
    reg [3:0] state, next_state;

    // Combinational signals
    wire hit;
    wire [2:0] hit_slot;
    wire has_empty;
    wire [2:0] empty_slot;
    wire [4:0] next_use [0:MAX_CACHE-1];
    wire [2:0] evict_slot;
    wire [4:0] max_next_use;

    integer i, j;

    // Write sequence memory
    always @(posedge clk) begin
        if (obj_write) begin
            seq[obj_addr] <= obj_data;
        end
    end

    // Hit detection
    always @(*) begin
        hit = 0;
        hit_slot = 0;
        for (i = 0; i < MAX_CACHE; i = i + 1) begin
            if (i < c && valid[i] && cache[i] == seq[current_index]) begin
                hit = 1;
                hit_slot = i;
            end
        end
    end

    // Empty slot detection
    always @(*) begin
        has_empty = 0;
        empty_slot = 0;
        for (i = 0; i < MAX_CACHE; i = i + 1) begin
            if (i < c && !valid[i] && !has_empty) begin
                has_empty = 1;
                empty_slot = i;
            end
        end
    end

    // Next use computation
    always @(*) begin
        for (i = 0; i < MAX_CACHE; i = i + 1) begin
            next_use[i] = a;
            if (i < c && valid[i]) begin
                for (j = current_index + 1; j < MAX_SEQ; j = j + 1) begin
                    if (j < a && seq[j] == cache[i] && next_use[i] == a) begin
                        next_use[i] = j;
                    end
                end
            end
        end
    end

    // Evict slot selection
    always @(*) begin
        max_next_use = 0;
        evict_slot = 0;
        for (i = 0; i < MAX_CACHE; i = i + 1) begin
            if (i < c && valid[i] && next_use[i] > max_next_use) begin
                max_next_use = next_use[i];
                evict_slot = i;
            end
        end
    end

    // State register and sequential logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            current_index <= 0;
            occupied <= 0;
            miss_count <= 0;
            done <= 0;
            for (i = 0; i < MAX_CACHE; i = i + 1) begin
                cache[i] <= 0;
                valid[i] <= 0;
            end
        end else begin
            state <= next_state;

            case (state)
                INIT: begin
                    occupied <= 0;
                    miss_count <= 0;
                    current_index <= 0;
                    for (i = 0; i < MAX_CACHE; i = i + 1) begin
                        valid[i] <= 0;
                        cache[i] <= 0;
                    end
                end

                ADD: begin
                    cache[empty_slot] <= seq[current_index];
                    valid[empty_slot] <= 1;
                    occupied <= occupied + 1;
                end

                EVICT: begin
                    cache[evict_slot] <= seq[current_index];
                    valid[evict_slot] <= 1;
                end

                NEXT: begin
                    current_index <= current_index + 1;
                end

                DONE: begin
                    done <= 1;
                end
            endcase
        end
    end

    // Next state logic
    always @(*) begin
        next_state = state; // default stay

        case (state)
            IDLE: begin
                if (start) begin
                    next_state = INIT;
                end
            end

            INIT: begin
                next_state = CHECK;
            end

            CHECK: begin
                if (hit) begin
                    next_state = NEXT;
                end else begin
                    next_state = MISS;
                end
            end

            MISS: begin
                if (has_empty) begin
                    next_state = ADD;
                end else begin
                    next_state = EVICT;
                end
            end

            ADD: begin
                next_state = NEXT;
            end

            EVICT: begin
                next_state = NEXT;
            end

            NEXT: begin
                if (current_index + 1 < a) begin
                    next_state = CHECK;
                end else begin
                    next_state = DONE;
                end
            end

            DONE: begin
                next_state = DONE;
            end

            default: next_state = IDLE;
        endcase
    end

    // Increment miss_count on ADD or EVICT
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            miss_count <= 0;
        end else begin
            if (state == ADD || state == EVICT) begin
                miss_count <= miss_count + 1;
            end
        end
    end

endmodule
