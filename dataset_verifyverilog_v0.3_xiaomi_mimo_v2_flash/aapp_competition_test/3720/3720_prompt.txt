module powers_game(
    input clk,
    input rst_n,
    input start,
    input [9:0] n,
    output reg winner,
    output reg done
);
    // State definitions
    localparam IDLE = 3'b000;
    localparam INIT = 3'b001;
    localparam LOOP = 3'b010;
    localparam CALC_L = 3'b011;
    localparam UPDATE = 3'b100;
    localparam FINAL = 3'b101;

    // Internal registers
    reg [2:0] state;
    reg [9:0] i;              // Current base
    reg [9:0] s;              // Sum of chain lengths
    reg [3:0] ans;            // Current XOR value
    reg [3:0] L;              // Current chain length
    reg [31:0] current_power; // For exponentiation
    reg [3:0] exponent;
    reg [9:0] sqrt_n;         // Precomputed sqrt(n)

    // Lookup table for arr[0:15] (Grundy numbers)
    reg [3:0] arr [0:15];
    initial begin
        arr[0] = 0;
        arr[1] = 1;
        arr[2] = 2;
        arr[3] = 1;
        arr[4] = 4;
        arr[5] = 3;
        arr[6] = 2;
        arr[7] = 1;
        arr[8] = 5;
        arr[9] = 6;
        arr[10] = 2;
        arr[11] = 1;
        arr[12] = 8;
        arr[13] = 7;
        arr[14] = 5;
        arr[15] = 9;
    end

    // Perfect power flags for i=3 to 31
    reg is_perfect [3:31];
    initial begin
        // Initialize all to 0
        for (int idx = 3; idx <= 31; idx++) is_perfect[idx] = 0;
        // Mark perfect powers
        is_perfect[4] = 1;  // 2^2
        is_perfect[8] = 1;  // 2^3
        is_perfect[9] = 1;  // 3^2
        is_perfect[16] = 1; // 2^4
        is_perfect[25] = 1; // 5^2
        is_perfect[27] = 1; // 3^3
        is_perfect[32] = 1; // 2^5 (but i max=31)
        is_perfect[36] = 1; // 6^2
        is_perfect[49] = 1; // 7^2
        is_perfect[64] = 1; // 8^2, 4^3, 2^6
        is_perfect[81] = 1; // 9^2, 3^4
        is_perfect[100] = 1; // 10^2
    end

    // Helper: Find floor of square root
    function [9:0] sqrt;
        input [9:0] x;
        integer i;
        begin
            sqrt = 0;
            for (i = 15; i >= 0; i = i - 1) begin
                if ((sqrt + (1 << i)) * (sqrt + (1 << i)) <= x)
                    sqrt = sqrt + (1 << i);
            end
        end
    endfunction

    // State machine
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            done <= 0;
            winner <= 0;
        end else begin
            case (state)
                IDLE: begin
                    done <= 0;
                    if (start) begin
                        state <= INIT;
                    end
                end

                INIT: begin
                    // Compute s0 = floor(log2(n))
                    if (n < 2) begin
                        ans <= arr[0];
                        s <= 0;
                    end else begin
                        // Priority encoder for highest bit
                        if (n[9]) ans <= arr[9];
                        else if (n[8]) ans <= arr[8];
                        else if (n[7]) ans <= arr[7];
                        else if (n[6]) ans <= arr[6];
                        else if (n[5]) ans <= arr[5];
                        else if (n[4]) ans <= arr[4];
                        else if (n[3]) ans <= arr[3];
                        else if (n[2]) ans <= arr[2];
                        else if (n[1]) ans <= arr[1];
                        else ans <= arr[0];
                        s <= (n[9]?9:(n[8]?8:(n[7]?7:(n[6]?6:(n[5]?5:(n[4]?4:(n[3]?3:(n[2]?2:(n[1]?1:0))))))));
                    end
                    i <= 3;
                    sqrt_n <= sqrt(n);
                    state <= LOOP;
                end

                LOOP: begin
                    if (i > sqrt_n) begin
                        state <= FINAL;
                    end else if (i <= 31 && is_perfect[i]) begin
                        i <= i + 1;
                    end else begin
                        L <= 0;
                        current_power <= 1;
                        exponent <= 0;
                        state <= CALC_L;
                    end
                end

                CALC_L: begin
                    // Compute current_power = i^(exponent+1)
                    if (exponent == 0) begin
                        current_power <= i;
                    end else begin
                        current_power <= current_power * i;
                    end
                    // Check after one cycle delay
                    if (exponent > 0 && current_power > n) begin
                        // If current_power (i^exponent) > n, then L is exponent-1
                        L <= exponent - 1;
                        state <= UPDATE;
                    end else if (exponent > 0 && current_power <= n) begin
                        L <= exponent;
                    end
                    exponent <= exponent + 1;
                    // Terminate if exponent too large or power exceeds n
                    if (exponent >= 10 || (exponent > 0 && current_power > n)) begin
                        state <= UPDATE;
                    end
                end

                UPDATE: begin
                    if (L >= 1 && L <= 15) begin
                        ans <= ans ^ arr[L];
                        s <= s + L;
                    end
                    i <= i + 1;
                    state <= LOOP;
                end

                FINAL: begin
                    // ans = ans XOR ((n - s) % 2)
                    if ((n - s) & 1) ans <= ans ^ 1;
                    // winner = 1 if ans != 0
                    winner <= (ans != 0);
                    done <= 1;
                    state <= IDLE;
                end

                default: state <= IDLE;
            endcase
        end
    end
endmodule
