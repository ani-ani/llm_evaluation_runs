module robber_password_counter (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [7:0] arr_0, arr_1, arr_2, arr_3, arr_4, arr_5, arr_6, arr_7,
    input wire [7:0] arr_8, arr_9, arr_10, arr_11, arr_12, arr_13, arr_14, arr_15,
    input wire [3:0] length,
    output reg [19:0] result,
    output reg done
);

// Maximum length and modulo
parameter MAX_LEN = 16;
parameter MOD = 20'd1000009;

// State definitions
localparam IDLE = 2'b00;
localparam COMPUTE = 2'b01;
localparam DONE = 2'b10;

// Registers
reg [1:0] state;
reg [3:0] i_reg;           // current index (1 to length)
reg [19:0] dp_im1, dp_im2, dp_im3;
reg [7:0] prev_char, prev_prev_char;
reg [7:0] char_array [0:MAX_LEN-1];
reg [3:0] len_reg;

// Combinational next values
reg [19:0] next_dp_im1, next_dp_im2, next_dp_im3;
reg [7:0] next_prev_char, next_prev_prev_char;
reg [3:0] next_i;
reg next_done;

// Vowel detection
function automatic is_vowel(input [7:0] c);
    begin
        is_vowel = (c == 8'h61) || (c == 8'h65) || (c == 8'h69) || (c == 8'h6F) || (c == 8'h75);
    end
endfunction

// Combinational logic for next state and outputs
always @(*) begin
    // Default: keep current values
    next_dp_im1 = dp_im1;
    next_dp_im2 = dp_im2;
    next_dp_im3 = dp_im3;
    next_prev_char = prev_char;
    next_prev_prev_char = prev_prev_char;
    next_i = i_reg;
    next_done = 0;

    case (state)
        COMPUTE: begin
            if (i_reg > len_reg) begin
                // All characters processed, go to DONE
                next_done = 1;
            end else begin
                // Compute dp_i
                reg [7:0] cur_char = char_array[i_reg-1];
                reg vowel = is_vowel(cur_char);
                reg [19:0] dp_i = dp_im1; // vowel or consonant untransformed
                if (!vowel && i_reg >= 3) begin
                    // Check transformed case: must have previous char 'o' and char before that equal to current
                    if (prev_char == 8'h6F && prev_prev_char == cur_char) begin
                        dp_i = dp_i + dp_im3;
                        if (dp_i >= MOD) dp_i = dp_i - MOD;
                    end
                end
                // Update next registers
                next_dp_im3 = dp_im2;
                next_dp_im2 = dp_im1;
                next_dp_im1 = dp_i;
                next_prev_prev_char = prev_char;
                next_prev_char = cur_char;
                next_i = i_reg + 1;
            end
        end
        default: begin
            // For IDLE and DONE, next values remain default
        end
    endcase
end

// Sequential updates
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 20'd0;
        i_reg <= 4'd0;
        dp_im1 <= 20'd0;
        dp_im2 <= 20'd0;
        dp_im3 <= 20'd0;
        prev_char <= 8'd0;
        prev_prev_char <= 8'd0;
        len_reg <= 4'd0;
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    // Load input array
                    char_array[0] <= arr_0;
                    char_array[1] <= arr_1;
                    char_array[2] <= arr_2;
                    char_array[3] <= arr_3;
                    char_array[4] <= arr_4;
                    char_array[5] <= arr_5;
                    char_array[6] <= arr_6;
                    char_array[7] <= arr_7;
                    char_array[8] <= arr_8;
                    char_array[9] <= arr_9;
                    char_array[10] <= arr_10;
                    char_array[11] <= arr_11;
                    char_array[12] <= arr_12;
                    char_array[13] <= arr_13;
                    char_array[14] <= arr_14;
                    char_array[15] <= arr_15;
                    len_reg <= length;
                    // Initialize DP
                    dp_im1 <= 20'd1; // dp[0]
                    dp_im2 <= 20'd0;
                    dp_im3 <= 20'd0;
                    prev_char <= 8'd0;
                    prev_prev_char <= 8'd0;
                    i_reg <= 4'd1;   // start with i=1
                    state <= COMPUTE;
                end
            end
            COMPUTE: begin
                // Update registers with next values
                dp_im1 <= next_dp_im1;
                dp_im2 <= next_dp_im2;
                dp_im3 <= next_dp_im3;
                prev_char <= next_prev_char;
                prev_prev_char <= next_prev_prev_char;
                i_reg <= next_i;
                if (next_done) begin
                    state <= DONE;
                    result <= next_dp_im1; // dp_im1 now holds dp[length]
                    done <= 1;
                end
            end
            DONE: begin
                state <= IDLE;
                done <= 0;
            end
            default: state <= IDLE;
        endcase
    end
end

endmodule
