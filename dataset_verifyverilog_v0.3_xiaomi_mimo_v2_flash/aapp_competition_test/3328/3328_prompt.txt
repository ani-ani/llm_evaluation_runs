Module DominoMinSum:
- Inputs: clk, rst_n, start, grid[4][4] (8-bit each), K (4-bit)
- Outputs: min_sum (16-bit), done
- Parameters: N=4, K_MAX=8
- Description: Implements DP over rows with state machine.

```verilog
module DominoMinSum (
    input clk,
    input rst_n,
    input start,
    input [7:0] grid [0:3][0:3],
    input [3:0] K,
    output reg [15:0] min_sum,
    output reg done
);

// State encoding
localparam IDLE = 2'b00;
localparam RESET = 2'b01;
localparam PROCESS_ROW = 2'b10;
localparam DONE = 2'b11;

reg [1:0] state, next_state;
reg [3:0] row_counter; // current row being processed
reg [3:0] mask; // occupancy mask from previous row
reg [3:0] k_count; // dominoes placed so far
reg [15:0] dp [0:15][0:8]; // dp[mask][k] = max covered sum, initialized to -inf
reg [15:0] dp_next [0:15][0:8];
reg [15:0] total_sum;
reg [15:0] best_covered;

// Precomputed placement patterns for each mask (16 masks, 128 subsets)
// Each entry: valid, new_mask, cur_cover_mask, num_dominoes
reg [0:0]  pat_valid [0:4095];
reg [3:0]  pat_new_mask [0:4095];
reg [3:0]  pat_cur_cover [0:4095];
reg [2:0]  pat_num_dom [0:4095];

// Helper function: compute total sum of grid
function [15:0] compute_total_sum;
    integer i, j;
    begin
        compute_total_sum = 0;
        for (i = 0; i < 4; i = i + 1)
            for (j = 0; j < 4; j = j + 1)
                compute_total_sum = compute_total_sum + grid[i][j];
    end
endfunction

// Task: process one row, compute dp_next from dp
// This is a placeholder; actual implementation would use loops and precomputed patterns
task process_row(input [3:0] r);
    integer m, k, s;
    reg [3:0] new_m, cur_cov;
    reg [2:0] num_dom;
    reg [15:0] added_weight;
    reg [15:0] cur_val, next_val;
    begin
        // Initialize dp_next to -inf
        for (m = 0; m < 16; m = m + 1)
            for (k = 0; k <= K; k = k + 1)
                dp_next[m][k] = 16'hFFFF;
        
        // Iterate over all masks and k
        for (m = 0; m < 16; m = m + 1) begin
            for (k = 0; k <= K; k = k + 1) begin
                if (dp[m][k] != 16'hFFFF) begin
                    // Iterate over all placement subsets (0..127)
                    for (s = 0; s < 128; s = s + 1) begin
                        // Index into precomputed patterns: m*128 + s
                        if (pat_valid[m*128 + s]) begin
                            new_m = pat_new_mask[m*128 + s];
                            cur_cov = pat_cur_cover[m*128 + s];
                            num_dom = pat_num_dom[m*128 + s];
                            if (k + num_dom <= K) begin
                                // Compute added_weight: sum of covered cells in current row and next row
                                added_weight = 0;
                                // Current row: add grid[r][c] if cur_cov[c] == 1
                                for (integer c = 0; c < 4; c = c + 1) begin
                                    if (cur_cov[c]) begin
                                        added_weight = added_weight + grid[r][c];
                                    end
                                end
                                // Next row: add grid[r+1][c] if new_m[c] == 1 (only if r < 3)
                                if (r < 3) begin
                                    for (integer c = 0; c < 4; c = c + 1) begin
                                        if (new_m[c]) begin
                                            added_weight = added_weight + grid[r+1][c];
                                        end
                                    end
                                end
                                // Update dp_next
                                if (dp[m][k] + added_weight > dp_next[new_m][k+num_dom]) begin
                                    dp_next[new_m][k+num_dom] = dp[m][k] + added_weight;
                                end
                            end
                        end
                    end
                end
            end
        end
        
        // Copy dp_next to dp for next row
        for (m = 0; m < 16; m = m + 1)
            for (k = 0; k <= K; k = k + 1)
                dp[m][k] <= dp_next[m][k];
    end
endtask

// State machine
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        min_sum <= 0;
    end else begin
        case (state)
            IDLE: if (start) begin
                // Initialize DP
                for (integer m = 0; m < 16; m = m + 1) begin
                    for (integer k = 0; k <= 8; k = k + 1) begin
                        dp[m][k] <= 16'hFFFF;
                    end
                end
                dp[0][0] <= 0;
                row_counter <= 0;
                best_covered <= 0;
                state <= RESET;
            end

            RESET: begin
                total_sum <= compute_total_sum();
                state <= PROCESS_ROW;
            end

            PROCESS_ROW: begin
                if (row_counter < 4) begin
                    process_row(row_counter);
                    row_counter <= row_counter + 1;
                end else begin
                    // Find best covered sum for exactly K dominoes
                    for (integer m = 0; m < 16; m = m + 1) begin
                        if (dp[m][K] != 16'hFFFF && dp[m][K] > best_covered)
                            best_covered <= dp[m][K];
                    end
                    min_sum <= total_sum - best_covered;
                    done <= 1;
                    state <= DONE;
                end
            end

            DONE: begin
                // Hold output
            end
        endcase
    end
end

endmodule
```
