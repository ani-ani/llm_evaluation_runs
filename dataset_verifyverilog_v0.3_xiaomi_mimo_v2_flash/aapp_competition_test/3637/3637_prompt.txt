module pizza_solver #(
    parameter NUM_FRIENDS = 4,
    parameter WISHES_PER_FRIEND = 8,
    parameter WISH_WIDTH = 5
)(
    input clk,
    input rst_n,
    input start,
    input [2:0] num_friends,   // actual number of friends (1-4)
    input [2:0] num_toppings,  // actual number of toppings (1-8)
    // Packed wishes data: each wish is 5 bits: [2:0] topping_index, sign, valid
    input [NUM_FRIENDS*WISHES_PER_FRIEND*WISH_WIDTH - 1:0] wishes_packed,
    // Packed wishes count per friend: each count is 3 bits (1-8)
    input [NUM_FRIENDS*3 - 1:0] wishes_count_packed,
    output reg [7:0] subset,
    output reg done,
    output reg valid
);

// Implementation details:
// - Use a state machine to brute-force all subsets of toppings (up to 256 subsets).
// - Unpack the wishes_packed and wishes_count_packed into arrays for easy access.
// - For each subset, check all friends' wishes: count satisfied wishes and compare to threshold.
// - Threshold: satisfied * 3 > total_wishes for each friend.
// - When a valid subset is found, output it and assert done and valid.
// - If no subset is found, assert done but valid=0 (should not happen per problem guarantee).

endmodule
