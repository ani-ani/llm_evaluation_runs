module basic_interpreter(
    input clk,
    input rst_n,
    input start,
    input [7:0] prog_len,           // Number of statements in program
    input [127:0] instruction,      // Current instruction input (from external memory)
    input [7:0] pc_in,              // Current program counter input
    output reg [7:0] pc_out,        // Next program counter output
    output reg read_en,             // Enable reading from instruction memory
    output reg [7:0] char_addr,     // Address for string character lookup
    input [7:0] char_data,          // Character data from string memory
    output reg char_valid,          // When to output character
    output reg [7:0] char_out,      // Output character for PRINT
    output reg done                 // Program execution complete
);

// Fixed-point format: 16-bit signed (sufficient for test cases)
// Variables A-Z map to registers 0-25

// Instruction format (128 bits = 16 bytes):
// [7:0]    - label
// [15:8]   - command type (0=LET, 1=IF, 2=PRINT, 3=PRINTLN)
// [23:16]  - operand A (variable index or immediate)
// [31:24]  - operand B (variable index or immediate)
// [39:32]  - operator (+, -, *, /, =, <, >, <=, >=, <>)
// [47:40]  - target variable (for LET)
// [55:48]  - goto label (for IF)
// [63:56]  - string length (for PRINT/PRINTLN)
// [127:64] - string data (up to 8 chars, padded)

reg [15:0] vars [0:25];  // Variables A-Z
reg [7:0] state;
reg [7:0] current_label;
reg [7:0] next_pc;
reg [15:0] op_result;
reg [15:0] temp_a;
reg [15:0] temp_b;
reg condition_met;

// String output state
reg [3:0] char_idx;
reg [3:0] string_len;
reg output_newline;

localparam IDLE = 0;
localparam FETCH = 1;
localparam DECODE = 2;
localparam EXEC_LET = 3;
localparam EXEC_IF = 4;
localparam EXEC_PRINT = 5;
localparam OUTPUT_STRING = 6;
localparam OUTPUT_NEWLINE = 7;
localparam UPDATE_PC = 8;
localparam DONE_STATE = 9;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        read_en <= 0;
        char_valid <= 0;
        pc_out <= 0;
        // Initialize variables
        for (integer i = 0; i < 26; i = i + 1) begin
            vars[i] <= 0;
        end
    end else begin
        char_valid <= 0;
        read_en <= 0;
        
        case (state)
            IDLE: begin
                if (start) begin
                    pc_out <= 0;
                    state <= FETCH;
                    done <= 0;
                end
            end
            
            FETCH: begin
                // Request instruction from memory
                read_en <= 1;
                state <= DECODE;
            end
            
            DECODE: begin
                // Parse instruction fields
                current_label <= instruction[7:0];
                temp_a <= {8'h00, instruction[23:16]}; // Zero-extend for now
                temp_b <= {8'h00, instruction[31:24]};
                string_len <= instruction[55:48];
                output_newline <= (instruction[15:8] == 3); // PRINTLN
                
                // Resolve variables
                if (instruction[15:8] == 0) begin // LET
                    // Check if operands are variables (bit 7 = 1 indicates variable)
                    temp_a <= instruction[23] ? vars[instruction[22:16]] : {8'h00, instruction[23:16]};
                    temp_b <= instruction[31] ? vars[instruction[30:24]] : {8'h00, instruction[31:24]};
                end else if (instruction[15:8] == 1) begin // IF
                    temp_a <= instruction[23] ? vars[instruction[22:16]] : {8'h00, instruction[23:16]};
                    temp_b <= instruction[31] ? vars[instruction[30:24]] : {8'h00, instruction[31:24]};
                end
                
                // Determine next state based on command
                case (instruction[15:8])
                    0: state <= EXEC_LET;
                    1: state <= EXEC_IF;
                    2, 3: begin
                        char_idx <= 0;
                        state <= EXEC_PRINT;
                    end
                    default: state <= UPDATE_PC;
                endcase
            end
            
            EXEC_LET: begin
                // Perform arithmetic based on operator [39:32]
                case (instruction[39:32])
                    8'h01: op_result = temp_a + temp_b;  // +
                    8'h02: op_result = temp_a - temp_b;  // -
                    8'h03: op_result = temp_a * temp_b;  // *
                    8'h04: op_result = temp_a / temp_b;  // / (truncates toward zero)
                    default: op_result = temp_a;
                endcase
                // Write to variable [47:40]
                vars[instruction[47:40]] <= op_result;
                state <= UPDATE_PC;
            end
            
            EXEC_IF: begin
                // Evaluate condition [39:32]
                case (instruction[39:32])
                    8'h05: condition_met = (temp_a == temp_b);  // =
                    8'h06: condition_met = (temp_a > temp_b);   // >
                    8'h07: condition_met = (temp_a < temp_b);   // <
                    8'h08: condition_met = (temp_a != temp_b);  // <>
                    8'h09: condition_met = (temp_a <= temp_b);  // <=
                    8'h0A: condition_met = (temp_a >= temp_b);  // >=
                    default: condition_met = 0;
                endcase
                
                if (condition_met) begin
                    // Jump to label [55:48]
                    // Since labels are sequential, we assume label = position + 1
                    // For simplicity, we'll just load the new PC from instruction
                    pc_out <= instruction[55:48];
                end else begin
                    pc_out <= pc_in + 1;
                end
                state <= UPDATE_PC;
            end
            
            EXEC_PRINT: begin
                if (string_len > 0 && char_idx < string_len) begin
                    // Output string character from instruction[127:64]
                    char_out <= instruction[64 + char_idx*8 +: 8];
                    char_valid <= 1;
                    char_idx <= char_idx + 1;
                    state <= OUTPUT_STRING;
                end else if (instruction[15:8] == 2) begin // PRINT without variable
                    state <= UPDATE_PC;
                end else begin
                    // Print variable from [23:16]
                    char_out <= 8'h30 + vars[instruction[23:16]][7:0]; // Simple digit for test
                    char_valid <= 1;
                    state <= (instruction[15:8] == 3) ? OUTPUT_NEWLINE : UPDATE_PC;
                end
            end
            
            OUTPUT_STRING: begin
                if (char_idx < string_len) begin
                    char_out <= instruction[64 + char_idx*8 +: 8];
                    char_valid <= 1;
                    char_idx <= char_idx + 1;
                end else begin
                    char_idx <= 0;
                    if (output_newline) begin
                        state <= OUTPUT_NEWLINE;
                    end else begin
                        state <= UPDATE_PC;
                    end
                end
            end
            
            OUTPUT_NEWLINE: begin
                char_out <= 8'h0A;  // Newline
                char_valid <= 1;
                state <= UPDATE_PC;
            end
            
            UPDATE_PC: begin
                if (instruction[15:8] != 1 || !condition_met) begin
                    // If not a taken IF, increment PC
                    if (instruction[15:8] != 1)
                        pc_out <= pc_in + 1;
                end
                
                // Check if done
                if (pc_in >= prog_len - 1 && (instruction[15:8] != 1 || !condition_met)) begin
                    state <= DONE_STATE;
                end else begin
                    state <= FETCH;
                end
            end
            
            DONE_STATE: begin
                done <= 1;
                if (!start) state <= IDLE;
            end
        endcase
    end
end

endmodule

// Helper module to convert Python-style BASIC to instruction memory
module program_loader (
    input [7:0] prog_index,
    output reg [127:0] instruction,
    output reg [7:0] prog_len
);

always @(*) begin
    case (prog_index)
        // Program 1: Simple loop
        0: begin instruction = {8'h00, 8'h00, 8'h00, 8'h01, 8'h01, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00, 8'h00}; prog_len = 6; end
        // This is a simplified loader - full implementation would parse real BASIC
        // For actual test, instructions need to be properly encoded
        default: begin instruction = 128'h0; prog_len = 0; end
    endcase
end

endmodule
