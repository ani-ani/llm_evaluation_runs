module graph_decoration_optimizer (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [2:0] node_count,          // Number of nodes (1-8)
    input wire [3:0] edge_count,          // Number of edges (0-16)
    input wire [2:0] edge_u [0:15],       // Source nodes for edges
    input wire [2:0] edge_v [0:15],       // Destination nodes for edges
    output reg [15:0] min_cost,           // Minimum cost result
    output reg valid,                     // Valid result
    output reg error                      // No solution found
);

    // Edge costs: 0, 1, or 2 (encoded as 2-bit values: 00=0, 01=1, 10=2)
    reg [1:0] current_costs [0:15];
    reg [4:0] best_cost;                  // Track best found cost
    reg [4:0] current_cost_sum;
    reg [15:0] candidates_tried;          // Bitmask for tried configurations
    
    // State machine states
    localparam IDLE = 3'b000;
    localparam VALIDATE_CONSTRAINTS = 3'b001;
    localparam CHECK_CYCLES = 3'b010;
    localparam UPDATE_BEST = 3'b011;
    localparam NEXT_CONFIG = 3'b100;
    localparam DONE = 3'b101;
    
    reg [2:0] state;
    reg [2:0] next_state;
    reg [3:0] edge_idx;                   // Current edge being considered
    reg [3:0] config_idx;                 // Current configuration index
    reg [2:0] node_degree [0:7];          // Track degrees per node
    
    // Constraint checking variables
    reg [2:0] check_node;
    reg [3:0] check_edge_i;
    reg [3:0] check_edge_j;
    reg [1:0] cost_a, cost_b;
    reg constraint_violated;
    
    // Cycle detection variables
    reg [7:0] visited_nodes;
    reg [3:0] cycle_edge_count;
    reg cycle_has_odd_length;
    reg [3:0] cycle_edge_idx;
    
    // Cycle structure: pre-defined small cycles for 8 nodes
    // We'll check simple cycles using edge adjacency
    reg [15:0] cycle_edge_mask;
    reg [2:0] cycle_nodes [0:7];
    reg [2:0] cycle_node_count;
    
    integer i, j, k;
    
    // Helper: Calculate mod 3 sum (a+b) % 3 != 1
    function automatic bool check_mod_constraint(input [1:0] a, input [1:0] b);
        reg [1:0] sum;
        begin
            sum = a + b;
            if (sum >= 3) sum = sum - 3;
            // Check if (sum % 3) != 1
            check_mod_constraint = (sum != 1);
        end
    endfunction
    
    // Helper: Calculate sum of costs
    function automatic [4:0] calculate_sum(input [15:0] mask);
        reg [4:0] sum;
        begin
            sum = 0;
            for (i = 0; i < 16; i = i + 1) begin
                if (mask[i]) begin
                    case (current_costs[i])
                        2'b00: sum = sum + 0;
                        2'b01: sum = sum + 1;
                        2'b10: sum = sum + 2;
                        default: sum = sum + 0;
                    endcase
                end
            end
            calculate_sum = sum;
        end
    endfunction
    
    // Sequential logic
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            valid <= 0;
            error <= 0;
            min_cost <= 16'hFFFF;
            best_cost <= 5'h1F;
            config_idx <= 0;
            candidates_tried <= 0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        // Initialize costs to all zeros
                        for (i = 0; i < 16; i = i + 1) begin
                            current_costs[i] <= 2'b00;
                        end
                        best_cost <= 5'h1F;
                        config_idx <= 0;
                        candidates_tried <= 0;
                        valid <= 0;
                        error <= 0;
                        state <= (edge_count > 0) ? NEXT_CONFIG : DONE;
                    end
                end
                
                NEXT_CONFIG: begin
                    // Generate next configuration
                    if (config_idx >= (3 ** edge_count) || candidates_tried[config_idx]) begin
                        // All configurations tried
                        if (best_cost == 5'h1F) begin
                            error <= 1;
                        end else begin
                            min_cost <= best_cost;
                            valid <= 1;
                        end
                        state <= DONE;
                    end else begin
                        // Decode config_idx into current_costs
                        decode_configuration(config_idx, current_costs, edge_count);
                        candidates_tried[config_idx] <= 1;
                        state <= VALIDATE_CONSTRAINTS;
                        check_node <= 0;
                        edge_idx <= 0;
                        current_cost_sum <= 0;
                    end
                end
                
                VALIDATE_CONSTRAINTS: begin
                    // Check mod-3 constraint for all nodes
                    if (check_node < node_count) begin
                        // Check all edges connected to this node
                        constraint_violated <= 0;
                        check_edge_i <= 0;
                        check_edge_j <= 1;
                        state <= 2'b10; // Go to inner loop
                    end else begin
                        // All nodes checked, now check cycles
                        state <= CHECK_CYCLES;
                        cycle_edge_idx <= 0;
                        cycle_has_odd_length <= 0;
                    end
                end
                
                CHECK_CYCLES: begin
                    // Check for simple cycles and sum parity
                    // Simplified: check only direct edge cycles and 3-node cycles
                    if (cycle_edge_idx < edge_count) begin
                        // Find if this edge is part of any cycle
                        // For 3-node cycle: edge(i,j), edge(j,k), edge(k,i)
                        // For this simplified version, just check total parity of all edges
                        cycle_edge_idx <= cycle_edge_idx + 1;
                    end else begin
                        // After checking all possible cycles
                        current_cost_sum <= calculate_sum(candidates_tried);
                        state <= UPDATE_BEST;
                    end
                end
                
                UPDATE_BEST: begin
                    // Check if configuration is valid (no constraint violation flag)
                    if (!constraint_violated) begin
                        // Check if total cost parity is odd
                        if (current_cost_sum[0] == 1) begin // odd
                            if (current_cost_sum < best_cost) begin
                                best_cost <= current_cost_sum;
                            end
                        end
                    end
                    state <= NEXT_CONFIG;
                    config_idx <= config_idx + 1;
                end
                
                DONE: begin
                    // Stay in done state
                end
            endcase
        end
    end
    
    // Task to decode configuration
    task automatic decode_configuration(
        input [4:0] idx,
        output reg [1:0] costs [0:15],
        input [3:0] count
    );
        integer k;
        reg [4:0] temp;
        begin
            temp = idx;
            for (k = 0; k < count; k = k + 1) begin
                costs[k] = temp % 3;
                temp = temp / 3;
            end
            // Fill remaining with zeros
            for (k = count; k < 16; k = k + 1) begin
                costs[k] = 0;
            end
        end
    endtask

endmodule

    input rst_n,
    input start,
    input [2:0] node_count,
    input [3:0] edge_count,
    input [2:0] edge_u_0, input [2:0] edge_v_0,

    input [2:0] edge_u_3, input [2:0] edge_v_3,

    input [2:0] edge_u_5, input [2:0] edge_v_5,

    output valid,
    output error
);
    // Wrapper for edge arrays - use concatenation for 8 elements
    wire [2:0] edge_u [0:7];
    wire [2:0] edge_v [0:7];
    
    assign edge_u[0] = edge_u_0;
    assign edge_u[1] = edge_u_1;
    assign edge_u[2] = edge_u_2;
    assign edge_u[3] = edge_u_3;
    assign edge_u[4] = edge_u_4;
    assign edge_u[5] = edge_u_5;
    assign edge_u[6] = edge_u_6;
    assign edge_u[7] = edge_u_7;
    
    assign edge_v[0] = edge_v_0;
    assign edge_v[1] = edge_v_1;
    assign edge_v[2] = edge_v_2;
    assign edge_v[3] = edge_v_3;

    graph_decoration_optimizer u0 (

        .edge_u(edge_u),
        .edge_v(edge_v),
        .min_cost(min_cost),
        .valid(valid),
        .error(error)
    );
endmodule
