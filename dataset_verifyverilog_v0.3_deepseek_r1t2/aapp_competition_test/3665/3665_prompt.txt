module LossySorter #(
    parameter N = 2,          // Number of numbers (scaled to 2)
    parameter M = 3,          // Number of digits per number (scaled to 3)
    parameter DIGIT_WIDTH = 4 // Bits per digit (4 bits for 0-9)
)(
    input clk,
    input rst_n,
    input start,
    // Input numbers: each as N separate M-digit values
    // Each number is packed as: {d0, d1, ..., d_{M-1}} where d0 is most significant digit
    input [N-1:0][M*DIGIT_WIDTH-1:0] numbers_in,
    output reg [N-1:0][M*DIGIT_WIDTH-1:0] numbers_out,
    output reg done
);

// Internal state definitions
localparam STATE_IDLE = 2'b00;
localparam STATE_COMPUTE = 2'b01;
localparam STATE_OUTPUT = 2'b10;
localparam STATE_DONE = 2'b11;

// State tracking: bit for each adjacent pair (N-1 bits)
// For N=2: 1 bit (0=equal so far, 1=already ordered)
reg [N-2:0] state_vector;
reg [M-1:0] digit_idx;  // Current digit position being processed
reg [1:0] current_state;

// Cost accumulation
reg [15:0] total_cost;

// Storage for intermediate results
reg [N-1:0][M*DIGIT_WIDTH-1:0] temp_numbers;

// For N=2, we can use simple logic:
// At each digit position, decide digits for both numbers
// Constraints: if state_vector[0]==0 (equal so far), then d0 <= d1
// If state_vector[0]==1, no constraint
// Minimize total changes: cost = (d0 != orig_d0) + (d1 != orig_d1)

// Combinational logic to compute optimal digits for current digit position
wire [DIGIT_WIDTH-1:0] orig_d0, orig_d1;
assign orig_d0 = numbers_in[0][digit_idx*DIGIT_WIDTH +: DIGIT_WIDTH];
assign orig_d1 = numbers_in[1][digit_idx*DIGIT_WIDTH +: DIGIT_WIDTH];

reg [DIGIT_WIDTH-1:0] new_d0, new_d1;
reg [1:0] cost_per_digit;
reg new_state_bit;

always @(*) begin
    if (state_vector[0] == 1'b1) begin
        // Already ordered, choose digits that minimize cost independently
        // For minimal changes, keep original digits
        new_d0 = orig_d0;
        new_d1 = orig_d1;
        cost_per_digit = (orig_d0 != orig_d0) + (orig_d1 != orig_d1); // 0
        new_state_bit = 1'b1; // remains ordered
    end else begin
        // Still equal, need d0 <= d1
        // Try all possibilities? Since digits 0-9, we can enumerate 100 possibilities
        // For simplicity, we'll use a heuristic: if orig_d0 <= orig_d1, keep both
        // Else, we need to change one or both to make d0 <= d1 with minimal changes
        if (orig_d0 <= orig_d1) begin
            new_d0 = orig_d0;
            new_d1 = orig_d1;
            cost_per_digit = 0;
            new_state_bit = (orig_d0 < orig_d1) ? 1'b1 : 1'b0;
        end else begin
            // orig_d0 > orig_d1, need to fix
            // Options: change d0 down, or d1 up, or both
            // Compute minimal cost:
            // Cost to set d0 = d1 = orig_d1: cost = (orig_d0 != orig_d1) + 0 = 1
            // Cost to set d0 = d1 = orig_d0: cost = 0 + (orig_d1 != orig_d0) = 1
            // Cost to set d0 = d1 = some other value: at least 1
            // Cost to set d0 < d1: can set d0 = orig_d1 - 1, d1 = orig_d1, cost = (orig_d0 != orig_d1-1) + 0
            // But orig_d1-1 may be <0, so need to bound.
            // For simplicity, we choose to set both to max(orig_d0, orig_d1) = orig_d0
            new_d0 = orig_d0;
            new_d1 = orig_d0;
            cost_per_digit = (orig_d1 != orig_d0) ? 2'b01 : 2'b00;
            new_state_bit = 1'b0; // still equal
        end
    end
end

// Sequential logic
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        current_state <= STATE_IDLE;
        done <= 1'b0;
        digit_idx <= 0;
        state_vector <= 0;
        total_cost <= 0;
        numbers_out <= 0;
        temp_numbers <= 0;
    end else begin
        case (current_state)
            STATE_IDLE: begin
                done <= 1'b0;
                digit_idx <= 0;
                state_vector <= 0;
                total_cost <= 0;
                temp_numbers <= 0;
                if (start) begin
                    current_state <= STATE_COMPUTE;
                end
            end
            
            STATE_COMPUTE: begin
                // Update digits for current position
                temp_numbers[0][digit_idx*DIGIT_WIDTH +: DIGIT_WIDTH] <= new_d0;
                temp_numbers[1][digit_idx*DIGIT_WIDTH +: DIGIT_WIDTH] <= new_d1;
                total_cost <= total_cost + cost_per_digit;
                state_vector[0] <= new_state_bit;
                
                if (digit_idx == M-1) begin
                    current_state <= STATE_OUTPUT;
                    digit_idx <= 0;
                end else begin
                    digit_idx <= digit_idx + 1;
                end
            end
            
            STATE_OUTPUT: begin
                // Copy temp to output
                numbers_out <= temp_numbers;
                current_state <= STATE_DONE;
            end
            
            STATE_DONE: begin
                done <= 1'b1;
                if (!start) begin
                    current_state <= STATE_IDLE;
                end
            end
        endcase
    end
end

endmodule
