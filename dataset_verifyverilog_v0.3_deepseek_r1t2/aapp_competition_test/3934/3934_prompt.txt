module tree_checker (
    input clk,
    input rst_n,
    input start,
    input [3:0] n,            // number of nodes (2-16)
    input edge_valid,
    input [3:0] edge_u,
    input [3:0] edge_v,
    output reg result,        // 1 for YES (no deg2), 0 for NO (has deg2)
    output reg done
);

// State definitions
localparam IDLE = 2'b00;
localparam LOAD = 2'b01;
localparam CHECK = 2'b10;

reg [1:0] state;
reg [3:0] deg [0:15]; // degree counters for 16 nodes (0-indexed: node1->deg[0])
reg [3:0] edge_count;
wire has_deg2;

// Combinational logic for has_deg2
assign has_deg2 = (deg[0]==4'd2) | (deg[1]==4'd2) | (deg[2]==4'd2) | (deg[3]==4'd2) |
                  (deg[4]==4'd2) | (deg[5]==4'd2) | (deg[6]==4'd2) | (deg[7]==4'd2) |
                  (deg[8]==4'd2) | (deg[9]==4'd2) | (deg[10]==4'd2) | (deg[11]==4'd2) |
                  (deg[12]==4'd2) | (deg[13]==4'd2) | (deg[14]==4'd2) | (deg[15]==4'd2);

// State transition and sequential logic
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
        edge_count <= 4'b0;
        // Reset degree array
        for (integer i=0; i<16; i=i+1) begin
            deg[i] <= 4'b0;
        end
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    // Reset degree array and edge_count
                    for (integer i=0; i<16; i=i+1) begin
                        deg[i] <= 4'b0;
                    end
                    edge_count <= 4'b0;
                    state <= LOAD;
                end
            end

            LOAD: begin
                if (edge_valid) begin
                    // Update degrees for the edge (convert 1-indexed to 0-indexed)
                    deg[edge_u - 1] <= deg[edge_u - 1] + 1;
                    deg[edge_v - 1] <= deg[edge_v - 1] + 1;
                    edge_count <= edge_count + 1;

                    // Check if this is the last edge (n-1 edges total)
                    if (edge_count + 1 == n - 1) begin
                        state <= CHECK;
                    end
                end
            end

            CHECK: begin
                done <= 1;
                result <= ~has_deg2; // If no degree 2, result=1 (YES), else 0 (NO)
                state <= IDLE;
            end

            default: state <= IDLE;
        endcase
    end
end

endmodule
