module CriticOrderSolver #(
    parameter N = 8,          // Maximum number of critics (1-8)
    parameter M_WIDTH = 4,    // Bits for m (0-15)
    parameter K_WIDTH = 14,   // Bits for k (up to 8*15=120)
    parameter IDX_WIDTH = 3   // Bits for critic indices (0-7)
)(
    input wire clk,
    input wire rst_n,          // Active-low synchronous reset
    input wire start,          // Start pulse (1 cycle)
    input wire [3:0] n,        // Actual number of critics (1-8)
    input wire [M_WIDTH-1:0] m, // Max score
    input wire [K_WIDTH-1:0] k, // Target sum
    input wire [M_WIDTH-1:0] a [0:N-1], // Opinions array
    output reg done,           // Asserted when computation complete
    output reg possible,       // 1 if solution exists
    output reg [IDX_WIDTH-1:0] p [0:N-1], // Permutation (1-based indices)
    output reg [3:0] valid_indices // Number of valid indices in p
);

// Internal state machine
localparam IDLE = 3'b000;
localparam CHECK_COND = 3'b001;
localparam SORT = 3'b010;
localparam BUILD_PERM = 3'b011;
localparam DONE = 3'b100;

reg [2:0] state;
reg [2:0] next_state;

// Intermediate storage
reg [M_WIDTH-1:0] a_sorted [0:N-1];
reg [IDX_WIDTH-1:0] idx_sorted [0:N-1];
reg [3:0] x; // Number of givers = k/m
reg [3:0] i, j; // Loop counters
reg [K_WIDTH-1:0] temp_k;

// Combinational logic for k % m check
wire k_mod_m_zero;
assign k_mod_m_zero = (k % m == 0);

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
    end else begin
        state <= next_state;
    end
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: if (start) next_state = CHECK_COND;
        CHECK_COND: begin
            if (!k_mod_m_zero || k < m || k > n * m || k == 0) begin
                next_state = DONE;
            end else begin
                x = k / m;
                if (x > n) next_state = DONE;
                else next_state = SORT;
            end
        end
        SORT: if (i >= n) next_state = BUILD_PERM;
        BUILD_PERM: next_state = DONE;
        DONE: next_state = IDLE;
    endcase
end

// Main logic
always @(posedge clk) begin
    if (state == IDLE) begin
        done <= 0;
        possible <= 0;
        valid_indices <= 0;
        i <= 0;
        j <= 0;
    end else if (state == CHECK_COND) begin
        // Condition checking already done in next_state
    end else if (state == SORT) begin
        // Bubble sort a into descending order
        // This is a simplified version for small N
        // In practice, use a proper sorting network
        if (i < n - 1) begin
            if (a[i] < a[i+1]) begin
                // Swap
                a_sorted[i] <= a[i+1];
                a_sorted[i+1] <= a[i];
                idx_sorted[i] <= i;
                idx_sorted[i+1] <= i+1;
            end else begin
                a_sorted[i] <= a[i];
                idx_sorted[i] <= i;
            end
            i <= i + 1;
        end
    end else if (state == BUILD_PERM) begin
        // Construct permutation based on sorted array
        // This is a placeholder for the complex logic
        // In actual implementation, would build the order
        // For now, output a simple solution
        if (x == 1) begin
            // Only first critic gives m
            p[0] <= idx_sorted[0] + 1; // 1-based
            for (integer k = 1; k < N; k = k + 1) begin
                if (k < n) p[k] <= idx_sorted[k] + 1;
                else p[k] <= 0;
            end
        end else begin
            // More complex ordering for x > 1
            // Would implement the greedy algorithm here
            // For demonstration, output identity
            for (integer k = 0; k < N; k = k + 1) begin
                if (k < n) p[k] <= k + 1;
                else p[k] <= 0;
            end
        end
        possible <= 1; // Simplified: assume always possible if conditions met
        valid_indices <= n;
        done <= 1;
    end else if (state == DONE) begin
        done <= 0;
    end
end

endmodule
