module intersection_area (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [1:0] P,          // Number of pine points (0-3)
    input wire [1:0] A,          // Number of aspen points (0-3)
    input wire [31:0] pine_x0, pine_y0,
    input wire [31:0] pine_x1, pine_y1,
    input wire [31:0] pine_x2, pine_y2,
    input wire [31:0] aspen_x0, aspen_y0,
    input wire [31:0] aspen_x1, aspen_y1,
    input wire [31:0] aspen_x2, aspen_y2,
    output reg [31:0] result,
    output reg done
);

// Fixed-point parameters (Q16.16)
parameter WI = 16;
parameter WF = 16;
parameter W = 32;
parameter W2 = 64; // for multiplications

// States
localparam IDLE = 0;
localparam LOAD = 1;
localparam ORIENT = 2;
localparam PREP_CLIP = 3;
localparam CHECK_INSIDE = 4;
localparam COMPUTE_INTERSECTION = 5;
localparam ADD_VERTEX = 6;
localparam NEXT_EDGE = 7;
localparam NEXT_CLIP_EDGE = 8;
localparam CALC_AREA = 9;
localparam DONE_STATE = 10;
localparam DIVIDE = 11;

// Registers for state
reg [4:0] state, next_state;
reg [2:0] poly_cnt, new_poly_cnt;
reg signed [W-1:0] triA_x [0:2], triA_y [0:2];
reg signed [W-1:0] triB_x [0:2], triB_y [0:2];
reg signed [W-1:0] poly_x [0:5], poly_y [0:5];
reg signed [W-1:0] new_poly_x [0:5], new_poly_y [0:5];
reg signed [W-1:0] clip_start_x, clip_start_y, clip_end_x, clip_end_y;
reg signed [W-1:0] edge_start_x, edge_start_y, edge_end_x, edge_end_y;
reg [2:0] clip_edge_idx, edge_idx;
reg [2:0] vertex_idx;
reg signed [W2-1:0] area_sum;
reg area_sign;
reg signed [W2-1:0] cross_uv_reg, cross_vw_reg;
reg signed [W-1:0] int_x, int_y;
reg [5:0] div_cnt;
reg div_start;
reg signed [W-1:0] div_numer, div_denom, div_quot;
reg div_done;

// Helper: absolute value
function automatic signed [W-1:0] abs_val(input signed [W-1:0] val);
    abs_val = (val < 0) ? -val : val;
endfunction

// Helper: signed min/max
function automatic signed [W-1:0] smin(input signed [W-1:0] a, b);
    smin = (a < b) ? a : b;
endfunction

function automatic signed [W-1:0] smax(input signed [W-1:0] a, b);
    smax = (a > b) ? a : b;
endfunction

// 32-bit signed divider (restoring algorithm)
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        div_done <= 0;
        div_quot <= 0;
        div_cnt <= 0;
    end else begin
        if (div_start && !div_done) begin
            if (div_cnt == 0) begin
                // Initialize
                div_quot <= 0;
                div_cnt <= 1;
            end else if (div_cnt <= 32) begin
                // One iteration per cycle
                // Shift quotient left by 1
                div_quot <= div_quot << 1;
                // Subtract denominator from remainder (not implemented fully here)
                // For simplicity, we use a behavioral divider for simulation
                // In synthesis, a proper divider would be needed
                div_cnt <= div_cnt + 1;
            end else begin
                // Done
                div_done <= 1;
                // Actually compute quotient properly:
                // We'll use a behavioral division for simulation correctness
                if (div_denom != 0)
                    div_quot <= div_numer / div_denom;
                else
                    div_quot <= 0;
            end
        end else if (div_done) begin
            div_done <= 0;
            div_cnt <= 0;
        end
    end
end

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) state <= IDLE;
    else state <= next_state;
end

// Next state logic
always @(*) begin
    next_state = state;
    case (state)
        IDLE: if (start) next_state = LOAD;
        LOAD: next_state = ORIENT;
        ORIENT: if (P >= 3 && A >= 3) next_state = PREP_CLIP; else next_state = DONE_STATE;
        PREP_CLIP: begin
            if (clip_edge_idx < 3) next_state = CHECK_INSIDE;
            else next_state = CALC_AREA;
        end
        CHECK_INSIDE: begin
            if (edge_idx < poly_cnt) next_state = COMPUTE_INTERSECTION;
            else next_state = NEXT_CLIP_EDGE;
        end
        COMPUTE_INTERSECTION: begin
            if (div_done) next_state = ADD_VERTEX;
            else next_state = COMPUTE_INTERSECTION;
        end
        ADD_VERTEX: next_state = NEXT_EDGE;
        NEXT_EDGE: begin
            if (edge_idx < poly_cnt) next_state = CHECK_INSIDE;
            else next_state = NEXT_CLIP_EDGE;
        end
        NEXT_CLIP_EDGE: begin
            if (clip_edge_idx < 3) next_state = PREP_CLIP;
            else next_state = CALC_AREA;
        end
        CALC_AREA: next_state = DONE_STATE;
        DONE_STATE: if (!start) next_state = IDLE;
        default: next_state = IDLE;
    endcase
end

// Datapath
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        // Reset all registers
        result <= 0;
        done <= 0;
        poly_cnt <= 0;
        new_poly_cnt <= 0;
        clip_edge_idx <= 0;
        edge_idx <= 0;
        vertex_idx <= 0;
        area_sum <= 0;
        area_sign <= 0;
        div_start <= 0;
        int_x <= 0;
        int_y <= 0;
    end else begin
        case (state)
            LOAD: begin
                // Load pine points
                triA_x[0] <= pine_x0; triA_y[0] <= pine_y0;
                triA_x[1] <= pine_x1; triA_y[1] <= pine_y1;
                triA_x[2] <= pine_x2; triA_y[2] <= pine_y2;
                // Load aspen points
                triB_x[0] <= aspen_x0; triB_y[0] <= aspen_y0;
                triB_x[1] <= aspen_x1; triB_y[1] <= aspen_y1;
                triB_x[2] <= aspen_x2; triB_y[2] <= aspen_y2;
                poly_cnt <= 0;
                new_poly_cnt <= 0;
                clip_edge_idx <= 0;
                edge_idx <= 0;
                area_sum <= 0;
                area_sign <= 0;
            end

            ORIENT: begin
                // Orient triangle A to CCW
                if (P >= 3) begin
                    // signed area = (x1-x0)*(y2-y0) - (y1-y0)*(x2-x0)
                    if ( ((triA_x[1] - triA_x[0]) * (triA_y[2] - triA_y[0]) - 
                          (triA_y[1] - triA_y[0]) * (triA_x[2] - triA_x[0])) < 0 ) begin
                        // swap vertices 1 and 2
                        triA_x[1] <= triA_x[2]; triA_y[1] <= triA_y[2];
                        triA_x[2] <= triA_x[1]; triA_y[2] <= triA_y[1];
                    end
                end
                // Orient triangle B to CCW
                if (A >= 3) begin
                    if ( ((triB_x[1] - triB_x[0]) * (triB_y[2] - triB_y[0]) - 
                          (triB_y[1] - triB_y[0]) * (triB_x[2] - triB_x[0])) < 0 ) begin
                        triB_x[1] <= triB_x[2]; triB_y[1] <= triB_y[2];
                        triB_x[2] <= triB_x[1]; triB_y[2] <= triB_y[1];
                    end
                end
                // Initialize polygon as triangle A
                poly_cnt <= 3;
                poly_x[0] <= triA_x[0]; poly_y[0] <= triA_y[0];
                poly_x[1] <= triA_x[1]; poly_y[1] <= triA_y[1];
                poly_x[2] <= triA_x[2]; poly_y[2] <= triA_y[2];
            end

            PREP_CLIP: begin
                // Set clip edge from triangle B
                clip_edge_idx <= clip_edge_idx + 1;
                clip_start_x <= triB_x[clip_edge_idx];
                clip_start_y <= triB_y[clip_edge_idx];
                // next vertex, wrap around
                if (clip_edge_idx == 2) begin
                    clip_end_x <= triB_x[0];
                    clip_end_y <= triB_y[0];
                end else begin
                    clip_end_x <= triB_x[clip_edge_idx+1];
                    clip_end_y <= triB_y[clip_edge_idx+1];
                end
                new_poly_cnt <= 0;
                edge_idx <= 0;
            end

            CHECK_INSIDE: begin
                // Set current edge from polygon
                edge_start_x <= poly_x[edge_idx];
                edge_start_y <= poly_y[edge_idx];
                if (edge_idx == poly_cnt-1) begin
                    edge_end_x <= poly_x[0];
                    edge_end_y <= poly_y[0];
                end else begin
                    edge_end_x <= poly_x[edge_idx+1];
                    edge_end_y <= poly_y[edge_idx+1];
                end
                // Determine inside status of start vertex
                // inside = ( (clip_end_x - clip_start_x)*(edge_start_y - clip_start_y) - 
                //            (clip_end_y - clip_start_y)*(edge_start_x - clip_start_x) ) >= 0
                // We'll compute this in next state using cross product
            end

            COMPUTE_INTERSECTION: begin
                // Compute cross products for edge start and end relative to clip edge
                // For start vertex
                // cross_start = (clip_end_x - clip_start_x)*(edge_start_y - clip_start_y) - 
                //               (clip_end_y - clip_start_y)*(edge_start_x - clip_start_x)
                // For end vertex
                // cross_end = (clip_end_x - clip_start_x)*(edge_end_y - clip_start_y) - 
                //             (clip_end_y - clip_start_y)*(edge_end_x - clip_start_x)
                // Here we start division if needed (intersection case)
                if (div_done && div_start) begin
                    // Division finished, compute intersection point
                    // t = cross_vw / cross_uv
                    // point = edge_start + t * (edge_end - edge_start)
                    // We'll compute using the stored cross values
                    // For simulation, we use behavioral division
                    div_start <= 0;
                    // After division, we compute intersection point
                    // We'll compute later in ADD_VERTEX state
                end else if (!div_start && !div_done) begin
                    // Determine if we need to compute intersection
                    // We'll compute cross products in parallel and decide
                    // For simplicity, we compute cross products here
                    // We need to decide if edge crosses the clip edge
                    // This is a simplified version; in practice, we compute both crosses
                    // We'll compute cross_uv = clip vector cross edge vector
                    // cross_vw = clip vector cross (edge_start - clip_start)
                    // cross_uw = clip vector cross (edge_end - clip_start)
                    // We'll store them for division
                    cross_uv_reg <= (clip_end_x - clip_start_x)*(edge_end_y - edge_start_y) - 
                                   (clip_end_y - clip_start_y)*(edge_end_x - edge_start_x);
                    cross_vw_reg <= (clip_end_x - clip_start_x)*(edge_start_y - clip_start_y) - 
                                   (clip_end_y - clip_start_y)*(edge_start_x - clip_start_x);
                    // If edge crosses, need division (cross_uv != 0 and cross_vw and cross_vw+cross_uv have opposite signs)
                    if (cross_uv_reg != 0) begin
                        div_numer <= cross_vw_reg[W-1:0]; // Truncate to W bits (simplified)
                        div_denom <= cross_uv_reg[W-1:0];
                        div_start <= 1;
                    end
                end
            end

            ADD_VERTEX: begin
                // Add vertices to new_poly based on inside status
                // This is a placeholder; actual logic would add start vertex if inside,
                // add intersection if crossing, etc.
                // For brevity, we assume we have computed int_x, int_y from division
                // In full implementation, we would add:
                //   if (start_inside) add edge_start
                //   if (start_inside != end_inside) add intersection
                // Here we just increment counters
                if (new_poly_cnt < 6) begin
                    new_poly_x[new_poly_cnt] <= edge_start_x; // placeholder
                    new_poly_y[new_poly_cnt] <= edge_start_y;
                    new_poly_cnt <= new_poly_cnt + 1;
                end
                edge_idx <= edge_idx + 1;
            end

            NEXT_EDGE: begin
                // Move to next edge
                if (edge_idx == poly_cnt) begin
                    // Finished all edges, replace polygon
                    poly_cnt <= new_poly_cnt;
                    for (integer i = 0; i < 6; i = i + 1) begin
                        poly_x[i] <= new_poly_x[i];
                        poly_y[i] <= new_poly_y[i];
                    end
                end
            end

            NEXT_CLIP_EDGE: begin
                // Prepare for next clip edge
                // Already incremented clip_edge_idx in PREP_CLIP
            end

            CALC_AREA: begin
                // Compute area of final polygon using shoelace
                // area_sum = 0.5 * |sum(x_i*y_{i+1} - x_{i+1}*y_i)|
                // For simplicity, we compute using a loop (unrolled in practice)
                // We'll compute sum of cross products
                // This is a placeholder; actual computation would be done in previous states
                // Output area in Q16.16 format
                result <= area_sum[W2-1:W]; // scale back to 32-bit
            end

            DONE_STATE: begin
                done <= 1;
                if (!start) done <= 0;
            end
        endcase
    end
end

endmodule
