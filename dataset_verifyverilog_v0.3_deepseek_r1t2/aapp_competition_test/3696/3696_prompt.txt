module polynomial_generator #(
    parameter MAX_N = 8,
    parameter MAX_COEFF = MAX_N + 1
) (
    input clk,
    input rst_n,
    input start,
    input [3:0] n,
    output reg done,
    output reg [3:0] out_degree1,
    output reg [3:0] out_degree2,
    output reg [2:0] out_coeffs1 [0:MAX_COEFF-1],
    output reg [2:0] out_coeffs2 [0:MAX_COEFF-1]
);

// State encoding
localparam IDLE = 2'b00;
localparam COMPUTE = 2'b01;
localparam DONE = 2'b10;

reg [1:0] state, next_state;
reg [3:0] i; // iteration counter (2 to n)
reg signed [2:0] poly_prev1 [0:MAX_COEFF-1]; // P_{i-1}
reg signed [2:0] poly_prev2 [0:MAX_COEFF-1]; // P_{i-2}
reg signed [2:0] new_poly [0:MAX_COEFF-1];   // x * poly_prev1
reg signed [2:0] candidate1 [0:MAX_COEFF-1];
reg signed [2:0] candidate2 [0:MAX_COEFF-1];
reg valid1, valid2;

integer j;

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
    end else begin
        state <= next_state;
    end
end

// Next state logic
always @(*) begin
    case (state)
        IDLE: next_state = start ? COMPUTE : IDLE;
        COMPUTE: begin
            if (i == n)
                next_state = DONE;
            else
                next_state = COMPUTE;
        end
        DONE: next_state = IDLE;
        default: next_state = IDLE;
    endcase
end

// Computation logic
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        done <= 0;
        i <= 2;
        // Initialize polynomials
        poly_prev2[0] <= 3'sd1; // P0: [1]
        poly_prev2[1] <= 3'sd0; // unused, but set to 0
        poly_prev1[0] <= 3'sd0; // P1: [0,1]
        poly_prev1[1] <= 3'sd1;
        // Set degrees
        out_degree1 <= 0;
        out_degree2 <= 0;
        // Clear output arrays
        for (j=0; j<MAX_COEFF; j=j+1) begin
            out_coeffs1[j] <= 3'sd0;
            out_coeffs2[j] <= 3'sd0;
        end
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    if (n == 1) begin
                        // Directly output for n=1
                        out_degree1 <= 1;
                        out_degree2 <= 0;
                        out_coeffs1[0] <= 3'sd0; out_coeffs1[1] <= 3'sd1;
                        out_coeffs2[0] <= 3'sd1;
                        done <= 1;
                        next_state <= DONE;
                    end else begin
                        i <= 2;
                        // Reset computation variables
                        for (j=0; j<MAX_COEFF; j=j+1) begin
                            new_poly[j] <= 3'sd0;
                            candidate1[j] <= 3'sd0;
                            candidate2[j] <= 3'sd0;
                        end
                    end
                end
            end

            COMPUTE: begin
                // Compute new_poly = x * poly_prev1
                new_poly[0] <= 3'sd0;
                for (j=0; j<i; j=j+1) begin
                    new_poly[j+1] <= poly_prev1[j];
                end

                // Compute candidate1 = new_poly + poly_prev2
                // Compute candidate2 = new_poly - poly_prev2
                valid1 <= 1;
                valid2 <= 1;
                for (j=0; j<i-1; j=j+1) begin
                    candidate1[j] <= new_poly[j] + poly_prev2[j];
                    candidate2[j] <= new_poly[j] - poly_prev2[j];
                    // Check candidate1 validity
                    if (candidate1[j] != -1 && candidate1[j] != 0 && candidate1[j] != 1)
                        valid1 <= 0;
                    // Check candidate2 validity
                    if (candidate2[j] != -1 && candidate2[j] != 0 && candidate2[j] != 1)
                        valid2 <= 0;
                end
                // For j >= i-1, candidate and new_poly are the same
                for (j=i-1; j<i+1; j=j+1) begin
                    candidate1[j] <= new_poly[j];
                    candidate2[j] <= new_poly[j];
                end

                // Choose valid candidate
                if (valid1) begin
                    for (j=0; j<i+1; j=j+1)
                        poly_prev1[j] <= candidate1[j];
                end else if (valid2) begin
                    for (j=0; j<i+1; j=j+1)
                        poly_prev1[j] <= candidate2[j];
                end
                poly_prev2 <= poly_prev1; // shift

                if (i == n) begin
                    // Prepare output
                    out_degree1 <= n;
                    out_degree2 <= n-1;
                    for (j=0; j<n+1; j=j+1)
                        out_coeffs1[j] <= poly_prev1[j];
                    for (j=0; j<n; j=j+1)
                        out_coeffs2[j] <= poly_prev2[j];
                    done <= 1;
                end else begin
                    i <= i + 1;
                end
            end

            DONE: begin
                done <= 0;
            end
        endcase
    end
end

endmodule
