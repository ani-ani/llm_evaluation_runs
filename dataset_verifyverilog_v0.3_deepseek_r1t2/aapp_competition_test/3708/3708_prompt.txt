module min_time_path (
    input wire clk,              // Clock signal
    input wire rst_n,            // Active-low reset
    input wire start,            // Start computation
    input wire setup_mode,       // Setup mode for volcano input
    input wire volcano_valid,    // Valid volcano coordinate
    input wire [3:0] volcano_x,  // Volcano x-coordinate (0-indexed)
    input wire [3:0] volcano_y,  // Volcano y-coordinate (0-indexed)
    input wire [3:0] n,          // Grid size (1-16)
    output reg done,             // Computation complete
    output reg [7:0] result      // Result: steps or 8'hFF for -1
);

    // State encoding
    localparam IDLE = 2'b00;
    localparam COMPUTE = 2'b01;
    localparam DONE = 2'b10;

    // Registers
    reg [1:0] state_reg;
    reg [3:0] i_reg, j_reg;       // Current cell indices
    reg [15:0] prev_row_reg;      // Previous row of DP
    reg [15:0] curr_row_reg;      // Current row of DP
    reg [15:0] blocked_rows [0:15]; // Blocked grid (16x16)

    // Combinational bit calculation
    wire bit_val;
    assign bit_val = blocked_rows[i_reg][j_reg] ? 0 :
                     (i_reg == 0 && j_reg == 0) ? 1 :
                     ((i_reg > 0 ? prev_row_reg[j_reg] : 0) |
                      (j_reg > 0 ? curr_row_reg[j_reg-1] : 0));

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Reset state
            state_reg <= IDLE;
            done <= 0;
            result <= 0;
            i_reg <= 0;
            j_reg <= 0;
            prev_row_reg <= 0;
            curr_row_reg <= 0;
            // Clear blocked grid
            for (int i = 0; i < 16; i++) 
                blocked_rows[i] <= 16'b0;
        end else begin
            done <= 0;
            case (state_reg)
                IDLE: begin
                    // Handle volcano setup
                    if (setup_mode && volcano_valid) 
                        blocked_rows[volcano_x][volcano_y] <= 1;
                    
                    if (start) begin
                        // Initialize computation
                        i_reg <= 0;
                        j_reg <= 0;
                        prev_row_reg <= 16'b0;
                        curr_row_reg <= 16'b0;
                        state_reg <= COMPUTE;
                    end
                end
                
                COMPUTE: begin
                    // Update current cell
                    curr_row_reg[j_reg] <= bit_val;
                    
                    // Move to next cell
                    if (j_reg == n-1) begin
                        j_reg <= 0;
                        if (i_reg == n-1) begin
                            // Grid complete
                            state_reg <= DONE;
                            result <= bit_val ? (n-1) << 1 : 8'hFF;
                        end else begin
                            // Next row
                            i_reg <= i_reg + 1;
                            prev_row_reg <= curr_row_reg;
                        end
                    end else begin
                        j_reg <= j_reg + 1;
                    end
                end
                
                DONE: begin
                    done <= 1;
                    state_reg <= IDLE;
                end
            endcase
        end
    end
endmodule
