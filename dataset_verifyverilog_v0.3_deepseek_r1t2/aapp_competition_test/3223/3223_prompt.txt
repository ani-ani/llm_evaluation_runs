module ForestBuilder #(
    parameter MAX_V = 8,
    parameter DATA_WIDTH = 8,
    parameter EDGE_WIDTH = 4
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] V,
    input wire [DATA_WIDTH-1:0] degrees [0:MAX_V-1],
    output reg possible,
    output reg done,
    output reg edge_valid,
    output reg [EDGE_WIDTH-1:0] edge_a,
    output reg [EDGE_WIDTH-1:0] edge_b
);

reg [DATA_WIDTH-1:0] deg [0:MAX_V-1];
reg [3:0] comp_id [0:MAX_V-1];
reg [3:0] V_reg;
reg [DATA_WIDTH-1:0] sum_deg, max_deg, m;
reg [3:0] u, v;
reg [3:0] idx;
reg [3:0] best_deg, best_u;
reg found_v;

localparam IDLE = 0;
localparam INIT = 1;
localparam FIND_U = 2;
localparam FIND_V = 3;
localparam ADD_EDGE = 4;
localparam CHECK_DONE = 5;
localparam FINISHED = 6;

reg [2:0] state, next_state;

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        possible <= 0;
        done <= 0;
        edge_valid <= 0;
        edge_a <= 0;
        edge_b <= 0;
        state <= IDLE;
    end else begin
        state <= next_state;
        edge_valid <= 0;
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    V_reg <= V;
                    for (integer i = 0; i < MAX_V; i = i + 1) begin
                        if (i < V) deg[i] <= degrees[i]; else deg[i] <= 0;
                        comp_id[i] <= i;
                    end
                    sum_deg <= 0;
                    max_deg <= 0;
                    m <= 0;
                    idx <= 0;
                    next_state <= INIT;
                end else next_state <= IDLE;
            end
            INIT: begin
                if (idx < V_reg) begin
                    if (deg[idx] > 0) begin
                        sum_deg <= sum_deg + deg[idx];
                        m <= m + 1;
                        if (deg[idx] > max_deg) max_deg <= deg[idx];
                    end
                    idx <= idx + 1;
                    next_state <= INIT;
                end else begin
                    if (sum_deg == 0) begin
                        possible <= 1; done <= 1; next_state <= IDLE;
                    end else if (sum_deg[0] == 1 || (m > 0 && (max_deg > (m-1) || sum_deg > 2*(m-1)))) begin
                        possible <= 0; done <= 1; next_state <= IDLE;
                    end else begin
                        possible <= 1;
                        best_deg <= 0;
                        best_u <= 0;
                        idx <= 0;
                        next_state <= FIND_U;
                    end
                end
            end
            FIND_U: begin
                if (idx < V_reg) begin
                    if (deg[idx] > best_deg) begin
                        best_deg <= deg[idx];
                        best_u <= idx;
                    end
                    idx <= idx + 1;
                    next_state <= FIND_U;
                end else begin
                    if (best_deg == 0) begin
                        done <= 1; next_state <= IDLE;
                    end else begin
                        u <= best_u;
                        best_deg <= 0;
                        idx <= 0;
                        found_v <= 0;
                        next_state <= FIND_V;
                    end
                end
            end
            FIND_V: begin
                if (idx < V_reg) begin
                    if (idx != u && deg[idx] > 0 && comp_id[idx] != comp_id[u]) begin
                        v <= idx;
                        found_v <= 1;
                    end
                    idx <= idx + 1;
                    next_state <= FIND_V;
                end else begin
                    if (found_v) next_state <= ADD_EDGE;
                    else begin
                        possible <= 0; done <= 1; next_state <= IDLE;
                    end
                    idx <= 0;
                end
            end
            ADD_EDGE: begin
                edge_a <= u + 1;
                edge_b <= v + 1;
                edge_valid <= 1;
                deg[u] <= deg[u] - 1;
                deg[v] <= deg[v] - 1;
                for (integer i = 0; i < MAX_V; i = i + 1) begin
                    if (i < V_reg && comp_id[i] == comp_id[v]) comp_id[i] <= comp_id[u];
                end
                next_state <= CHECK_DONE;
            end
            CHECK_DONE: begin
                if (idx < V_reg) begin
                    if (deg[idx] > 0) begin
                        idx <= 0;
                        best_deg <= 0;
                        next_state <= FIND_U;
                    end else begin
                        idx <= idx + 1;
                        next_state <= CHECK_DONE;
                    end
                end else begin
                    done <= 1;
                    next_state <= IDLE;
                end
            end
            default: next_state <= IDLE;
        endcase
    end
end

endmodule
