module min_trucks(
    input clk,
    input rst_n,
    input start,
    input [3:0] C,        // Number of clients (1-8)
    input [63:0] reach,   // Reachability matrix: bit i*8+j = 1 if truck can go from client i to client j
    output reg [3:0] result,
    output reg done
);

// Internal memory for DP (256 x 4 bits)
reg [3:0] dp0 [0:255];  // Current DP state
reg [3:0] dp1 [0:255];  // Next DP state

// State machine states
localparam IDLE = 0;
localparam INIT = 1;
localparam LOOP_I = 2;
localparam COPY_DP = 3;
localparam INNER_MASK = 4;
localparam INNER_J = 5;
localparam SWAP_DP = 6;
localparam FIND_MAX = 7;
localparam DONE = 8;

reg [3:0] state;
reg [2:0] i;           // Current client index (0-7)
reg [7:0] mask;        // Current mask for right nodes
reg [2:0] j;           // Current right node index
reg [3:0] max_match;   // Maximum matching found
reg [7:0] addr;        // Memory address counter
reg [3:0] temp_val;    // Temporary storage

// Helper: Check if bit is set in reach matrix
wire edge_ij = reach[i*8 + j];

// Helper: Check if right node j is in mask
wire mask_has_j = mask[j];

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        done <= 0;
        result <= 0;
        i <= 0;
        mask <= 0;
        j <= 0;
        max_match <= 0;
        addr <= 0;
        // Reset DP memories
        for (addr = 0; addr < 256; addr = addr + 1) begin
            dp0[addr] <= 0;
            dp1[addr] <= 0;
        end
    end else begin
        case (state)
            IDLE: begin
                done <= 0;
                if (start) begin
                    state <= INIT;
                    i <= 0;
                    addr <= 0;
                end
            end

            INIT: begin
                // Initialize dp0[0] = 0, others = 0 (already reset)
                if (addr == 255) begin
                    dp0[0] <= 0;  // Ensure first is zero
                    state <= LOOP_I;
                end else begin
                    addr <= addr + 1;
                end
            end

            LOOP_I: begin
                if (i >= C) begin
                    state <= FIND_MAX;
                end else begin
                    state <= COPY_DP;
                    addr <= 0;
                end
            end

            COPY_DP: begin
                // Copy dp0 to dp1
                dp1[addr] <= dp0[addr];
                if (addr == 255) begin
                    state <= INNER_MASK;
                    mask <= 0;
                end else begin
                    addr <= addr + 1;
                end
            end

            INNER_MASK: begin
                if (mask == 8'hFF) begin
                    state <= SWAP_DP;
                end else begin
                    j <= 0;
                    state <= INNER_J;
                end
            end

            INNER_J: begin
                if (j >= C) begin
                    mask <= mask + 1;
                    state <= INNER_MASK;
                end else if (!mask_has_j && edge_ij) begin
                    // Found valid edge: try to update dp1[mask | (1<<j)]
                    if (dp0[mask] != 0 || mask == 0) begin  // Only if dp0[mask] is valid
                        temp_val <= dp0[mask] + 1;
                        if (dp0[mask] + 1 > dp1[mask | (1<<j)]) begin
                            dp1[mask | (1<<j)] <= dp0[mask] + 1;
                        end
                    end
                    j <= j + 1;
                end else begin
                    j <= j + 1;
                end
            end

            SWAP_DP: begin
                // Copy dp1 back to dp0 for next iteration
                addr <= 0;
                state <= COPY_BACK;
            end

            COPY_BACK: begin
                dp0[addr] <= dp1[addr];
                if (addr == 255) begin
                    i <= i + 1;
                    state <= LOOP_I;
                end else begin
                    addr <= addr + 1;
                end
            end

            FIND_MAX: begin
                // Find maximum value in dp0
                if (addr == 255) begin
                    // Done searching
                    result <= C - max_match;
                    state <= DONE;
                end else begin
                    if (dp0[addr] > max_match)
                        max_match <= dp0[addr];
                    addr <= addr + 1;
                end
            end

            DONE: begin
                done <= 1;
                if (!start) begin
                    state <= IDLE;
                    done <= 0;
                end
            end
        endcase
    end
end

endmodule
