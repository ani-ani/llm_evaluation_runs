module StoneGame(
    input [3:0] n,
    input [31:0] a0, a1, a2, a3, a4, a5, a6, a7,
    output reg win
);
    // Internal array
    wire [31:0] arr [0:7];
    assign arr[0] = a0;
    assign arr[1] = a1;
    assign arr[2] = a2;
    assign arr[3] = a3;
    assign arr[4] = a4;
    assign arr[5] = a5;
    assign arr[6] = a6;
    assign arr[7] = a7;

    // Sorting and processing
    reg [31:0] temp [0:7];
    integer i, j, k;
    reg [31:0] swap;
    reg [2:0] dup_count;
    reg [31:0] dup_value;
    reg triple;
    reg lose_condition;
    reg found_dup_value_minus_one;
    reg [31:0] sum_val;
    reg [31:0] base;
    reg [31:0] S;

    always @(*) begin
        // Initialize with padding
        for (i = 0; i < 8; i = i + 1) begin
            if (i < n)
                temp[i] = arr[i];
            else
                temp[i] = 32'hFFFF_FFFF;
        end

        // Bubble sort
        for (i = 0; i < 8; i = i + 1) begin
            for (j = 0; j < 7 - i; j = j + 1) begin
                if (temp[j] > temp[j+1]) begin
                    swap = temp[j];
                    temp[j] = temp[j+1];
                    temp[j+1] = swap;
                end
            end
        end

        // Duplicate detection
        dup_count = 0;
        dup_value = 0;
        triple = 0;
        lose_condition = 0;
        
        for (i = 0; i < n-1; i = i + 1) begin
            if (temp[i] == temp[i+1]) begin
                dup_count = dup_count + 1;
                dup_value = temp[i];
                if (i+2 < n && temp[i] == temp[i+2])
                    triple = 1;
            end
        end

        if (triple || dup_count >= 2)
            lose_condition = 1;
        else if (dup_count == 1) begin
            if (dup_value == 0)
                lose_condition = 1;
            else begin
                found_dup_value_minus_one = 0;
                for (k = 0; k < n; k = k + 1) begin
                    if (temp[k] == dup_value - 1)
                        found_dup_value_minus_one = 1;
                end
                if (found_dup_value_minus_one)
                    lose_condition = 1;
            end
        end

        // Parity calculation
        sum_val = 0;
        for (i = 0; i < n; i = i + 1)
            sum_val = sum_val + temp[i];
        base = (n * (n-1)) >> 1;
        S = sum_val - base;

        // Win determination
        win = lose_condition ? 0 : S[0];
    end
endmodule
