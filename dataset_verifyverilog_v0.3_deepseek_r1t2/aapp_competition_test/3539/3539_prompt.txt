module BipartiteBattle (
    input clk,
    input rst_n,
    input start,
    input [2:0] N,
    input [7:0] a_i [0:7],
    input [7:0] b_i [0:7],
    output reg [31:0] result,
    output reg done
);
    parameter MOD = 1000000007;
    
    reg [31:0] total_edges;
    reg [31:0] cnt;
    reg [2:0] idx;
    reg computing;
    reg [31:0] power;
    reg [31:0] exponent;
    
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            result <= 0;
            done <= 0;
            total_edges <= 0;
            cnt <= 0;
            idx <= 0;
            computing <= 0;
            power <= 0;
            exponent <= 0;
        end else begin
            if (start && !computing) begin
                total_edges <= 0;
                cnt <= 0;
                idx <= 0;
                computing <= 1;
                done <= 0;
            end else if (computing) begin
                if (idx < N) begin
                    total_edges <= total_edges + a_i[idx] * b_i[idx];
                    if (a_i[idx][0] && b_i[idx][0]) begin
                        cnt <= cnt + 1;
                    end
                    idx <= idx + 1;
                end else if (idx == N) begin
                    if (cnt[0]) begin
                        exponent <= total_edges - 1;
                        power <= 1;
                        idx <= idx + 1;
                    end else begin
                        result <= 0;
                        done <= 1;
                        computing <= 0;
                    end
                end else if (idx > N) begin
                    if (exponent > 0) begin
                        power <= (power * 2) % MOD;
                        exponent <= exponent - 1;
                    end else begin
                        result <= power;
                        done <= 1;
                        computing <= 0;
                    end
                end
            end else begin
                done <= 0;
            end
        end
    end
endmodule
