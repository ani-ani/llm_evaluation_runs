module train_compensation #(
    parameter MAX_N = 8,           // Maximum stations (1-8)
    parameter MAX_M = 64,          // Maximum trains (1-64)
    parameter TIME_WIDTH = 12,     // Time value width (0-4095)
    parameter STATION_WIDTH = 3,   // Station index width (0-7)
    parameter REFUND_DELAY = 1800  // Required delay for refund
)(
    input wire clk,
    input wire rst_n,
    input wire start,
    
    // Configuration
    input wire [STATION_WIDTH:0] N,  // Number of stations
    input wire [STATION_WIDTH:0] M,  // Number of trains
    
    // Train data input (streaming)
    input wire [47:0] train_data,    // Packed: {station[3:0], S[11:0], T[11:0], L[11:0]}
    input wire train_valid,
    input wire train_last,
    
    // Output
    output reg [TIME_WIDTH-1:0] result,
    output reg result_valid,
    output reg impossible
);

// Internal storage for trains
reg [STATION_WIDTH-1:0] train_stations [MAX_M-1:0];
reg [TIME_WIDTH-1:0] train_S_arr [MAX_M-1:0];
reg [TIME_WIDTH-1:0] train_T_arr [MAX_M-1:0];
reg [TIME_WIDTH-1:0] train_L_arr [MAX_M-1:0];
reg train_valid_arr [MAX_M-1:0];

// State machine
reg [2:0] state;
localparam IDLE = 3'd0;
localparam LOAD = 3'd1;
localparam COMPUTE = 3'd2;
localparam DONE = 3'd3;

// Computation registers
reg [STATION_WIDTH:0] load_idx;
reg [STATION_WIDTH:0] candidate_idx;
reg [TIME_WIDTH-1:0] best_time;
reg best_found;

// Sub-state for journey simulation
reg [STATION_WIDTH:0] sim_station;
reg [TIME_WIDTH-1:0] sim_time;
reg sim_feasible;
reg [1:0] sim_state;  // 0=on-time, 1=actual, 2=done
reg [STATION_WIDTH:0] sim_train_idx;

// Helper task to find next train
task find_next_train;
    input [STATION_WIDTH-1:0] station;
    input [TIME_WIDTH-1:0] current_time;
    input is_actual;
    output found;
    output [TIME_WIDTH-1:0] arrival;
    integer i;
    begin
        found = 0; arrival = 0;
        for (i = 0; i < MAX_M; i = i + 1) begin
            if (train_valid_arr[i] && train_stations[i] == station) begin
                if (is_actual) begin
                    if (train_S_arr[i] + train_L_arr[i] >= current_time) begin
                        arrival = train_T_arr[i] + train_L_arr[i];
                        found = 1; return;
                    end
                end else begin
                    if (train_S_arr[i] >= current_time) begin
                        arrival = train_T_arr[i];
                        found = 1; return;
                    end
                end
            end
        end
    end
endtask

// Main state machine
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= IDLE;
        result_valid <= 0;
        impossible <= 0;
        load_idx <= 0;
        candidate_idx <= 0;
        best_time <= {TIME_WIDTH{1'b1}};
        best_found <= 0;
        for (integer i = 0; i < MAX_M; i = i + 1)
            train_valid_arr[i] <= 0;
    end else begin
        case (state)
            IDLE: begin
                if (start) begin
                    state <= LOAD;
                    load_idx <= 0;
                    result_valid <= 0;
                end
            end
            
            LOAD: begin
                if (train_valid && load_idx < M) begin
                    train_stations[load_idx] <= train_data[47:44];
                    train_S_arr[load_idx] <= train_data[43:32];
                    train_T_arr[load_idx] <= train_data[31:20];
                    train_L_arr[load_idx] <= train_data[19:8];
                    train_valid_arr[load_idx] <= 1;
                    load_idx <= load_idx + 1;
                end else if (train_last || load_idx >= M) begin
                    state <= COMPUTE;
                    candidate_idx <= 0;
                    best_time <= {TIME_WIDTH{1'b1}};
                    best_found <= 0;
                end
            end
            
            COMPUTE: begin
                if (candidate_idx < M) begin
                    // Check if from station 1 (index 0)
                    if (train_valid_arr[candidate_idx] && train_stations[candidate_idx] == 0) begin
                        // Initialize on-time simulation
                        sim_station <= 1;
                        sim_time <= train_S_arr[candidate_idx];
                        sim_feasible <= 1;
                        sim_state <= 0;  // on-time
                        sim_train_idx <= 0;
                        // Will continue in next cycle
                    end else begin
                        candidate_idx <= candidate_idx + 1;
                    end
                end else begin
                    state <= DONE;
                    if (best_found) begin
                        result <= best_time;
                        result_valid <= 1;
                        impossible <= 0;
                    end else begin
                        result <= 0;
                        result_valid <= 1;
                        impossible <= 1;
                    end
                end
            end
            
            DONE: begin
                if (!start) begin
                    state <= IDLE;
                    result_valid <= 0;
                end
            end
        endcase
        
        // Journey simulation sub-state machine (when in COMPUTE and sim_state active)
        if (state == COMPUTE && sim_state < 2'd2) begin
            if (sim_station < N && sim_feasible) begin
                // Find next train
                if (sim_state == 0) begin  // on-time
                    reg found;
                    reg [TIME_WIDTH-1:0] arrival;
                    find_next_train(sim_station, sim_time, 0, found, arrival);
                    if (found) begin
                        sim_time <= arrival;
                        sim_station <= sim_station + 1;
                    end else begin
                        sim_feasible <= 0;
                    end
                end else begin  // actual
                    reg found;
                    reg [TIME_WIDTH-1:0] arrival;
                    find_next_train(sim_station, sim_time, 1, found, arrival);
                    if (found) begin
                        sim_time <= arrival;
                        sim_station <= sim_station + 1;
                    end else begin
                        sim_feasible <= 0;
                    end
                end
            end else begin
                // Simulation complete for this state
                if (sim_state == 0) begin
                    // On-time complete
                    if (sim_feasible) begin
                        // Store A_on and start actual simulation
                        // [Note: In full implementation, store A_on in a register]
                        sim_state <= 1;
                        sim_station <= 1;
                        sim_time <= train_S_arr[candidate_idx];
                        sim_feasible <= 1;
                    end else begin
                        // On-time not feasible, skip
                        candidate_idx <= candidate_idx + 1;
                        sim_state <= 2;
                    end
                end else begin
                    // Actual complete
                    if (sim_feasible) begin
                        // Both simulations done, check condition
                        // [Here check if sim_time - a_on >= REFUND_DELAY]
                        // and update best_time if needed
                    end else begin
                        // Actual not feasible - satisfies condition
                        if (train_S_arr[candidate_idx] < best_time) begin
                            best_time <= train_S_arr[candidate_idx];
                            best_found <= 1;
                        end
                    end
                    candidate_idx <= candidate_idx + 1;
                    sim_state <= 2;
                end
            end
        end
    end
end

endmodule
