module beautiful_sequence(
    input clk,
    input rst_n,
    input start,
    input [7:0] a,  // count of 0s
    input [7:0] b,  // count of 1s
    input [7:0] c,  // count of 2s
    input [7:0] d,  // count of 3s
    output reg [7:0] seq_out [0:31],  // output sequence (max 32 elements)
    output reg [5:0] seq_len,          // length of output sequence
    output reg done
);

// Internal state
reg [7:0] counts [0:3];  // counts for 0,1,2,3
reg [7:0] temp_counts [0:3];  // temporary copy for simulation
reg [2:0] current_start;  // current starting point
reg [5:0] idx;            // index in sequence
reg [2:0] current_num;    // current number in sequence
reg [2:0] next_num;       // next number to try
reg [1:0] state;          // state machine: 0=idle, 1=run, 2=check, 3=done
reg [2:0] attempts;       // which start we are trying

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        done <= 0;
        state <= 0;
        seq_len <= 0;
        attempts <= 0;
    end else begin
        case (state)
            0: begin  // idle
                if (start) begin
                    // Initialize counts
                    counts[0] <= a;
                    counts[1] <= b;
                    counts[2] <= c;
                    counts[3] <= d;
                    attempts <= 0;
                    state <= 1;
                    done <= 0;
                end
            end
            
            1: begin  // try next starting point
                if (attempts < 4) begin
                    if (counts[attempts] > 0) begin
                        // Make copy of counts
                        temp_counts[0] <= counts[0];
                        temp_counts[1] <= counts[1];
                        temp_counts[2] <= counts[2];
                        temp_counts[3] <= counts[3];
                        current_start <= attempts;
                        current_num <= attempts;
                        idx <= 0;
                        seq_out[0] <= attempts;
                        temp_counts[attempts] <= temp_counts[attempts] - 1;
                        state <= 2;
                    end else begin
                        attempts <= attempts + 1;
                    end
                end else begin
                    // All attempts failed
                    state <= 3;
                    done <= 1;
                end
            end
            
            2: begin  // run sequence construction
                if (idx < 31) begin
                    // Find next number
                    case (current_num)
                        0: begin
                            if (temp_counts[1] > 0) begin
                                next_num <= 1;
                                temp_counts[1] <= temp_counts[1] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 1;
                                current_num <= 1;
                            end else begin
                                // Cannot continue
                                state <= 1;  // try next start
                                attempts <= attempts + 1;
                            end
                        end
                        1: begin
                            if (temp_counts[0] > 0) begin
                                next_num <= 0;
                                temp_counts[0] <= temp_counts[0] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 0;
                                current_num <= 0;
                            end else if (temp_counts[2] > 0) begin
                                next_num <= 2;
                                temp_counts[2] <= temp_counts[2] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 2;
                                current_num <= 2;
                            end else begin
                                state <= 1;
                                attempts <= attempts + 1;
                            end
                        end
                        2: begin
                            if (temp_counts[3] > 0) begin
                                next_num <= 3;
                                temp_counts[3] <= temp_counts[3] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 3;
                                current_num <= 3;
                            end else if (temp_counts[1] > 0) begin
                                next_num <= 1;
                                temp_counts[1] <= temp_counts[1] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 1;
                                current_num <= 1;
                            end else begin
                                state <= 1;
                                attempts <= attempts + 1;
                            end
                        end
                        3: begin
                            if (temp_counts[2] > 0) begin
                                next_num <= 2;
                                temp_counts[2] <= temp_counts[2] - 1;
                                idx <= idx + 1;
                                seq_out[idx + 1] <= 2;
                                current_num <= 2;
                            end else begin
                                state <= 1;
                                attempts <= attempts + 1;
                            end
                        end
                    endcase
                end else begin
                    // Sequence too long
                    state <= 1;
                    attempts <= attempts + 1;
                end
                
                // Check if all used
                if (temp_counts[0] == 0 && temp_counts[1] == 0 && 
                    temp_counts[2] == 0 && temp_counts[3] == 0) begin
                    seq_len <= idx + 1;
                    state <= 3;
                    done <= 1;
                end
            end
            
            3: begin  // done
                // Stay in done state
            end
        endcase
    end
end

endmodule
