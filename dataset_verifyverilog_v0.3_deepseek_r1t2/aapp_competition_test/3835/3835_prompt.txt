module restore_array (
    input clk,
    input rst_n,
    input start,
    input load,
    input [31:0] data_in,
    output reg [31:0] array_out_0,
    output reg [31:0] array_out_1,
    output reg [31:0] array_out_2,
    output reg [31:0] array_out_3,
    output reg [31:0] array_out_4,
    output reg [31:0] array_out_5,
    output reg [31:0] array_out_6,
    output reg [31:0] array_out_7,
    output reg done
);

parameter N = 8;
parameter MEM_SIZE = 64;

reg [31:0] mem [0:63];
reg [5:0] load_counter;

reg [3:0] state;
reg [3:0] i;
reg [3:0] j, k;
reg [31:0] A, B, C;
reg [63:0] product;
reg [63:0] quotient;
reg [31:0] sqrt_val;
reg [5:0] bit_index;
reg [5:0] address;

wire [31:0] data_out;
assign data_out = mem[address];

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        load_counter <= 0;
        state <= 0;
        done <= 0;
        i <= 0;
        array_out_0 <= 0;
        array_out_1 <= 0;
        array_out_2 <= 0;
        array_out_3 <= 0;
        array_out_4 <= 0;
        array_out_5 <= 0;
        array_out_6 <= 0;
        array_out_7 <= 0;
    end else begin
        case (state)
            0: begin // IDLE
                done <= 0;
                if (start) begin
                    i <= 0;
                    state <= 1;
                end
            end
            1: begin // READ1
                j <= (i + 1) % N;
                k <= (i + 2) % N;
                address <= i * N + ((i + 1) % N);
                state <= 2;
            end
            2: begin // READ2
                A <= data_out;
                address <= i * N + ((i + 2) % N);
                state <= 3;
            end
            3: begin // READ3
                B <= data_out;
                address <= j * N + k;
                state <= 4;
            end
            4: begin // MULT
                C <= data_out;
                product <= A * B;
                state <= 5;
            end
            5: begin // DIV
                quotient <= product / C;
                state <= 6;
            end
            6: begin // SQRT_START
                sqrt_val <= 0;
                bit_index <= 31;
                state <= 7;
            end
            7: begin // SQRT_LOOP
                if ((sqrt_val + (32'd1 << bit_index)) * (sqrt_val + (32'd1 << bit_index)) <= quotient) begin
                    sqrt_val <= sqrt_val + (32'd1 << bit_index);
                end
                if (bit_index == 0) begin
                    state <= 8;
                end else begin
                    bit_index <= bit_index - 1;
                end
            end
            8: begin // SQRT_DONE
                case (i)
                    0: array_out_0 <= sqrt_val;
                    1: array_out_1 <= sqrt_val;
                    2: array_out_2 <= sqrt_val;
                    3: array_out_3 <= sqrt_val;
                    4: array_out_4 <= sqrt_val;
                    5: array_out_5 <= sqrt_val;
                    6: array_out_6 <= sqrt_val;
                    7: array_out_7 <= sqrt_val;
                endcase
                i <= i + 1;
                if (i < N-1) state <= 1;
                else begin
                    state <= 9;
                    done <= 1;
                end
            end
            9: begin // DONE
                // Computation complete
            end
        endcase

        if (load) begin
            if (load_counter < MEM_SIZE) begin
                mem[load_counter] <= data_in;
                load_counter <= load_counter + 1;
            end
        end
    end
end
endmodule
