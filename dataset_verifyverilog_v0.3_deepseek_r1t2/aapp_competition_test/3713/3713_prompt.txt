module longest_alternating_subsequence(
    input [4:0] n,           // String length (1-16)
    input [15:0] s,          // Binary string (s[0] = first character)
    output reg [4:0] result  // Result: min(n, changes + 3)
);

integer i;
reg [4:0] changes;

always @(*) begin
    changes = 0;
    // Count adjacent differences
    for (i = 0; i < 15; i = i + 1) begin
        if (i < n - 1) begin
            if (s[i] != s[i + 1])
                changes = changes + 1;
        end
    end
    // Compute result = min(n, changes + 3)
    if (changes + 3 > n)
        result = n;
    else
        result = changes + 3;
end

endmodule
