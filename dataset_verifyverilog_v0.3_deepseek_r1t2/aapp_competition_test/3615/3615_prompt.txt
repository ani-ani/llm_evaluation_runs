module tower_coverage (
    input wire clk,
    input wire rst_n,
    input wire start,
    input wire [3:0] n,
    input wire [15:0] x0, y0, x1, y1, x2, y2, x3, y3,
    input wire [15:0] x4, y4, x5, y5, x6, y6, x7, y7,
    output reg [7:0] result,
    output reg done
);

// This module computes the maximum number of towers that can be in a connected
// subregion after adding one tower. The input coordinates are in Q8.8 format:
// integer value = actual_km * 256.
// The module assumes up to 8 towers.
// The algorithm:
// 1. Compute pairwise distances and build connected components using union-find.
// 2. For each pair of components, check if the minimum distance between any
//    tower in the two components is <= 4 km (i.e., squared distance <= 262144).
// 3. The maximum number of towers is either the largest component size + 1,
//    or the sum of two components that can be connected + 1.
// 4. Output the maximum.
//
// The module uses a state machine to perform the computation.
// The computation may take many cycles; done is asserted when finished.
// Reset: active low, resets all internal state.
// Start: pulse high to start computation.
//
// Implementation details:
// - Use union-find data structure with 8 elements.
// - Compute squared distances using 32-bit arithmetic.
// - Use parameters for thresholds.

parameter COORD_WIDTH = 16;
parameter SQ_DIST_THRESH = 262144; // 4 * 256 * 256 = 4 * 65536
parameter MAX_TOWERS = 8;
parameter MAX_RESULT = 9;

// State definitions
localparam S_IDLE = 0;
localparam S_RESET = 1;
localparam S_COMPUTE_DIST = 2;
localparam S_BUILD_COMPONENTS = 3;
localparam S_ASSIGN_COMPONENTS = 4;
localparam S_COMPUTE_MIN_DIST = 5;
localparam S_EVALUATE = 6;
localparam S_DONE = 7;

reg [2:0] state, next_state;

// Storage for coordinates
reg [COORD_WIDTH-1:0] x_reg [0:MAX_TOWERS-1];
reg [COORD_WIDTH-1:0] y_reg [0:MAX_TOWERS-1];

// Union-Find arrays
reg [2:0] parent [0:MAX_TOWERS-1];
reg [3:0] size   [0:MAX_TOWERS-1]; // component size

// Component assignment
reg [2:0] comp_id [0:MAX_TOWERS-1];
reg [3:0] comp_size [0:MAX_TOWERS-1]; // size per component
reg [3:0] num_components;

// Pairwise distance matrix (store only when needed)
reg [31:0] dist_sq [0:MAX_TOWERS-1][0:MAX_TOWERS-1];
reg edge_exists [0:MAX_TOWERS-1][0:MAX_TOWERS-1];

// Temporary variables for loops
reg [3:0] i, j, k, m, n_idx;
reg [31:0] dx, dy, dx_sq, dy_sq;
reg [31:0] min_dist_sq;
reg [7:0] current_max;
reg [7:0] temp_total;
reg found_root;

// Union-Find functions (implemented as tasks)
task find;
    input [2:0] x;
    output [2:0] root;
    begin
        root = x;
        while (parent[root] != root) begin
            root = parent[root];
        end
        // Path compression (optional, not implemented for simplicity)
    end
endtask

task union_sets;
    input [2:0] a, b;
    begin
        find(a, root_a);
        find(b, root_b);
        if (root_a != root_b) begin
            if (size[root_a] < size[root_b]) begin
                parent[root_a] = root_b;
                size[root_b] = size[root_b] + size[root_a];
            end else begin
                parent[root_b] = root_a;
                size[root_a] = size[root_a] + size[root_b];
            end
        end
    end
endtask

// State transition
always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        state <= S_IDLE;
        done <= 0;
        result <= 0;
    end else begin
        state <= next_state;
    end
end

// Next state logic
always @(*) begin
    case (state)
        S_IDLE: begin
            if (start) next_state = S_RESET;
            else next_state = S_IDLE;
        end
        S_RESET: next_state = S_COMPUTE_DIST;
        S_COMPUTE_DIST: begin
            // After finishing distance computation
            if (i >= n && j >= n) next_state = S_BUILD_COMPONENTS;
            else next_state = S_COMPUTE_DIST;
        end
        S_BUILD_COMPONENTS: begin
            // After processing all edges
            if (i >= n && j >= n) next_state = S_ASSIGN_COMPONENTS;
            else next_state = S_BUILD_COMPONENTS;
        end
        S_ASSIGN_COMPONENTS: next_state = S_COMPUTE_MIN_DIST;
        S_COMPUTE_MIN_DIST: begin
            // After checking all component pairs
            if (i >= num_components && j >= num_components) next_state = S_EVALUATE;
            else next_state = S_COMPUTE_MIN_DIST;
        end
        S_EVALUATE: next_state = S_DONE;
        S_DONE: next_state = S_DONE;
        default: next_state = S_IDLE;
    endcase
end

// Main computation logic (always block triggered by state changes)
always @(posedge clk) begin
    if (state == S_RESET) begin
        // Initialize coordinates from inputs
        x_reg[0] <= x0; y_reg[0] <= y0;
        x_reg[1] <= x1; y_reg[1] <= y1;
        x_reg[2] <= x2; y_reg[2] <= y2;
        x_reg[3] <= x3; y_reg[3] <= y3;
        x_reg[4] <= x4; y_reg[4] <= y4;
        x_reg[5] <= x5; y_reg[5] <= y5;
        x_reg[6] <= x6; y_reg[6] <= y6;
        x_reg[7] <= x7; y_reg[7] <= y7;
        n_idx <= n;
        // Initialize union-find
        for (i = 0; i < MAX_TOWERS; i = i + 1) begin
            parent[i] <= i;
            size[i] <= 1;
            comp_id[i] <= 0;
            comp_size[i] <= 0;
        end
        // Reset distance matrix flags
        for (i = 0; i < MAX_TOWERS; i = i + 1) begin
            for (j = 0; j < MAX_TOWERS; j = j + 1) begin
                edge_exists[i][j] <= 0;
                dist_sq[i][j] <= 0;
            end
        end
        i <= 0;
        j <= 0;
        num_components <= 0;
        current_max <= 0;
        temp_total <= 0;
    end

    else if (state == S_COMPUTE_DIST) begin
        // Compute pairwise distances and store edges
        if (i < n_idx) begin
            if (j < n_idx) begin
                if (i != j) begin
                    dx <= (x_reg[i] > x_reg[j]) ? (x_reg[i] - x_reg[j]) : (x_reg[j] - x_reg[i]);
                    dy <= (y_reg[i] > y_reg[j]) ? (y_reg[i] - y_reg[j]) : (y_reg[j] - y_reg[i]);
                    // Wait one cycle for multiplication (assumed combinational)
                    // In real hardware, we would need to break this into multiple cycles
                    // For simplicity, we assume multiplication can be done in one cycle
                    dx_sq <= dx * dx;
                    dy_sq <= dy * dy;
                    dist_sq[i][j] <= dx_sq + dy_sq;
                    if (dx_sq + dy_sq <= SQ_DIST_THRESH) begin
                        edge_exists[i][j] <= 1;
                        edge_exists[j][i] <= 1;
                    end
                end
                j <= j + 1;
            end else begin
                j <= 0;
                i <= i + 1;
            end
        end
    end

    else if (state == S_BUILD_COMPONENTS) begin
        // Process edges to union components
        if (i < n_idx) begin
            if (j < n_idx) begin
                if (edge_exists[i][j]) begin
                    // Perform union of i and j
                    union_sets(i, j);
                end
                j <= j + 1;
            end else begin
                j <= 0;
                i <= i + 1;
            end
        end
    end

    else if (state == S_ASSIGN_COMPONENTS) begin
        // Assign component IDs and compute sizes
        num_components <= 0;
        for (i = 0; i < n_idx; i = i + 1) begin
            // Find root of i
            find(i, root_i);
            // Check if this root has been assigned an ID
            found_root = 0;
            for (k = 0; k < num_components; k = k + 1) begin
                if (comp_id[k] == root_i) begin
                    comp_size[k] <= comp_size[k] + 1;
                    found_root = 1;
                end
            end
            if (!found_root) begin
                comp_id[num_components] <= root_i;
                comp_size[num_components] <= 1;
                num_components <= num_components + 1;
            end
        end
        i <= 0;
        j <= 0;
    end

    else if (state == S_COMPUTE_MIN_DIST) begin
        // Compute minimum distance between each pair of components
        // We store the minimum distance in a register min_dist_sq
        // For simplicity, we recompute for each pair
        if (i < num_components) begin
            if (j < num_components) begin
                if (i != j) begin
                    // Compute min distance between any tower in component i and any in component j
                    min_dist_sq <= 32'hFFFFFFFF; // Initialize to large value
                    for (m = 0; m < n_idx; m = m + 1) begin
                        for (k = 0; k < n_idx; k = k + 1) begin
                            // Check if m is in component i and k in component j
                            if (comp_id[m] == comp_id[i] && comp_id[k] == comp_id[j]) begin
                                if (dist_sq[m][k] < min_dist_sq) begin
                                    min_dist_sq <= dist_sq[m][k];
                                end
                            end
                        end
                    end
                    // If min_dist_sq <= SQ_DIST_THRESH, then components can be connected
                    if (min_dist_sq <= SQ_DIST_THRESH) begin
                        temp_total <= comp_size[i] + comp_size[j] + 1;
                        if (temp_total > current_max) current_max <= temp_total;
                    end
                end
                j <= j + 1;
            end else begin
                j <= 0;
                i <= i + 1;
            end
        end
    end

    else if (state == S_EVALUATE) begin
        // Also consider connecting a single component (size + 1)
        for (i = 0; i < num_components; i = i + 1) begin
            temp_total <= comp_size[i] + 1;
            if (temp_total > current_max) current_max <= temp_total;
        end
        result <= current_max;
    end

    else if (state == S_DONE) begin
        done <= 1;
    end
end

endmodule
